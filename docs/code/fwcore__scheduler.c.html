1/* 每輪先按可執行 VF 的需求分配 reserve，再按權重分配剩餘 credit；運算結果不代表 DMA 已完成。 */
2#include "fw_internal.h"
3
4fw_status fw_schedule_round(struct fw_context *ctx, const fw_u32 backlog[FW_VFS], fw_u32 service[FW_VFS])
5{
6    fw_u32 demand[FW_VFS], allocation[FW_VFS], remainders[FW_VFS];
7    fw_u32 left=FW_CAP_CREDIT, weights, floor_sum, available;
8    fw_u16 i, j, winner;
9    unsigned pass;
10    fw_u8 capped;
11    struct fw_hw_snapshot hw;
12    fw_status rc;
13    if (!ctx || !backlog || !service) return FW_INVAL;
14    if (ctx->round==FW_U64_MAX) return FW_OVERFLOW;
15    for (i=0; i<FW_VFS; ++i) {
16        rc=fw_hw(ctx,i,&hw); if (rc!=FW_OK) return rc;
17        demand[i]=(ctx->vf[i].state==FW_ACTIVE && !hw.fenced && !hw.killed && hw.bme &&
18            hw.config_valid && !hw.outstanding)?backlog[i]:0;
19        allocation[i]=demand[i]<ctx->vf[i].reserve?demand[i]:ctx->vf[i].reserve;
20        if (allocation[i]>left || (demand[i] && !ctx->vf[i].reserve)) return FW_STATE;
21        left-=allocation[i]; demand[i]-=allocation[i];
22    }
23    /* 至多移出八個 VF；沒有飽和者時一次 floor/remainder 即完成。 */
24    for (pass=0; pass<FW_VFS+1U && left; ++pass) {
25        weights=0;
26        for (i=0; i<FW_VFS; ++i) if (demand[i]) weights+=ctx->vf[i].reserve;
27        if (!weights) break;
28        capped=0; available=left;
29        for (i=0; i<FW_VFS; ++i) {
30            if (demand[i] && ((fw_u64)available*ctx->vf[i].reserve)/weights>=demand[i]) {
31                allocation[i]+=demand[i]; left-=demand[i]; demand[i]=0; capped=1;
32            }
33        }
34        if (capped) continue;
35        /* 無 VF 先達需求上限時，用整數商分配，再依餘數補足；同餘數輪替優先順序。 */
36        floor_sum=0;
37        for (i=0; i<FW_VFS; ++i) {
38            fw_u64 product=demand[i]?(fw_u64)left*ctx->vf[i].reserve:0;
39            fw_u32 whole=(fw_u32)(product/weights);
40            allocation[i]+=whole; demand[i]-=whole; floor_sum+=whole;
41            remainders[i]=(fw_u32)(product%weights);
42        }
43        left-=floor_sum;
44        for (j=0; j<FW_VFS && left; ++j) {
45            winner=FW_VFS;
46            for (i=0; i<FW_VFS; ++i) {
47                if (!demand[i] || !remainders[i]) continue;
48                if (winner==FW_VFS || remainders[i]>remainders[winner] ||
49                    (remainders[i]==remainders[winner] &&
50                     (i+FW_VFS-(fw_u32)(ctx->round%FW_VFS))%FW_VFS<
51                     (winner+FW_VFS-(fw_u32)(ctx->round%FW_VFS))%FW_VFS)) winner=i;
52            }
53            if (winner==FW_VFS) break;
54            ++allocation[winner]; --demand[winner]; --left; remainders[winner]=0;
55        }
56        break;
57    }
58    /* 先檢查全部累計值是否溢位，再發布 service，避免只更新一部分 VF。 */
59    for (i=0; i<FW_VFS; ++i)
60        if (ctx->vf[i].serviced>FW_U64_MAX-allocation[i]) return FW_OVERFLOW;
61    for (i=0; i<FW_VFS; ++i) { service[i]=allocation[i]; ctx->vf[i].serviced+=allocation[i]; }
62    ++ctx->round;
63    return FW_OK;
64}
