10static uint64_t read_vf(void *opaque, hwaddr addr, unsigned size)
11{
12    ToyVF *vf = opaque;
13    ToyPF *pf = vf->pf;
14    ToySlot *s = &pf->slots[vf->index];
15    if (size != 4 || (addr & 3)) {
16        toy_saturate(&pf->bad_mmio);
17        return 0;
18    }
19    switch (addr) {
20    case TOY_VF_MAGIC: return TOY_VF_MAGIC_VALUE;
21    case TOY_VF_ABI: return TOY_ABI_VERSION;
22    case TOY_VF_STATUS: return s->latch_status;
23    case TOY_VF_EPOCH_LO: return (uint32_t)s->latch_epoch;
24    case TOY_VF_EPOCH_HI: return s->latch_epoch >> 32;
25    case TOY_VF_SESSION_LO: return (uint32_t)s->latch_session;
26    case TOY_VF_SESSION_HI: return s->latch_session >> 32;
27    case TOY_VF_SQ_BASE_LO: return (uint32_t)s->staging.sq_base;
28    case TOY_VF_SQ_BASE_HI: return s->staging.sq_base >> 32;
29    case TOY_VF_SQ_ENTRIES: return s->staging.sq_entries;
30    case TOY_VF_CQ_BASE_LO: return (uint32_t)s->staging.cq_base;
31    case TOY_VF_CQ_BASE_HI: return s->staging.cq_base >> 32;
32    case TOY_VF_CQ_ENTRIES: return s->staging.cq_entries;
33    case TOY_VF_CONFIG_EPOCH_LO: return (uint32_t)s->staging.epoch;
34    case TOY_VF_CONFIG_EPOCH_HI: return s->staging.epoch >> 32;
35    case TOY_VF_CONFIG_SESSION_LO: return (uint32_t)s->staging.session;
36    case TOY_VF_CONFIG_SESSION_HI: return s->staging.session >> 32;
37    case TOY_VF_SQ_HEAD: return s->sq_head;
38    case TOY_VF_SQ_TAIL: return s->sq_tail;
39    case TOY_VF_CQ_HEAD: return s->cq_head;
40    case TOY_VF_CQ_TAIL: return s->cq_tail;
41    case TOY_VF_QUEUE_STATUS: return s->queue_status;
42    case TOY_VF_QUEUE_ERROR: return s->queue_error;
43    case TOY_VF_IRQ_CAUSE: return s->irq_cause;
44    default: return 0;
45    }
46}

48static void set_half(uint64_t *target, uint32_t value, bool high)
49{
50    if (high) {
51        *target = (*target & UINT32_MAX) | ((uint64_t)value << 32);
52    } else {
53        *target = (*target & ~((uint64_t)UINT32_MAX)) | value;
54    }
55}

57/* Staging 寄存器可分次寫入；COMMIT 驗證完整設定後才生效，並停在 READY_PENDING 等 PF REARM。 */
58static void config_commit(ToyPF *pf, unsigned index)
59{
60    ToySlot *s = &pf->slots[index];
61    fw_status result;
62    if (s->killed || !s->fenced || s->queue_status == TOY_QUEUE_STATUS_RUNNING ||
63        (pf->core.vf[index].state != FW_ACTIVE && pf->core.vf[index].state != FW_RECOVERING)) {
64        s->queue_error = FW_STATE;
65        return;
66    }
67    result = fw_queue_validate(&s->staging, pf->session, s->epoch);
68    if (result != FW_OK || s->ready_gen == UINT32_MAX) {
69        s->queue_error = result == FW_OK ? FW_OVERFLOW : result;
70        return;
71    }
72    s->config = s->staging;
73    s->sq_head = s->sq_tail = s->cq_head = s->cq_tail = 0;
74    s->ready_gen++;
75    s->config_valid = s->ready_pending = true;
76    s->queue_status = TOY_QUEUE_STATUS_READY_PENDING;
77    s->queue_error = FW_OK;
78}

80static void write_vf(void *opaque, hwaddr addr, uint64_t value, unsigned size)
81{
82    ToyVF *vf = opaque;
83    ToyPF *pf = vf->pf;
84    ToySlot *s = &pf->slots[vf->index];
85    uint32_t val = value;
86    if (size != 4 || (addr & 3)) {
87        toy_saturate(&pf->bad_mmio);
88        return;
89    }
90    switch (addr) {
91    /* 先 latch 再分次讀 32-bit halves，避免讀到不同時刻的 session/epoch。 */
92    case TOY_VF_EPOCH_LATCH:
93        if (val != 1) { break; }
94        s->latch_epoch = s->epoch;
95        s->latch_session = pf->session;
96        s->latch_status = toy_hw_flags(pf, vf->index);
97        return;
98    case TOY_VF_SQ_BASE_LO: set_half(&s->staging.sq_base, val, false); return;
99    case TOY_VF_SQ_BASE_HI: set_half(&s->staging.sq_base, val, true); return;
100    case TOY_VF_CQ_BASE_LO: set_half(&s->staging.cq_base, val, false); return;
101    case TOY_VF_CQ_BASE_HI: set_half(&s->staging.cq_base, val, true); return;
102    case TOY_VF_CONFIG_EPOCH_LO: set_half(&s->staging.epoch, val, false); return;
103    case TOY_VF_CONFIG_EPOCH_HI: set_half(&s->staging.epoch, val, true); return;
104    case TOY_VF_CONFIG_SESSION_LO: set_half(&s->staging.session, val, false); return;
105    case TOY_VF_CONFIG_SESSION_HI: set_half(&s->staging.session, val, true); return;
106    case TOY_VF_SQ_ENTRIES: s->staging.sq_entries = val; return;
107    case TOY_VF_CQ_ENTRIES: s->staging.cq_entries = val; return;
108    case TOY_VF_CONFIG_COMMIT:
109        if (val != 1) { break; }
110        config_commit(pf, vf->index);
111        return;
112    case TOY_VF_SQ_TAIL:
113        if (!s->config_valid || s->queue_status != TOY_QUEUE_STATUS_RUNNING) {
114            s->queue_error = FW_STATE;
115            return;
116        }
117        if ((uint32_t)(val - s->sq_head) > s->config.sq_entries ||
118            (uint32_t)(val - s->sq_tail) > s->config.sq_entries) {
119            toy_fail(pf, vf->index, TOY_LAST_FAULT_BAD_QUEUE);
120        } else {
121            s->sq_tail = val;
122        }
123        return;
124    case TOY_VF_CQ_HEAD:
125        if ((uint32_t)(val - s->cq_head) > (uint32_t)(s->cq_tail - s->cq_head)) {
126            toy_fail(pf, vf->index, TOY_LAST_FAULT_BAD_QUEUE);
127        } else {
128            s->cq_head = val;
129            toy_schedule(pf);
130        }
131        return;
132    case TOY_VF_DOORBELL:
133        if (val != 1) { break; }
134        if (s->fenced || s->killed) {
135            toy_saturate(&s->fence_blocked);
136        } else {
137            toy_schedule(pf);
138        }
139        return;
140    case TOY_VF_IRQ_CAUSE:
141        if (val & ~3u) { break; }
142        s->irq_cause &= ~val;
143        return;
144    default: break;
145    }
146    toy_saturate(&pf->bad_mmio);
147}

161static void vf_realize(PCIDevice *dev, Error **errp)
162{
163    ToyVF *vf = TOY_VF(dev);
164    int result;
165    vf->pf = TOY_PF(pcie_sriov_get_pf(dev));
166    vf->index = pcie_sriov_vf_number(dev);
167    if (vf->index >= FW_VFS) {
168        error_setg(errp, "Invalid VF index");
169        return;
170    }
171    dev->config_write = vf_config_write;
172    memory_region_init_io(&vf->bar0, OBJECT(vf), &toy_vf_mmio_ops, vf,
173                          "toy-vf-mmio", TOY_LIMITS_VF_BAR0_BYTES);
174    memory_region_init(&vf->bar2, OBJECT(vf), "toy-vf-msix", TOY_LIMITS_BAR2_BYTES);
175    pcie_sriov_vf_register_bar(dev, 0, &vf->bar0);
176    pcie_sriov_vf_register_bar(dev, 2, &vf->bar2);
177    result = pcie_endpoint_cap_init(dev, 0x80);
178    if (result < 0) {
179        error_setg(errp, "VF PCIe capability initialization failed");
180        return;
181    }
182    result = msix_init(dev, 2, &vf->bar2, 2, 0, &vf->bar2, 2, 0x1000, 0x50, errp);
183    if (result) {
184        pcie_cap_exit(dev);
185        return;
186    }
187    msix_vector_use(dev, 0);
188    msix_vector_use(dev, 1);
189    pcie_cap_flr_init(dev);
190    pcie_ari_init(dev, 0x100);
191    pcie_acs_init(dev, 0x120);
192    vf->pf->slots[vf->index].vf = vf;
193    vf->pf->slots[vf->index].fenced = true;
194    toy_clear_queue(vf->pf, vf->index);
195    fprintf(stderr, "TOY_VF_REALIZE vf=%u rid=%04x reserve=%u\n", vf->index, pci_get_bdf(dev),
196            vf->pf->core.vf[vf->index].reserve);
197}

199static void vf_exit(PCIDevice *dev)
200{
201    ToyVF *vf = TOY_VF(dev);
202    toy_fence(vf->pf, vf->index, 0, false);
203    toy_clear_queue(vf->pf, vf->index);
204    vf->pf->slots[vf->index].vf = NULL;
205    msix_unuse_all_vectors(dev);
206    msix_uninit(dev, &vf->bar2, &vf->bar2);
207    pcie_cap_exit(dev);
208    fprintf(stderr, "TOY_VF_UNREALIZE vf=%u reserve=%u\n", vf->index, vf->pf->core.vf[vf->index].reserve);
209}

211static void vf_reset(Object *obj, ResetType type)
212{
213    ToyVF *vf = TOY_VF(obj);
214    if (vf->pf && vf->pf->initialized) {
215        /* pinned qdev 在 realized=true 之前重設熱插入的新物件；這不是 PCI FLR。 */
216        if (!DEVICE(vf)->realized) {
217            fw_status status = toy_fence(vf->pf, vf->index, 0, false);
218            toy_clear_queue(vf->pf, vf->index);
219            if (status != FW_OK) {
220                toy_kill(vf->pf, vf->index);
221            }
222            fprintf(stderr, "TOY_VF_POWER_ON vf=%u epoch=%" PRIu64 " status=%u\n",
223                    vf->index, vf->pf->slots[vf->index].epoch, status);
224            return;
225        }
226        toy_vf_flr(vf->pf, vf->index);
227    }
228}

235static void vf_class_init(ObjectClass *klass, void *data)
236{
237    PCIDeviceClass *pc = PCI_DEVICE_CLASS(klass);
238    DeviceClass *dc = DEVICE_CLASS(klass);
239    pc->realize = vf_realize;
240    pc->exit = vf_exit;
241    pc->vendor_id = TOY_PCI_VENDOR;
242    pc->device_id = TOY_PCI_VF;
243    pc->revision = TOY_PCI_REVISION;
244    pc->class_id = TOY_PCI_CLASS_CODE >> 8;
245    RESETTABLE_CLASS(klass)->phases.hold = vf_reset;
246    dc->user_creatable = false;
247    dc->desc = "SR-IOV firmware lab VF";
248}
