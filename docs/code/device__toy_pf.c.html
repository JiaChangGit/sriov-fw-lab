11static uint32_t read_slot(ToyPF *pf, unsigned vf, uint32_t reg)
12{
13    ToyView *s = &pf->views[vf];
14    struct fw_vf *snap = &pf->snapshot.vf[vf];
15    switch (reg) {
16    case TOY_PF_VF_VIEW_HW_STATUS: return s->hw_flags;
17    case TOY_PF_VF_VIEW_EPOCH_LO: return (uint32_t)pf->snapshot.hw[vf].epoch;
18    case TOY_PF_VF_VIEW_EPOCH_HI: return pf->snapshot.hw[vf].epoch >> 32;
19    case TOY_PF_VF_VIEW_STATE: return snap->state;
20    case TOY_PF_VF_VIEW_RESERVE: return snap->reserve;
21    case TOY_PF_VF_VIEW_OUTSTANDING: return s->outstanding;
22    case TOY_PF_VF_VIEW_STALE_DESC: return s->stale_desc;
23    case TOY_PF_VF_VIEW_STALE_CPL: return s->stale_cpl;
24    case TOY_PF_VF_VIEW_FENCE_BLOCKED: return s->fence_blocked;
25    case TOY_PF_VF_VIEW_DMA_FAULT: return s->dma_fault;
26    case TOY_PF_VF_VIEW_LAST_FAULT: return s->last_fault;
27    case TOY_PF_VF_VIEW_SERVICED_LO: return (uint32_t)snap->serviced;
28    case TOY_PF_VF_VIEW_SERVICED_HI: return snap->serviced >> 32;
29    case TOY_PF_VF_VIEW_IRQ_SUPPRESSED: return s->irq_suppressed;
30    case TOY_PF_VF_VIEW_READY_GEN: return s->ready_gen;
31    default: return 0;
32    }
33}

35/* Driver 持鎖發出 SNAPSHOT 後，後續寄存器讀取使用同一份快照。 */
36fw_status toy_snapshot(ToyPF *pf)
37{
38    struct fw_snapshot core;
39    ToyView views[FW_VFS];
40    fw_status rc=fw_snapshot(&pf->core, &core);
41    if (rc!=FW_OK) { return rc; }
42    /* 全部在同一 BQL 區段取樣，成功後才一次提交；MMIO 讀取不再混入 live 值。 */
43    for (unsigned i=0; i<FW_VFS; i++) {
44        ToySlot *s=&pf->slots[i];
45        views[i]=(ToyView) {
46            .hw_flags=toy_hw_flags(pf,i), .outstanding=s->pending.valid,
47            .stale_desc=s->stale_desc, .stale_cpl=s->stale_cpl,
48            .fence_blocked=s->fence_blocked, .dma_fault=s->dma_fault,
49            .last_fault=s->last_fault, .irq_suppressed=s->irq_suppressed,
50            .ready_gen=s->ready_gen,
51        };
52    }
53    pf->snapshot=core;
54    memcpy(pf->views,views,sizeof(views));
55    return FW_OK;
56}

58static uint64_t read_pf(void *opaque, hwaddr addr, unsigned size)
59{
60    ToyPF *pf = opaque;
61    if (size != 4 || (addr & 3)) {
62        toy_saturate(&pf->bad_mmio);
63        return 0;
64    }
65    if (addr >= TOY_PF_VF_BASE && addr < TOY_PF_VF_BASE + FW_VFS * TOY_PF_VF_STRIDE) {
66        return read_slot(pf, (addr - TOY_PF_VF_BASE) / TOY_PF_VF_STRIDE,
67                         (addr - TOY_PF_VF_BASE) % TOY_PF_VF_STRIDE);
68    }
69    if (addr >= TOY_PF_MBX_ARG0 && addr <= TOY_PF_MBX_ARG3) {
70        return pf->mbx_staging.arg[(addr - TOY_PF_MBX_ARG0) / 4];
71    }
72    if (addr >= TOY_PF_MBX_REPLY0 && addr <= TOY_PF_MBX_REPLY3) {
73        return pf->mbx_reply.value[(addr - TOY_PF_MBX_REPLY0) / 4];
74    }
75    switch (addr) {
76    case TOY_PF_MAGIC: return TOY_PF_MAGIC_VALUE;
77    case TOY_PF_ABI: return TOY_ABI_VERSION;
78    case TOY_PF_CAP_VFS: return FW_VFS;
79    case TOY_PF_CAP_QUEUES: return FW_VFS;
80    case TOY_PF_CAP_VECTORS: return 16;
81    case TOY_PF_CAP_CREDIT: return FW_CAP_CREDIT;
82    case TOY_PF_CAP_FEATURES: return TOY_FEATURES_DMA_FENCE | TOY_FEATURES_VF_KILL | TOY_FEATURES_MODELED_FLR;
83    case TOY_PF_SESSION_LO: return (uint32_t)pf->snapshot.session;
84    case TOY_PF_SESSION_HI: return pf->snapshot.session >> 32;
85    case TOY_PF_EVENT: return pf->event;
86    case TOY_PF_BAD_MMIO: return pf->bad_mmio;
87    case TOY_PF_FENCE_TOKEN: return pf->fence_token;
88    case TOY_PF_HW_RESULT: return pf->hw_result;
89    case TOY_PF_MBX_SEQ: return pf->mbx_staging.seq;
90    case TOY_PF_MBX_OPCODE: return pf->mbx_staging.opcode;
91    case TOY_PF_MBX_TARGET: return pf->mbx_staging.target;
92    case TOY_PF_MBX_STATUS: return pf->mbx_status;
93    case TOY_PF_MBX_RESULT: return pf->mbx_reply.status;
94    case TOY_PF_MBX_ACK_SEQ: return pf->mbx_reply.seq;
95    default: return 0;
96    }
97}

99/* 發布回覆與通知；故障注入可丟一次回覆，用來檢查原 request 的重送行為。 */
100static void mailbox_complete(ToyPF *pf)
101{
102    if (pf->drop_mbx_reply_once) {
103        pf->drop_mbx_reply_once = false;
104        pf->mbx_status = TOY_MBX_STATUS_IDLE;
105        fprintf(stderr, "TOY_FAULT_ACK kind=drop_mbx_reply_once seq=%u\n", pf->mbx_reply.seq);
106        memset(&pf->mbx_reply, 0, sizeof(pf->mbx_reply));
107        return;
108    }
109    pf->mbx_status = TOY_MBX_STATUS_DONE;
110    pf->event |= TOY_PF_EVENTS_MBX_DONE;
111    msix_notify(PCI_DEVICE(pf), 0);
112}

114void toy_mbx_poll(ToyPF *pf)
115{
116    unsigned vf=pf->mbx_current.target;
117    if (!pf->core.mailbox_pending || pf->fw_hang_all ||
118        (vf<FW_VFS && pf->fw_hang_vf && toy_fault_target(pf,vf))) { return; }
119    if (fw_poll_mailbox(&pf->core,&pf->mbx_reply)!=FW_BUSY) { mailbox_complete(pf); }
120}

122void toy_mbx_work(void *opaque)
123{
124    ToyPF *pf = opaque;
125    struct fw_request *req = &pf->mbx_current;
126    /* BH 已取出；故障注入可能保留 BUSY，但不代表仍有排程中的 BH。 */
127    pf->mbx_pending = false;
128    if (pf->fw_hang_all || (pf->fw_hang_vf && req->target < FW_VFS && toy_fault_target(pf, req->target))) {
129        fprintf(stderr, "TOY_FAULT_ACK kind=mailbox_hang vf=%u seq=%u\n", req->target, req->seq);
130        return;
131    }
132    fw_handle_mailbox(&pf->core, req, &pf->mbx_reply);
133    toy_schedule(pf);
134    if (!pf->core.mailbox_pending) { mailbox_complete(pf); }
135}

137static void write_pf(void *opaque, hwaddr addr, uint64_t value, unsigned size)
138{
139    ToyPF *pf = opaque;
140    uint32_t val = value;
141    if (size != 4 || (addr & 3)) {
142        toy_saturate(&pf->bad_mmio);
143        return;
144    }
145    if (addr >= TOY_PF_VF_BASE && addr < TOY_PF_VF_BASE + FW_VFS * TOY_PF_VF_STRIDE) {
146        unsigned vf = (addr - TOY_PF_VF_BASE) / TOY_PF_VF_STRIDE;
147        unsigned reg = (addr - TOY_PF_VF_BASE) % TOY_PF_VF_STRIDE;
148        if (reg != TOY_PF_VF_VIEW_HW_CTL || !val || (val & ~3u)) {
149            toy_saturate(&pf->bad_mmio);
150            pf->hw_result = FW_INVAL;
151            return;
152        }
153        pf->hw_result = FW_OK;
154        if (val & TOY_VF_HW_CONTROL_FENCE_BEGIN) {
155            pf->hw_result = toy_fence(pf, vf, pf->fence_token, true);
156        }
157        if (val & TOY_VF_HW_CONTROL_KILL) {
158            toy_kill(pf, vf);
159            toy_schedule(pf);
160        }
161        return;
162    }
163    if (addr >= TOY_PF_MBX_ARG0 && addr <= TOY_PF_MBX_ARG3) {
164        pf->mbx_staging.arg[(addr - TOY_PF_MBX_ARG0) / 4] = val;
165        return;
166    }
167    if (addr >= TOY_CAPSULE_STAGE_OFFSET && addr < TOY_CAPSULE_STAGE_OFFSET + TOY_CAPSULE_STAGE_LENGTH) {
168        pf->hw_result = FW_UNSUP;
169        return;
170    }
171    switch (addr) {
172    case TOY_PF_SNAPSHOT:
173        if (val != 1) { break; }
174        pf->hw_result = toy_snapshot(pf);
175        return;
176    case TOY_PF_EVENT:
177        if (val & ~15u) { break; }
178        pf->event &= ~val;
179        return;
180    case TOY_PF_CONTROL:
181        if (!val || (val & ~3u)) { break; }
182        if (val & TOY_PF_CONTROL_FENCE_ALL) {
183            pf->hw_result = FW_OK;
184            for (unsigned i = 0; i < FW_VFS; i++) {
185                fw_status status = toy_fence(pf, i, 0, false);
186                if (status != FW_OK) { pf->hw_result = status; }
187            }
188        }
189        if (val & TOY_PF_CONTROL_GLOBAL_RESET) {
190            pf->hw_result = toy_global_reset(pf);
191        }
192        return;
193    case TOY_PF_FENCE_TOKEN: pf->fence_token = val; return;
194    case TOY_PF_MBX_SEQ: pf->mbx_staging.seq = val; return;
195    case TOY_PF_MBX_OPCODE: pf->mbx_staging.opcode = val; return;
196    case TOY_PF_MBX_TARGET: pf->mbx_staging.target = val; return;
197    case TOY_PF_MBX_SUBMIT:
198        if (val != 1) { break; }
199        if (pf->mbx_status == TOY_MBX_STATUS_BUSY) {
200            pf->hw_result = FW_BUSY;
201            return;
202        }
203        pf->mbx_current = pf->mbx_staging;
204        pf->mbx_current.session = pf->session;
205        pf->mbx_status = TOY_MBX_STATUS_BUSY;
206        pf->mbx_pending = true;
207        qemu_bh_schedule(pf->mbx_bh);
208        return;
209    case TOY_PF_MBX_ACK:
210        if (val != 1) { break; }
211        if (pf->mbx_status == TOY_MBX_STATUS_DONE) {
212            pf->mbx_status = TOY_MBX_STATUS_IDLE;
213        }
214        return;
215    default: break;
216    }
217    toy_saturate(&pf->bad_mmio);
218}

232static void pf_realize(PCIDevice *dev, Error **errp)
233{
234    ToyPF *pf = TOY_PF(dev);
235    int result;
236    if (pf->total_vfs != FW_VFS) {
237        error_setg(errp, "total-vfs must be 8 for ABI 1.0");
238        return;
239    }
240    dev->config_write = pf_config_write;
241    memory_region_init_io(&pf->bar0, OBJECT(pf), &pf_mmio_ops, pf, "toy-pf-mmio", TOY_LIMITS_PF_BAR0_BYTES);
242    pci_register_bar(dev, 0, PCI_BASE_ADDRESS_MEM_TYPE_64, &pf->bar0);
243    memory_region_init(&pf->bar2, OBJECT(pf), "toy-pf-msix", TOY_LIMITS_BAR2_BYTES);
244    pci_register_bar(dev, 2, PCI_BASE_ADDRESS_SPACE_MEMORY, &pf->bar2);
245    result = pcie_endpoint_cap_init(dev, 0x80);
246    if (result < 0) {
247        error_setg(errp, "PF PCIe capability initialization failed");
248        return;
249    }
250    result = msix_init(dev, 2, &pf->bar2, 2, 0, &pf->bar2, 2, 0x1000, 0x50, errp);
251    if (result) {
252        pcie_cap_exit(dev);
253        return;
254    }
255    msix_vector_use(dev, 0);
256    msix_vector_use(dev, 1);
257    pcie_cap_flr_init(dev);
258    pcie_ari_init(dev, 0x100);
259    pcie_sriov_pf_init(dev, 0x140, TYPE_TOY_VF, TOY_PCI_VF, FW_VFS, FW_VFS, 1, 1);
260    pcie_sriov_pf_init_vf_bar(dev, 0, PCI_BASE_ADDRESS_MEM_TYPE_64, TOY_LIMITS_VF_BAR0_BYTES);
261    pcie_sriov_pf_init_vf_bar(dev, 2, PCI_BASE_ADDRESS_SPACE_MEMORY, TOY_LIMITS_BAR2_BYTES);
262    /* 沒有 function-to-function P2P；endpoint ACS optional bits 為 0。 */
263    /* v9.2.0 的 SR-IOV reset 會以 quad 清至 0x183；保留間距避免覆蓋 ACS。 */
264    pcie_acs_init(dev, 0x200);
265    pf->session = 1;
266    for (unsigned i = 0; i < FW_VFS; i++) {
267        pf->slots[i].epoch = pf->slots[i].token = 1;
268        pf->slots[i].fenced = true;
269    }
270    toy_adapter_init(pf);
271    pf->worker = timer_new_ms(QEMU_CLOCK_VIRTUAL, toy_worker, pf);
272    pf->mbx_bh = qemu_bh_new(toy_mbx_work, pf);
273    pf->initialized = true;
274    pf->hw_result = toy_snapshot(pf);
275}

277static void pf_reset(Object *obj, ResetType type)
278{
279    ToyPF *pf = TOY_PF(obj);
280    if (!pf->initialized) { return; }
281    /* machine/PF reset 先消除 PCI presentation，再做全域 session transaction。 */
282    for (unsigned i = 0; i < FW_VFS; i++) { toy_kill(pf, i); }
283    pcie_sriov_pf_reset(PCI_DEVICE(pf));
284    qemu_bh_cancel(pf->mbx_bh);
285    pf->mbx_pending = false;
286    timer_del(pf->worker);
287    pf->mbx_status = TOY_MBX_STATUS_IDLE;
288    pf->event = 0;
289    pf->hw_result = toy_global_reset(pf);
290    if (toy_snapshot(pf) != FW_OK) {
291        pf->hw_result = FW_IO;
292    }
293}

295static void pf_exit(PCIDevice *dev)
296{
297    ToyPF *pf = TOY_PF(dev);
298    if (pf->initialized) {
299        for (unsigned i = 0; i < FW_VFS; i++) { toy_kill(pf, i); }
300        timer_del(pf->worker);
301        timer_free(pf->worker);
302        pf->worker = NULL;
303        qemu_bh_cancel(pf->mbx_bh);
304        pf->mbx_pending = false;
305        qemu_bh_delete(pf->mbx_bh);
306        pf->mbx_bh = NULL;
307        pcie_sriov_pf_exit(dev);
308        pf->initialized = false;
309    }
310    msix_unuse_all_vectors(dev);
311    msix_uninit(dev, &pf->bar2, &pf->bar2);
312    pcie_cap_exit(dev);
313}

315static void telemetry_get(Object *obj, Visitor *v, const char *name, void *opaque, Error **errp)
316{
317    ToyPF *pf = TOY_PF(obj);
318    uint32_t value = 0;
319    /* 只讀實際物件與工作狀態，不透過 fault setter 也不更動裝置。 */
320    if (!strcmp(name, "live-vfs")) {
321        for (unsigned i = 0; i < FW_VFS; i++) {
322            value += pf->slots[i].vf != NULL;
323        }
324    } else if (!strcmp(name, "pending-completions")) {
325        for (unsigned i = 0; i < FW_VFS; i++) {
326            value += pf->slots[i].pending.valid;
327        }
328    } else if (!strcmp(name, "worker-pending")) {
329        value = pf->worker && timer_pending(pf->worker);
330    } else if (!strcmp(name, "mailbox-pending")) {
331        value = pf->mbx_pending;
332    }
333    visit_type_uint32(v, name, &value, errp);
334}

416static const char *checkpoint_names[FW_CP_COUNT] = {
417    "provision.before_create", "provision.after_create", "provision.before_commit",
418    "provision.abort_release", "destroy.release",
419};

421fw_status toy_checkpoint(ToyPF *pf, unsigned vf, enum fw_checkpoint point)
422{
423    for (unsigned i = 0; i < ARRAY_SIZE(pf->failpoints); i++) {
424        ToyFailpoint *slot = &pf->failpoints[i];
425        if (slot->point != point || slot->fired ||
426            (slot->vf != UINT32_MAX && slot->vf != vf)) {
427            continue;
428        }
429        toy_saturate(&slot->seen);
430        if (slot->seen == slot->hit) {
431            slot->fired = 1;
432            fprintf(stderr, "TOY_FAILPOINT_HIT slot=%s name=%s vf=%u hit=%u\n",
433                    i ? "unwind" : "forward", checkpoint_names[point], vf, slot->hit);
434            return FW_IO;
435        }
436    }
437    return FW_OK;
438}

440static void failpoint_get(Object *obj, Visitor *v, const char *name, void *opaque, Error **errp)
441{
442    ToyFailpoint *slot = &TOY_PF(obj)->failpoints[(uintptr_t)opaque];
443    const char *field = strrchr(name, '-') + 1;
444    uint32_t value;

446        char *value = (char *)(slot->point < 0 ? "" : checkpoint_names[slot->point]);
447        visit_type_str(v, name, &value, errp);
448        return;
449    }
450    value = !strcmp(field, "hit") ? slot->hit : !strcmp(field, "vf") ? slot->vf :

452    visit_type_uint32(v, name, &value, errp);

455static void failpoint_set(Object *obj, Visitor *v, const char *name, void *opaque, Error **errp)
456{
457    ToyPF *pf = TOY_PF(obj);
458    ToyFailpoint *slot = &pf->failpoints[(uintptr_t)opaque];
459    const char *field = strrchr(name, '-') + 1;
460    if (!strcmp(field, "name")) {
461        char *value = NULL;
462        int point = -1;
463        if (!visit_type_str(v, name, &value, errp)) { return; }
464        for (unsigned i = 0; i < FW_CP_COUNT; i++) {
465            if (!strcmp(value, checkpoint_names[i])) { point = i; }
466        }
467        if (*value && point < 0) {
468            error_setg(errp, "Unknown named failpoint: %s", value);
469            g_free(value);
470            return;
471        }
472        g_free(value);
473        slot->point = point;
474        slot->seen = slot->fired = 0;
475    } else {
476        uint32_t value;
477        if (!visit_type_uint32(v, name, &value, errp)) { return; }
478        if (!strcmp(field, "hit")) {
479            if (!value) { error_setg(errp, "failpoint-hit must be nonzero"); return; }
480            slot->hit = value;
481        } else {
482            if (value >= FW_VFS && value != UINT32_MAX) {
483                error_setg(errp, "failpoint-vf must be 0..7 or UINT32_MAX"); return;
484            }
485            slot->vf = value;
486        }
487    }
488    toy_saturate(&pf->fault_ack);
489    fprintf(stderr, "TOY_FAULT_ACK property=%s ack=%u\n", name, pf->fault_ack);
490}

492static void pf_instance_init(Object *obj)
493{
494    ToyPF *pf = TOY_PF(obj);
495    static const struct { const char *name; size_t offset; } boolean[] = {
496        { "fw-hang-vf", offsetof(ToyPF, fw_hang_vf) },
497        { "fw-hang-all", offsetof(ToyPF, fw_hang_all) },
498        { "drop-mbx-reply-once", offsetof(ToyPF, drop_mbx_reply_once) },
499        { "ignore-quiesce", offsetof(ToyPF, ignore_quiesce) },
500        { "ignore-reset", offsetof(ToyPF, ignore_reset) },
501    };
502    const char *numbers[] = { "fault-vf", "fault-after", "delay-cpl-ticks" };
503    const char *telemetry[] = { "live-vfs", "pending-completions", "worker-pending", "mailbox-pending" };
504    for (unsigned i = 0; i < ARRAY_SIZE(boolean); i++) {
505        object_property_add(obj, boolean[i].name, "bool", bool_get, bool_set, NULL,
506                            (void *)(uintptr_t)boolean[i].offset);
507    }
508    for (unsigned i = 0; i < ARRAY_SIZE(numbers); i++) {
509        object_property_add(obj, numbers[i], "uint32", fault_get, fault_set, NULL, NULL);
510    }
511    for (unsigned i = 0; i < ARRAY_SIZE(telemetry); i++) {
512        object_property_add(obj, telemetry[i], "uint32", telemetry_get, NULL, NULL, NULL);
513    }
514    object_property_add(obj, "fault-ack", "uint32", fault_get, NULL, NULL, NULL);
515    object_property_add(obj, "virtual-tick", "uint64", virtual_tick_get, NULL, NULL, NULL);
516    object_property_add_bool(obj, "try-stale-cpl", trigger_get, stale_set);
517    object_property_add_bool(obj, "try-irq-fenced", trigger_get, irq_set);
518    for (unsigned i = 0; i < ARRAY_SIZE(pf->failpoints); i++) {
519        const char *fields[] = { "name", "hit", "vf", "seen", "fired" };
520        pf->failpoints[i] = (ToyFailpoint){ .point = -1, .hit = 1, .vf = UINT32_MAX };
521        for (unsigned j = 0; j < ARRAY_SIZE(fields); j++) {
522            g_autofree char *name = g_strdup_printf("%sfailpoint-%s", i ? "unwind-" : "", fields[j]);
523            object_property_add(obj, name, j ? "uint32" : "str", failpoint_get,
524                                j < 3 ? failpoint_set : NULL, NULL, (void *)(uintptr_t)i);
525        }
526    }
527}
