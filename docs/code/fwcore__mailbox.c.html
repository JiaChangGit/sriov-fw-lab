4static fw_u8 same_request(const struct fw_request *a, const struct fw_request *b)
5{
6    unsigned i;
7    if (a->session!=b->session || a->seq!=b->seq || a->opcode!=b->opcode || a->target!=b->target) return 0;
8    for (i=0; i<4; ++i) if (a->arg[i]!=b->arg[i]) return 0;
9    return 1;
10}

12static fw_status shape(const struct fw_context *ctx, const struct fw_request *req)
13{
14    unsigned i, used=0;
15    if (req->opcode<FW_VERSION_NEG || req->opcode>FW_GLOBAL_RESET) return FW_UNSUP;
16    if (req->opcode==FW_VERSION_NEG) {
17        if (req->target!=FW_GLOBAL || req->arg[3] || req->arg[1]>req->arg[2]) return FW_INVAL;
18        if (req->arg[0]!=1 || req->arg[1]>0) return FW_UNSUP;
19        return FW_OK;
20    }
21    if (!ctx->negotiated) return FW_NO_NEG;
22    if (req->opcode==FW_GLOBAL_RESET) {
23        if (req->target!=FW_GLOBAL) return FW_INVAL;
24        if (!ctx->hal.begin_global || !ctx->hal.end_global) return FW_UNSUP;
25    } else if (req->target>=FW_VFS) return FW_INVAL;
26    if (req->opcode==FW_PROVISION || req->opcode==FW_SET_RESERVE) {
27        used=1;
28        if (!req->arg[0] || req->arg[0]>FW_CAP_CREDIT) return FW_INVAL;
29    }
30    if (req->opcode==FW_REARM) { used=3; if (!req->arg[2]) return FW_INVAL; }
31    for (i=used; i<4; ++i) if (req->arg[i]) return FW_INVAL;
32    return FW_OK;
33}

35static fw_status run_checkpoint(struct fw_context *ctx, fw_u16 vf, enum fw_checkpoint point)
36{
37    return ctx->hal.checkpoint ? ctx->hal.checkpoint(ctx->hal.opaque,vf,point) : FW_OK;
38}

40/* 釋放失敗時走 FAILED 並保留配額，避免後續配置重用尚未確認回收的資源。 */
41static void rollback(struct fw_context *ctx, fw_u16 vf, fw_u8 destroying)
42{
43    fw_status rc=run_checkpoint(ctx,vf,destroying?FW_CP_DESTROY_RELEASE:FW_CP_ABORT_RELEASE);
44    if (rc==FW_OK) rc=fw_vf_event(ctx,vf,destroying?FW_RELEASE_OK:FW_ABORT_OK);
45    fw_note(ctx,destroying?"destroy_release_status":"provision_abort_status",vf,rc);
46    if (rc!=FW_OK) {
47        rc=fw_vf_event(ctx,vf,destroying?FW_RELEASE_FAIL:FW_ABORT_FAIL);
48        fw_note(ctx,"rollback_quarantine_status",vf,rc);
49    }
50}

52static fw_status progress(struct fw_context *ctx, fw_u16 vf, fw_u32 opcode,
53                          fw_u64 epoch, fw_u64 deadline)
54{
55    enum fw_state expected=opcode==FW_QUIESCE?FW_QUIESCING:FW_RESETTING;
56    fw_status rc;
57    if (ctx->vf[vf].epoch!=epoch) return FW_STATE;
58    if (ctx->vf[vf].state==FW_FAILED && ctx->hal.now_tick(ctx->hal.opaque)>=deadline) return FW_TIMEOUT;
59    if (ctx->vf[vf].state!=expected) return FW_STATE;
60    /* 期限優先於晚到的完成回呼，100 tick 的邊界不依 host timeout。 */
61    if (ctx->hal.now_tick(ctx->hal.opaque)>=deadline) {
62        rc=fw_vf_event(ctx,vf,FW_EV_TIMEOUT);
63        return rc==FW_OK?FW_TIMEOUT:rc;
64    }
65    rc=ctx->hal.mailbox_progress?ctx->hal.mailbox_progress(ctx->hal.opaque,vf,opcode):FW_OK;
66    if (rc!=FW_OK) return rc;
67    return fw_vf_event(ctx,vf,opcode==FW_QUIESCE?FW_DRAIN_OK:FW_RESET_DONE);
68}

70fw_status fw_poll_mailbox(struct fw_context *ctx, struct fw_reply *reply)
71{
72    fw_status rc;
73    if (!ctx || !reply) return FW_INVAL;
74    fw_zero(reply,sizeof(*reply));
75    if (!ctx->mailbox_pending) return FW_STATE;
76    rc=progress(ctx,(fw_u16)ctx->cached_request.target,ctx->cached_request.opcode,
77                ctx->pending_epoch,ctx->pending_deadline);
78    ctx->cached_reply.status=rc;
79    *reply=ctx->cached_reply;
80    if (rc!=FW_BUSY) {
81        ctx->mailbox_pending=0; ctx->pending_epoch=0; ctx->pending_deadline=0;
82        fw_note(ctx,"mailbox_completed",(fw_u16)ctx->cached_request.target,rc);
83    }
84    return rc;
85}

87fw_status fw_handle_mailbox(struct fw_context *ctx, const struct fw_request *req, struct fw_reply *reply)
88{
89    fw_status rc;
90    fw_u16 vf;
91    struct fw_hw_snapshot hw;
92    fw_u64 session, epochs[FW_VFS];
93    if (!ctx || !req || !reply) return FW_INVAL;
94    fw_zero(reply,sizeof(*reply)); reply->seq=req->seq;
95    if (!req->seq) { reply->status=FW_INVAL; return FW_INVAL; }
96    if (req->session!=ctx->session) { reply->status=FW_STALE_SEQ; return FW_STALE_SEQ; }
97    /* 重送必須逐欄相同；同序號卻改內容是衝突，不能再次執行副作用。 */
98    if (ctx->cache_valid && req->seq==ctx->last_seq) {
99        if (!same_request(req,&ctx->cached_request)) { reply->status=FW_SEQ_CONFLICT; return FW_SEQ_CONFLICT; }
100        *reply=ctx->cached_reply; return reply->status;
101    }
102    if (req->seq<=ctx->last_seq) { reply->status=FW_STALE_SEQ; return FW_STALE_SEQ; }
103    rc=shape(ctx,req);
104    if (rc!=FW_OK) { reply->status=rc; return rc; }
105    /* 尚在等 quiesce/reset 的交易由 poll 推進；新 request 不可蓋掉它的 cache。 */
106    if (ctx->mailbox_pending) { reply->status=FW_BUSY; return FW_BUSY; }
107    vf=(fw_u16)req->target;
108    switch (req->opcode) {
109    case FW_VERSION_NEG:
110        ctx->negotiated=1; reply->value[0]=1U<<16; rc=FW_OK;
111        break;
112    case FW_GET_STATE:
113        rc=fw_hw(ctx,vf,&hw);
114        if (rc==FW_OK) {
115            reply->value[0]=(fw_u32)ctx->vf[vf].state;
116            reply->value[1]=(fw_u32)hw.epoch; reply->value[2]=(fw_u32)(hw.epoch>>32);
117            reply->value[3]=(fw_u32)hw.fenced|((fw_u32)hw.killed<<1)|((fw_u32)hw.pci_present<<2)|
118                ((fw_u32)hw.bme<<3)|((fw_u32)hw.config_valid<<4)|((fw_u32)hw.ready_pending<<5);
119        }
120        break;
121    case FW_PROVISION:
122        rc=run_checkpoint(ctx,vf,FW_CP_BEFORE_CREATE);
123        if (rc==FW_OK) rc=fw_create(ctx,vf,req->arg[0]);
124        if (rc==FW_OK) {
125            rc=run_checkpoint(ctx,vf,FW_CP_AFTER_CREATE);
126            if (rc==FW_OK) rc=run_checkpoint(ctx,vf,FW_CP_BEFORE_COMMIT);
127            if (rc==FW_OK) rc=fw_vf_event(ctx,vf,FW_COMMIT);
128            if (rc!=FW_OK) rollback(ctx,vf,0);
129        }
130        break;
131    case FW_QUIESCE:
132        rc=fw_vf_event(ctx,vf,FW_FENCE);
133        if (rc==FW_OK) rc=progress(ctx,vf,req->opcode,ctx->vf[vf].epoch,ctx->vf[vf].deadline);
134        break;
135    case FW_RESET:
136        rc=progress(ctx,vf,req->opcode,ctx->vf[vf].epoch,ctx->vf[vf].deadline);
137        break;
138    case FW_REARM:
139        rc=fw_rearm(ctx,vf,(fw_u64)req->arg[0]|((fw_u64)req->arg[1]<<32),req->arg[2]);
140        break;
141    case FW_DESTROY_CMD:
142        rc=fw_vf_event(ctx,vf,FW_DESTROY);
143        if (rc==FW_OK) {
144            rc=run_checkpoint(ctx,vf,FW_CP_DESTROY_RELEASE);
145            if (rc==FW_OK) rc=fw_vf_event(ctx,vf,FW_RELEASE_OK);
146            fw_note(ctx,"destroy_release_status",vf,rc);
147            if (rc!=FW_OK) {
148                fw_status failed=fw_vf_event(ctx,vf,FW_RELEASE_FAIL);
149                fw_note(ctx,"rollback_quarantine_status",vf,failed);
150            }
151        }
152        break;
153    case FW_SET_RESERVE: rc=fw_set_reserve(ctx,vf,req->arg[0]); break;
154    case FW_GLOBAL_RESET:
155        if (!ctx->hal.owners_closed(ctx->hal.opaque,FW_GLOBAL)) { rc=FW_STATE; break; }
156        rc=ctx->hal.begin_global(ctx->hal.opaque,&session,epochs);
157        if (rc!=FW_OK) break;
158        rc=fw_global_recover(ctx,session,epochs);
159        if (rc!=FW_OK) { reply->status=rc; return rc; }
160        rc=ctx->hal.end_global(ctx->hal.opaque);
161        reply->status=rc; reply->value[0]=(fw_u32)session; reply->value[1]=(fw_u32)(session>>32);
162        /* 舊 session 的 transaction 不能重新污染剛清空的協商與 cache。 */
163        return rc;
164    default: rc=FW_UNSUP; break;
165    }
166    reply->status=rc; ctx->last_seq=req->seq; ctx->cache_valid=1;
167    ctx->cached_request=*req; ctx->cached_reply=*reply;
168    if (rc==FW_BUSY && (req->opcode==FW_QUIESCE || req->opcode==FW_RESET)) {
169        ctx->mailbox_pending=1;
170        ctx->pending_epoch=ctx->vf[vf].epoch; ctx->pending_deadline=ctx->vf[vf].deadline;
171    }
172    return rc;
173}
