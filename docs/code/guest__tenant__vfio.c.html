18int lab_write_text(const char *path, const char *text)
19{
20    size_t length=strlen(text);
21    int fd=open(path,O_WRONLY|O_CLOEXEC),saved;
22    ssize_t n;
23    if(fd<0) return -1;
24    do { n=write(fd,text,length); } while(n<0 && errno==EINTR);
25    saved=errno;
26    if(close(fd) && n==(ssize_t)length) return -1;
27    if(n!=(ssize_t)length) { errno=n<0?saved:EIO; return -1; }
28    return 0;
29}

31int lab_read_text(const char *path, char *buffer, size_t length)
32{
33    int fd,saved;
34    ssize_t n;
35    if(length<2) { errno=EINVAL; return -1; }
36    fd=open(path,O_RDONLY|O_CLOEXEC);
37    if(fd<0) return -1;
38    do { n=read(fd,buffer,length-1); } while(n<0 && errno==EINTR);
39    saved=errno;
40    if(close(fd) && n>=0) return -1;
41    if(n<0) { errno=saved; return -1; }
42    if(n==(ssize_t)length-1) { errno=EOVERFLOW; return -1; }
43    buffer[n]=0;
44    return 0;
45}

47int lab_read_u64(const char *path, uint64_t *value)
48{
49    char buffer[128],*end;
50    unsigned long long n;
51    if(lab_read_text(path,buffer,sizeof(buffer))) return -1;
52    errno=0; n=strtoull(buffer,&end,0);
53    if(errno || end==buffer || (*end && *end!='\n')) { errno=EINVAL; return -1; }
54    *value=n;
55    return 0;
56}

58int lab_link_basename(const char *path, char *buffer, size_t length)
59{
60    char target[PATH_MAX],*base;
61    ssize_t n=readlink(path,target,sizeof(target)-1);
62    if(n<0) return -1;
63    if(n==(ssize_t)sizeof(target)-1) { errno=EOVERFLOW; return -1; }
64    target[n]=0; base=strrchr(target,'/'); base=base?base+1:target;
65    if(strlen(base)>=length) { errno=ENAMETOOLONG; return -1; }
66    strcpy(buffer,base);
67    return 0;
68}

70int lab_ensure_char(const char *sysfs_dev, const char *path)
71{
72    char text[64];
73    unsigned major_num,minor_num;
74    struct stat st;
75    if(!stat(path,&st)) {
76        if(S_ISCHR(st.st_mode)) return 0;
77        errno=EINVAL; return -1;
78    }
79    if(errno!=ENOENT || lab_read_text(sysfs_dev,text,sizeof(text))) return -1;
80    if(sscanf(text,"%u:%u",&major_num,&minor_num)!=2) { errno=EINVAL; return -1; }
81    return mknod(path,S_IFCHR|0600,makedev(major_num,minor_num));
82}

95static int iommu_info(struct lab_vfio *vf)
96{
97    unsigned char buffer[16384] __attribute__((aligned(8)));
98    struct vfio_iommu_type1_info *info=(void *)buffer;
99    uint32_t offset,seen[64],count=0;
100    uint64_t page=(uint64_t)vf->bytes;
101    int found=0;
102    memset(buffer,0,sizeof(buffer)); info->argsz=sizeof(buffer);
103    if(ioctl(vf->container,VFIO_IOMMU_GET_INFO,info)) return -1;
104    if(info->argsz<sizeof(*info) || info->argsz>sizeof(buffer) || !(info->flags&VFIO_IOMMU_INFO_PGSIZES) || !(info->iova_pgsizes&page)) {
105        errno=EPROTO; return -1;
106    }
107    vf->page_sizes=info->iova_pgsizes;
108    offset=info->cap_offset;
109    while(offset) {
110        const struct vfio_info_cap_header *header;
111        unsigned j;
112        if(count==64 || offset>info->argsz-sizeof(*header) || (offset&3U)) { errno=EPROTO; return -1; }
113        for(j=0;j<count;++j) if(seen[j]==offset) { errno=EPROTO; return -1; }
114        seen[count++]=offset;
115        header=(const void *)(buffer+offset);
116        if(header->id==VFIO_IOMMU_TYPE1_INFO_CAP_IOVA_RANGE) {
117            const struct vfio_iommu_type1_info_cap_iova_range *range=(const void *)header;
118            size_t available=info->argsz-offset;
119            if(available<sizeof(*range) || header->version!=1 ||
120               range->nr_iovas>(available-sizeof(*range))/sizeof(range->iova_ranges[0])) { errno=EPROTO; return -1; }
121            for(j=0;j<range->nr_iovas;++j) {
122                uint64_t start=range->iova_ranges[j].start,end=range->iova_ranges[j].end,choice;
123                printf("{\"event\":\"iova_range\",\"vf\":\"%s\",\"start\":\"0x%" PRIx64 "\",\"end\":\"0x%" PRIx64 "\"}\n",vf->bdf,start,end);
124                if(start>UINT64_MAX-(page-1)) continue;
125                choice=(start+page-1)&~(page-1);
126                if(choice<0x100000 && end>=0x100000+page-1) choice=0x100000;
127                if(!found && choice<=end && page-1<=end-choice) { vf->iova=choice; found=1; }
128            }
129        }
130        offset=header->next;
131    }
132    if(!found) { errno=ENOTSUP; return -1; }
133    return 0;
134}

136/* 逐步填入已取得的資源；中途失敗仍由呼叫者呼叫 lab_vfio_close，不能只 free 結構。 */
137int lab_vfio_open(struct lab_vfio *vf, const char *bdf, unsigned group_id)
138{
139    char path[PATH_MAX],node[128],sysdev[128];
140    struct vfio_group_status group={.argsz=sizeof(group)};
141    struct vfio_irq_info irq={.argsz=sizeof(irq),.index=VFIO_PCI_MSIX_IRQ_INDEX};
142    long page=sysconf(_SC_PAGESIZE);
143    uint32_t magic=0,identity=0;
144    if(strlen(bdf)>=sizeof(vf->bdf) || page<=0 || ((unsigned long)page&((unsigned long)page-1))) { errno=EINVAL; return -1; }
145    strcpy(vf->bdf,bdf); vf->group_id=group_id; vf->bytes=(size_t)page;
146    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/driver_override",bdf);
147    if(lab_write_text(path,"vfio-pci\n")) return -1;
148    if(lab_write_text("/sys/bus/pci/drivers_probe",bdf)) return -1;
149    vf->bound=1;
150    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/driver",bdf);
151    if(lab_link_basename(path,node,sizeof(node)) || strcmp(node,"vfio-pci")) { errno=ENODEV; return -1; }
152    if(mkdir("/dev/vfio",0755) && errno!=EEXIST) return -1;
153    if(lab_ensure_char("/sys/class/misc/vfio/dev","/dev/vfio/vfio")) return -1;
154    snprintf(node,sizeof(node),"/dev/vfio/%u",group_id);
155    snprintf(sysdev,sizeof(sysdev),"/sys/class/vfio/%u/dev",group_id);
156    if(lab_ensure_char(sysdev,node)) return -1;
157    vf->container=open("/dev/vfio/vfio",O_RDWR|O_CLOEXEC);
158    if(vf->container<0) return -1;
159    if(ioctl(vf->container,VFIO_GET_API_VERSION)!=VFIO_API_VERSION ||
160       ioctl(vf->container,VFIO_CHECK_EXTENSION,VFIO_TYPE1_IOMMU)<=0) { errno=ENOTSUP; return -1; }
161    vf->group=open(node,O_RDWR|O_CLOEXEC);
162    if(vf->group<0 || ioctl(vf->group,VFIO_GROUP_GET_STATUS,&group)) return -1;
163    if(!(group.flags&VFIO_GROUP_FLAGS_VIABLE)) { errno=EBUSY; return -1; }
164    if(ioctl(vf->group,VFIO_GROUP_SET_CONTAINER,&vf->container)) return -1;
165    vf->attached=1;
166    if(ioctl(vf->container,VFIO_SET_IOMMU,VFIO_TYPE1_IOMMU)) return -1;
167    vf->device=ioctl(vf->group,VFIO_GROUP_GET_DEVICE_FD,bdf);
168    if(vf->device<0) return -1;
169    if(fcntl(vf->device,F_SETFD,FD_CLOEXEC)) return -1;
170    vf->info.argsz=sizeof(vf->info);
171    if(ioctl(vf->device,VFIO_DEVICE_GET_INFO,&vf->info)) return -1;
172    if(!(vf->info.flags&VFIO_DEVICE_FLAGS_PCI) || !(vf->info.flags&VFIO_DEVICE_FLAGS_RESET) ||
173        vf->info.num_regions<=VFIO_PCI_CONFIG_REGION_INDEX || vf->info.num_irqs<=VFIO_PCI_MSIX_IRQ_INDEX) { errno=EPROTO; return -1; }
174    if(region(vf->device,VFIO_PCI_BAR0_REGION_INDEX,&vf->bar0) ||
175       region(vf->device,VFIO_PCI_BAR2_REGION_INDEX,&vf->bar2) ||
176       region(vf->device,VFIO_PCI_CONFIG_REGION_INDEX,&vf->config) ||
177       ioctl(vf->device,VFIO_DEVICE_GET_IRQ_INFO,&irq)) return -1;
178    if(vf->bar0.size!=TOY_LIMITS_VF_BAR0_BYTES || vf->bar2.size!=TOY_LIMITS_BAR2_BYTES ||
179       irq.count!=TOY_LIMITS_VECTORS_PER_FUNCTION || !(irq.flags&VFIO_IRQ_INFO_EVENTFD)) { errno=EPROTO; return -1; }
180    if(pread(vf->device,&identity,sizeof(identity),(off_t)vf->config.offset)!=(ssize_t)sizeof(identity) ||
181       identity!=(TOY_PCI_VENDOR|(TOY_PCI_VF<<16))) { errno=EPROTO; return -1; }
182    if(pread(vf->device,&magic,sizeof(magic),(off_t)vf->bar0.offset)!=(ssize_t)sizeof(magic) || magic!=TOY_VF_MAGIC_VALUE) { errno=EPROTO; return -1; }
183    if(iommu_info(vf)) return -1;
184    printf("{\"event\":\"vfio_open\",\"vf\":\"%s\",\"group\":%u,\"container_fd\":%d,\"group_fd\":%d,\"device_fd\":%d,\"vectors\":%u,\"page_sizes\":\"0x%" PRIx64 "\",\"reset_supported\":true}\n",
185           bdf,group_id,vf->container,vf->group,vf->device,irq.count,vf->page_sizes);
186    return 0;
187}

189int lab_vfio_map_probe(struct lab_vfio *vf)
190{
191    struct vfio_iommu_type1_dma_map map={.argsz=sizeof(map),.flags=VFIO_DMA_MAP_FLAG_READ|VFIO_DMA_MAP_FLAG_WRITE};
192    vf->buffer=mmap(NULL,vf->bytes,PROT_READ|PROT_WRITE,MAP_PRIVATE|MAP_ANONYMOUS,-1,0);
193    if(vf->buffer==MAP_FAILED) { vf->buffer=NULL; return -1; }
194    memset(vf->buffer,(int)(vf->group_id&255U),vf->bytes);
195    map.vaddr=(uint64_t)(uintptr_t)vf->buffer; map.iova=vf->iova; map.size=vf->bytes;
196    if(ioctl(vf->container,VFIO_IOMMU_MAP_DMA,&map)) return -1;
197    vf->mapped=1;
198    printf("{\"event\":\"vfio_map\",\"vf\":\"%s\",\"iova\":\"0x%" PRIx64 "\",\"size\":%zu,\"permissions\":\"READ_WRITE\"}\n",vf->bdf,vf->iova,vf->bytes);
199    return 0;
200}

202/* 依已取得的欄位回收部分初始化狀態；任何一步失敗都保留失敗回傳值。 */
203int lab_vfio_close(struct lab_vfio *vf)
204{
205    char path[PATH_MAX];
206    int failed=0;
207    if(vf->mapped) {
208        struct vfio_iommu_type1_dma_unmap unmap={.argsz=sizeof(unmap),.iova=vf->iova,.size=vf->bytes};
209        if(ioctl(vf->container,VFIO_IOMMU_UNMAP_DMA,&unmap) || unmap.size!=vf->bytes) {
210            fprintf(stderr,"VFIO_UNMAP_FAILED vf=%s actual=%" PRIu64 " expected=%zu errno=%d\n",vf->bdf,(uint64_t)unmap.size,vf->bytes,errno);
211            failed=1;
212        } else {
213            vf->mapped=0;
214            printf("{\"event\":\"vfio_unmap\",\"vf\":\"%s\",\"size\":%zu}\n",vf->bdf,vf->bytes);
215        }
216    }
217    /* UNMAP 失敗時先關 owner fds，使 VFIO 撤銷映射後才釋放 userspace RAM。 */
218    if(vf->device>=0) { if(close(vf->device)) failed=1; vf->device=-1; }
219    if(vf->attached) { if(ioctl(vf->group,VFIO_GROUP_UNSET_CONTAINER)) failed=1; else vf->attached=0; }
220    if(vf->group>=0) { if(close(vf->group)) failed=1; vf->group=-1; }
221    if(vf->container>=0) { if(close(vf->container)) failed=1; vf->container=-1; }
222    vf->mapped=0;
223    if(vf->buffer) { if(munmap(vf->buffer,vf->bytes)) failed=1; vf->buffer=NULL; }
224    if(vf->bound) {
225        if(lab_write_text("/sys/bus/pci/drivers/vfio-pci/unbind",vf->bdf)) failed=1;
226        else vf->bound=0;
227    }
228    if(vf->bdf[0]) {
229        snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/driver_override",vf->bdf);
230        if(lab_write_text(path,"\n")) failed=1;
231    }
232    return failed?-1:0;
