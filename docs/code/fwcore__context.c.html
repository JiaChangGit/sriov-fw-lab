4void fw_zero(void *ptr, fw_size size)
5{
6    fw_u8 *p = (fw_u8 *)ptr;
7    fw_size i;
8    for (i = 0; i < size; ++i) p[i] = 0;

11void fw_copy(void *dst, const void *src, fw_size size)
12{
13    fw_u8 *d = (fw_u8 *)dst;
14    const fw_u8 *s = (const fw_u8 *)src;
15    fw_size i;
16    for (i = 0; i < size; ++i) d[i] = s[i];

19fw_status fw_hw(const struct fw_context *ctx, fw_u16 vf, struct fw_hw_snapshot *out)
20{
21    fw_status rc;
22    if (!ctx || !out || vf >= FW_VFS || !ctx->hal.read_hw) return FW_INVAL;
23    fw_zero(out, sizeof(*out));
24    rc = ctx->hal.read_hw(ctx->hal.opaque, vf, out);
25    if (rc != FW_OK) return rc;
26    if (out->session != ctx->session || out->epoch < ctx->vf[vf].epoch) return FW_STATE;
27    return FW_OK;
28}

35fw_status fw_deadline(const struct fw_context *ctx, fw_u64 delta, fw_u64 *out)
36{
37    fw_u64 now = ctx->hal.now_tick(ctx->hal.opaque);
38    if (now > FW_U64_MAX - delta) return FW_OVERFLOW;
39    *out = now + delta;
40    return FW_OK;
41}

43/* 先在區域變數建立完整 context；任一 VF 快照不符時，不覆寫呼叫者的舊 context。 */
44fw_status fw_init(struct fw_context *ctx, const struct fw_hal *hal)
45{
46    struct fw_context fresh;
47    struct fw_hw_snapshot hw;
48    fw_u16 i;
49    fw_status rc;
50    if (!ctx || !hal || !hal->now_tick || !hal->read_hw || !hal->cancel_io ||
51        !hal->clear_queue || !hal->set_gate || !hal->kill || !hal->owners_closed) return FW_INVAL;
52    fw_zero(&fresh, sizeof(fresh));
53    fresh.hal = *hal;
54    for (i = 0; i < FW_VFS; ++i) {
55        fw_zero(&hw, sizeof(hw));
56        rc = hal->read_hw(hal->opaque, i, &hw);
57        if (rc != FW_OK) return rc;
58        if (!hw.session || !hw.epoch || !hw.fenced || hw.config_valid || hw.outstanding)
59            return FW_STATE;
60        if (i == 0) fresh.session = hw.session;
61        if (hw.session != fresh.session) return FW_STATE;
62        fresh.vf[i].epoch = hw.epoch;
63    }
64    *ctx = fresh;
65    return FW_OK;
66}

68/* 讀完所有 VF 才複製到 out，讓呼叫者不會收到半份成功的 snapshot。 */
69fw_status fw_snapshot(const struct fw_context *ctx, struct fw_snapshot *out)
70{
71    struct fw_snapshot snap;
72    fw_u16 i;
73    fw_status rc;
74    if (!ctx || !out) return FW_INVAL;
75    fw_zero(&snap, sizeof(snap));
76    snap.session=ctx->session; snap.round=ctx->round;
77    snap.used_queues=ctx->used_queues; snap.used_vectors=ctx->used_vectors;
78    snap.used_credit=ctx->used_credit; snap.last_seq=ctx->last_seq;
79    snap.negotiated=ctx->negotiated; snap.cache_valid=ctx->cache_valid;
80    for (i=0; i<FW_VFS; ++i) {
81        snap.vf[i]=ctx->vf[i];
82        rc=fw_hw(ctx,i,&snap.hw[i]);
83        if (rc!=FW_OK) return rc;
84    }
85    *out=snap;
86    return FW_OK;
87}

89/* 由各 VF 重新加總 ledger，並檢查 FREE/open-gate 等跨欄位條件。 */
90fw_status fw_check_invariants(const struct fw_context *ctx)
91{
92    fw_u32 queues=0, credit=0;
93    fw_u16 i;
94    struct fw_hw_snapshot hw;
95    fw_status rc;
96    if (!ctx || !ctx->session) return FW_INVAL;
97    for (i=0; i<FW_VFS; ++i) {
98        if (ctx->vf[i].state >= FW_STATE_COUNT || ctx->vf[i].reserve > FW_CAP_CREDIT) return FW_STATE;
99        rc=fw_hw(ctx,i,&hw);
100        if (rc!=FW_OK) return rc;
101        if (hw.outstanding>1) return FW_STATE;
102        if (ctx->vf[i].state==FW_FREE && (ctx->vf[i].reserve || hw.config_valid || hw.outstanding)) return FW_STATE;
103        if (ctx->vf[i].reserve) { ++queues; credit+=ctx->vf[i].reserve; }
104        if (!hw.fenced && (hw.killed || !hw.config_valid || !hw.bme || ctx->vf[i].state!=FW_ACTIVE)) return FW_STATE;
105    }
106    if (credit>FW_CAP_CREDIT || queues!=ctx->used_queues || 2U*queues!=ctx->used_vectors || credit!=ctx->used_credit)
107        return FW_STATE;
108    return FW_OK;
109}

111fw_status fw_global_recover(struct fw_context *ctx, fw_u64 new_session, const fw_u64 epochs[FW_VFS])
112{
113    struct fw_hw_snapshot hw;
114    fw_u16 i;
115    fw_status rc;
116    if (!ctx || !epochs || !new_session) return FW_INVAL;
117    if (ctx->session==FW_U64_MAX) return FW_OVERFLOW;
118    if (new_session<=ctx->session || !ctx->hal.owners_closed(ctx->hal.opaque, FW_GLOBAL)) return FW_STATE;
119    for (i=0; i<FW_VFS; ++i) {
120        if (ctx->vf[i].epoch==FW_U64_MAX) return FW_OVERFLOW;
121        rc=ctx->hal.read_hw(ctx->hal.opaque,i,&hw);
122        if (rc!=FW_OK) return rc;
123        if (epochs[i]<=ctx->vf[i].epoch || hw.epoch!=epochs[i] || hw.session!=new_session ||
124            !hw.fenced || hw.config_valid || hw.outstanding || hw.pci_present) return FW_STATE;
125    }
126    /* 只有完成全部前置驗證後才提交，kill 仍由硬體保留到 end_global。 */
127    for (i=0; i<FW_VFS; ++i) {
128        fw_zero(&ctx->vf[i],sizeof(ctx->vf[i]));
129        ctx->vf[i].epoch=epochs[i];
130    }
131    ctx->session=new_session; ctx->round=0;
132    ctx->used_queues=0; ctx->used_vectors=0; ctx->used_credit=0;
133    ctx->last_seq=0; ctx->negotiated=0; ctx->cache_valid=0;
134    ctx->mailbox_pending=0; ctx->pending_epoch=0; ctx->pending_deadline=0;
135    fw_zero(&ctx->cached_request,sizeof(ctx->cached_request));
136    fw_zero(&ctx->cached_reply,sizeof(ctx->cached_reply));
137    fw_note(ctx,"global_recovered",FW_GLOBAL,new_session);
138    return FW_OK;
139}

141fw_status fw_poll_deadlines(struct fw_context *ctx)
142{
143    fw_u16 i;
144    fw_u64 now;
145    fw_status rc=FW_OK, one;
146    struct fw_hw_snapshot hw;
147    if (!ctx) return FW_INVAL;
148    now=ctx->hal.now_tick(ctx->hal.opaque);
149    for (i=0; i<FW_VFS; ++i) {
150        enum fw_state s=ctx->vf[i].state;
151        if ((s==FW_QUIESCING || s==FW_RESETTING || s==FW_RECOVERING) && now>=ctx->vf[i].deadline) {
152            /* PCI 移除後已無 READY 端點；等待 DESTROY，保留 quota 與 fence。 */
153            if (s==FW_RECOVERING) {
154                one=fw_hw(ctx,i,&hw);
155                if (one!=FW_OK) { rc=one; continue; }
156                if (!hw.pci_present) continue;
157            }
158            one=fw_vf_event(ctx,i,s==FW_RECOVERING?FW_VERIFY_FAIL:FW_EV_TIMEOUT);
159            if (one!=FW_OK) rc=one;
160        }
161    }
162    return rc;
