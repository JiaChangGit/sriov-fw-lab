4static fw_u64 get_le(const fw_u8 *p, unsigned n)
5{
6    fw_u64 value=0;
7    unsigned i;
8    for (i=0; i<n; ++i) value|=(fw_u64)p[i]<<(8U*i);
9    return value;
10}
11
12static void put_le(fw_u8 *p, fw_u64 value, unsigned n)
13{
14    unsigned i;
15    for (i=0; i<n; ++i) p[i]=(fw_u8)(value>>(8U*i));
16}

18static fw_status validate_sq(const struct fw_sq *sq)
19{
20    if (!sq->len || sq->len>4096 || sq->flags || (sq->op!=1 && sq->op!=2) ||
21        sq->addr>FW_U64_MAX-(fw_u64)sq->len) return FW_BAD_DESC;
22    return FW_OK;
23}

25fw_status fw_decode_sq(const fw_u8 bytes[32], struct fw_sq *out)
26{
27    struct fw_sq sq;
28    fw_status rc;
29    if (!bytes || !out) return FW_INVAL;
30    fw_zero(&sq,sizeof(sq));
31    sq.addr=get_le(bytes,8); sq.len=(fw_u32)get_le(bytes+8,4);
32    sq.op=bytes[12]; sq.flags=bytes[13]; sq.id=(fw_u16)get_le(bytes+14,2);
33    sq.epoch=get_le(bytes+16,8); sq.arg=get_le(bytes+24,8);
34    rc=validate_sq(&sq); if (rc!=FW_OK) return rc;
35    *out=sq;
36    return FW_OK;
37}

39fw_status fw_encode_sq(const struct fw_sq *sq, fw_u8 bytes[32])
40{
41    fw_status rc;
42    if (!sq || !bytes) return FW_INVAL;
43    rc=validate_sq(sq); if (rc!=FW_OK) return rc;
44    put_le(bytes,sq->addr,8); put_le(bytes+8,sq->len,4);
45    bytes[12]=sq->op; bytes[13]=sq->flags; put_le(bytes+14,sq->id,2);
46    put_le(bytes+16,sq->epoch,8); put_le(bytes+24,sq->arg,8);
47    return FW_OK;
48}

50fw_status fw_decode_cq(const fw_u8 bytes[32], struct fw_cq *out)
51{
52    struct fw_cq cq;
53    unsigned i;
54    if (!bytes || !out) return FW_INVAL;
55    for (i=25; i<32; ++i) if (bytes[i]) return FW_BAD_DESC;
56    if (bytes[24]>1 || get_le(bytes+2,2)>FW_IO || get_le(bytes+4,4)>4096) return FW_BAD_DESC;
57    fw_zero(&cq,sizeof(cq));
58    cq.id=(fw_u16)get_le(bytes,2); cq.status=(fw_u16)get_le(bytes+2,2);
59    cq.len=(fw_u32)get_le(bytes+4,4); cq.epoch=get_le(bytes+8,8);
60    cq.result=get_le(bytes+16,8); cq.phase=bytes[24];
61    *out=cq;
62    return FW_OK;
63}
64
65fw_status fw_encode_cq(const struct fw_cq *cq, fw_u8 bytes[32])
66{
67    if (!cq || !bytes) return FW_INVAL;
68    if (cq->phase>1 || cq->status>FW_IO || cq->len>4096) return FW_BAD_DESC;
69    fw_zero(bytes,32);
70    put_le(bytes,cq->id,2); put_le(bytes+2,cq->status,2); put_le(bytes+4,cq->len,4);
71    put_le(bytes+8,cq->epoch,8); put_le(bytes+16,cq->result,8); bytes[24]=cq->phase;
72    return FW_OK;
73}

75static fw_u8 valid_entries(fw_u32 n)
76{
77    return (fw_u8)(n>=16 && n<=256 && !(n&(n-1U)));
78}

80/* 驗證 queue 大小、對齊、世代及 SQ/CQ 位址範圍；不在這裡建立 IOMMU mapping。 */
81fw_status fw_queue_validate(const struct fw_queue_config *cfg, fw_u64 session, fw_u64 epoch)
82{
83    fw_u64 bytes, sq_end, cq_end;
84    if (!cfg) return FW_INVAL;
85    if (!valid_entries(cfg->sq_entries) || cfg->cq_entries!=cfg->sq_entries ||
86        (cfg->sq_base&63U) || (cfg->cq_base&63U) || cfg->epoch!=epoch ||
87        cfg->session!=session || !session || !epoch) return FW_BAD_QUEUE;
88    bytes=(fw_u64)cfg->sq_entries*32U;
89    if (cfg->sq_base>FW_U64_MAX-bytes || cfg->cq_base>FW_U64_MAX-bytes) return FW_BAD_QUEUE;
90    sq_end=cfg->sq_base+bytes; cq_end=cfg->cq_base+bytes;
91    if (cfg->sq_base<cq_end && cfg->cq_base<sq_end) return FW_BAD_QUEUE;
92    return FW_OK;
93}

95/* u32 減法允許 counter 回繞；有效差值仍必須小於或等於 ring 容量。 */
96fw_status fw_ring_pending(fw_u32 tail, fw_u32 head, fw_u32 entries, fw_u32 *pending)
97{
98    fw_u32 count=tail-head;
99    if (!pending) return FW_INVAL;
100    if (!valid_entries(entries) || count>entries) return FW_BAD_QUEUE;
101    *pending=count;
102    return FW_OK;
103}

105fw_u8 fw_ring_phase(fw_u32 counter, fw_u32 entries)
106{
107    if (!valid_entries(entries)) return 0;
108    return (fw_u8)(1U^((counter/entries)&1U));
