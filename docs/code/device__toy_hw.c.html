23uint32_t toy_hw_flags(ToyPF *pf, unsigned vf)
24{
25    ToySlot *s = &pf->slots[vf];
26    return (s->fenced ? TOY_HW_FLAGS_FENCED : 0) |
27           (s->killed ? TOY_HW_FLAGS_KILLED : 0) |
28           (s->vf ? TOY_HW_FLAGS_PCI_PRESENT : 0) |
29           (s->vf && (pci_get_word(s->vf->parent_obj.config + PCI_COMMAND) & PCI_COMMAND_MASTER) ? TOY_HW_FLAGS_BME : 0) |
30           (s->config_valid ? TOY_HW_FLAGS_CONFIG_VALID : 0) |
31           (s->ready_pending ? TOY_HW_FLAGS_READY_PENDING : 0);

34/* 保留舊 operation 的副本供 stale-callback 測試；移出 pending 後不再由正常 worker 完成。 */
35static void cancel_pending(ToySlot *s)
36{
37    if (s->pending.valid) {
38        s->retired = s->pending;
39        s->pending.valid = false;
40    }
41}

45    if (s->vf) {

52fw_status toy_fence(ToyPF *pf, unsigned vf, uint32_t token, bool deduplicate)
53{
54    ToySlot *s = &pf->slots[vf];
55    if (deduplicate) {
56        if (!token) {
57            return FW_INVAL;
58        }
59        if (token < s->last_fence_token) {
60            return FW_STALE_SEQ;
61        }
62        if (token == s->last_fence_token) {
63            s->fenced = true;
64            cancel_pending(s);
65            clear_pending_irq(s);
66            return FW_OK;
67        }
68    }
69    /* 順序不可依賴 firmware 回應；同一 BQL 區段先關 gate 再換代。 */
70    s->fenced = true;
71    cancel_pending(s);
72    clear_pending_irq(s);
73    if (s->epoch == UINT64_MAX || s->token == UINT64_MAX) {
74        s->killed = true;
75        return FW_OVERFLOW;
76    }
77    s->epoch++;
78    s->token++;
79    if (deduplicate) {
80        s->last_fence_token = token;
81        s->last_fence_epoch = s->epoch;
82    }
83    return FW_OK;
84}

86void toy_kill(ToyPF *pf, unsigned vf)
87{
88    ToySlot *s = &pf->slots[vf];
89    s->fenced = s->killed = true;
90    if (s->token != UINT64_MAX) {
91        s->token++;
92    }
93    cancel_pending(s);
94    clear_pending_irq(s);
95}

97void toy_clear_queue(ToyPF *pf, unsigned vf)
98{
99    ToySlot *s = &pf->slots[vf];
100    cancel_pending(s);
101    clear_pending_irq(s);
102    memset(&s->config, 0, sizeof(s->config));
103    memset(&s->staging, 0, sizeof(s->staging));
104    s->config_valid = s->ready_pending = false;
105    s->sq_head = s->sq_tail = s->cq_head = s->cq_tail = 0;
106    s->queue_status = TOY_QUEUE_STATUS_UNCONFIGURED;
107    s->queue_error = FW_OK;
108    s->dma_trace_seen = 0;
109}

111void toy_vf_flr(ToyPF *pf, unsigned vf)
112{
113    fw_status status = toy_fence(pf, vf, 0, false);
114    toy_clear_queue(pf, vf);
115    if (status == FW_OK && pf->core.vf[vf].state != FW_FREE) {
116        status = fw_vf_event(&pf->core, vf, FW_FLR);
117    }
118    if (status != FW_OK) {
119        toy_kill(pf, vf);
120    }
121    pf->event |= TOY_PF_EVENTS_FLR_EVENT;
122    msix_notify(PCI_DEVICE(pf), 1);
123    toy_schedule(pf);
124    fprintf(stderr, "TOY_FLR vf=%u epoch=%" PRIu64 " status=%u\n", vf, pf->slots[vf].epoch, status);
125}

127void toy_fail(ToyPF *pf, unsigned vf, uint32_t reason)
128{
129    ToySlot *s = &pf->slots[vf];
130    fw_status status;
131    s->last_fault = reason;
132    status = toy_fence(pf, vf, 0, false);
133    toy_kill(pf, vf);
134    /* 不虛構 timeout：沿已定義的 fence/drain/reset/verify-fail 事件收斂。 */
135    if (status == FW_OK && pf->core.vf[vf].state == FW_ACTIVE) {
136        status = fw_vf_event(&pf->core, vf, FW_FENCE);
137    }
138    if (status == FW_OK && pf->core.vf[vf].state == FW_QUIESCING) {
139        status = fw_vf_event(&pf->core, vf, FW_DRAIN_OK);
140    }
141    if (status == FW_OK && pf->core.vf[vf].state == FW_RESETTING) {
142        status = fw_vf_event(&pf->core, vf, FW_RESET_DONE);
143    }
144    if (status == FW_OK && pf->core.vf[vf].state == FW_RECOVERING) {
145        status = fw_vf_event(&pf->core, vf, FW_VERIFY_FAIL);
146    }
147    s->queue_status = TOY_QUEUE_STATUS_ERROR;
148    s->queue_error = reason == TOY_LAST_FAULT_DMA ? FW_DMA : FW_BAD_QUEUE;
149    pf->event |= TOY_PF_EVENTS_VF_HEALTH;
150    msix_notify(PCI_DEVICE(pf), 1);
151    fprintf(stderr, "TOY_FAULT vf=%u rid=%04x reason=%u core_status=%u\n", vf,
152            s->vf ? pci_get_bdf(PCI_DEVICE(s->vf)) : 0xffff, reason, status);
153}

155static bool same_generation(ToyPF *pf, ToySlot *s, const ToyOperation *op)
156{
157    return !op || (op->session == pf->session && op->epoch == s->epoch && op->token == s->token);
158}

160/* 每次 DMA/IRQ 前重查存取資格；op 為 NULL 時沒有舊工作 tuple，仍需檢查目前硬體狀態。 */
161static bool gate(ToyPF *pf, unsigned vf, const ToyOperation *op)
162{
163    ToySlot *s = &pf->slots[vf];
164    bool allowed = s->vf && !s->fenced && !s->killed && s->config_valid &&
165                   pf->core.vf[vf].state == FW_ACTIVE && same_generation(pf, s, op) &&
166                   (pci_get_word(s->vf->parent_obj.config + PCI_COMMAND) & PCI_COMMAND_MASTER);
167    if (!allowed) {
168        toy_saturate(&s->fence_blocked);
169    }
170    return allowed;
171}

173enum ToyDmaRole { DMA_SQ, DMA_PAYLOAD_READ, DMA_PAYLOAD_WRITE, DMA_CQ, DMA_CQ_PHASE };
174
175static bool dma(ToyPF *pf, unsigned vf, const ToyOperation *op,
176                uint64_t addr, void *buffer, uint32_t length, bool write,
177                enum ToyDmaRole role)
178{
179    ToySlot *s = &pf->slots[vf];
180    MemTxResult result;
181    if (!gate(pf, vf, op)) {
182        return false;
183    }
184    /* 一律以該 VF 的 requester 進入 PCI DMA helper。 */
185    result = write ? pci_dma_write(PCI_DEVICE(s->vf), addr, buffer, length) :
186                     pci_dma_read(PCI_DEVICE(s->vf), addr, buffer, length);
187    if (result != MEMTX_OK) {
188        toy_saturate(&s->dma_fault);
189        fprintf(stderr, "TOY_DMA_ERROR vf=%u rid=%04x addr=%" PRIx64 " length=%u write=%u\n",
190                vf, pci_get_bdf(PCI_DEVICE(s->vf)), addr, length, write);
191        return false;
192    }
193    /* 每次配置僅記錄各用途首筆成功 DMA，requester 與實際 helper 使用同一 VF。 */
194    if (!(s->dma_trace_seen & (1u << role))) {
195        static const char *const names[] = { "sq", "payload_read", "payload_write", "cq", "cq_phase" };
196        s->dma_trace_seen |= 1u << role;
197        fprintf(stderr, "TOY_DMA_OK vf=%u rid=%04x role=%s addr=%" PRIx64 " length=%u write=%u epoch=%" PRIu64 "\n",
198                vf, pci_get_bdf(PCI_DEVICE(s->vf)), names[role], addr, length, write, s->epoch);
199    }
200    return true;
201}

203void toy_irq(ToyPF *pf, unsigned vf, unsigned vector, const ToyOperation *op)
204{
205    ToySlot *s = &pf->slots[vf];
206    if (!gate(pf, vf, op)) {
207        toy_saturate(&s->irq_suppressed);
208        return;
209    }
210    s->irq_cause |= 1u << vector;
211    msix_notify(PCI_DEVICE(s->vf), vector);
212}

214/* 延遲工作帶著提交當時的 session/epoch/token；先拒絕舊世代，再碰 payload 或 CQ。 */
215void toy_complete(ToyPF *pf, unsigned vf, ToyOperation *op)
216{
217    ToySlot *s = &pf->slots[vf];
218    uint8_t payload[4096], completion[32];
219    uint64_t address;
220    uint32_t done = 0;
221    bool payload_fault = false;
222    struct fw_cq cq;
223    if (!op->valid) {
224        return;
225    }
226    if (!same_generation(pf, s, op)) {
227        toy_saturate(&s->stale_cpl);
228        fprintf(stderr, "TOY_STALE_CPL vf=%u old_session=%" PRIu64 " live_session=%" PRIu64
229                " old_epoch=%" PRIu64 " live_epoch=%" PRIu64 " old_token=%" PRIu64
230                " live_token=%" PRIu64 " count=%u\n", vf, op->session, pf->session,
231                op->epoch, s->epoch, op->token, s->token, s->stale_cpl);
232        op->valid = false;
233        return;
234    }
235    if (!gate(pf, vf, op)) {
236        op->valid = false;
237        return;
238    }
239    if (op->status == FW_OK) {
240        if (op->sq.op == TOY_SQ_OPS_FILL) {
241            memset(payload, (uint8_t)op->sq.arg, op->sq.len);
242        }
243        while (done < op->sq.len) {
244            uint32_t chunk = MIN(op->sq.len - done, 4096 - ((op->sq.addr + done) & 4095));
245            if (!dma(pf, vf, op, op->sq.addr + done, payload + done, chunk,
246                     op->sq.op == TOY_SQ_OPS_FILL,
247                     op->sq.op == TOY_SQ_OPS_FILL ? DMA_PAYLOAD_WRITE : DMA_PAYLOAD_READ)) {
248                payload_fault = true;
249                op->status = FW_DMA;
250                break;
251            }
252            done += chunk;
253        }
254        if (!payload_fault && op->sq.op == TOY_SQ_OPS_CRC32) {
255            op->result = fw_crc32(payload, op->sq.len);
256        }
257    }
258    memset(&cq, 0, sizeof(cq));
259    cq.id = op->sq.id;
260    cq.status = op->status;
261    cq.len = op->status == FW_BAD_DESC ? 0 : op->sq.len;
262    cq.epoch = op->epoch;
263    cq.result = op->result;
264    cq.phase = fw_ring_phase(op->cq_index, s->config.cq_entries);
265    if (fw_encode_cq(&cq, completion) != FW_OK) {
266        toy_fail(pf, vf, TOY_LAST_FAULT_BAD_DESC);
267        op->valid = false;
268        return;
269    }
270    address = s->config.cq_base + (op->cq_index & (s->config.cq_entries - 1)) * 32;
271    /* phase 為獨立最後一筆 DMA；CQ_TAIL 及 IRQ 僅在其成功後發布。 */
272    if (!dma(pf, vf, op, address, completion, 24, true, DMA_CQ) ||
273        !dma(pf, vf, op, address + 25, completion + 25, 7, true, DMA_CQ) ||
274        !dma(pf, vf, op, address + 24, completion + 24, 1, true, DMA_CQ_PHASE)) {
275        toy_fail(pf, vf, TOY_LAST_FAULT_DMA);
276        op->valid = false;
277        return;
278    }
279    s->cq_tail++;
280    toy_irq(pf, vf, 0, op);
281    op->valid = false;
282    if (payload_fault) {
283        toy_fail(pf, vf, TOY_LAST_FAULT_DMA);
284    }
285}

287/* 每個 VF 最多保留一筆 delayed operation；CQ 已滿時不能再 fetch SQ，避免覆寫未消費結果。 */
288static bool service_one(ToyPF *pf, unsigned vf)
289{
290    ToySlot *s = &pf->slots[vf];
291    uint8_t bytes[32];
292    ToyOperation op = { 0 };
293    fw_status status;
294    if (!gate(pf, vf, NULL) || s->pending.valid || s->sq_head == s->sq_tail ||
295        (uint32_t)(s->cq_tail - s->cq_head) >= s->config.cq_entries) {
296        return false;
297    }
298    if (!dma(pf, vf, NULL, s->config.sq_base + (s->sq_head & (s->config.sq_entries - 1)) * 32,
299             bytes, sizeof(bytes), false, DMA_SQ)) {
300        toy_fail(pf, vf, TOY_LAST_FAULT_DMA);
301        return false;
302    }
303    s->sq_head++;
304    op.valid = true;
305    op.session = pf->session;
306    op.epoch = s->epoch;
307    op.token = s->token;
308    op.cq_index = s->cq_tail;
309    status = fw_decode_sq(bytes, &op.sq);
310    /* epoch 在任何 payload/CQ 寫入前比較，與格式錯誤分開計數。 */
311    if (ldq_le_p(bytes + 16) != s->epoch) {
312        toy_saturate(&s->stale_desc);
313        return true;
314    }
315    if (status != FW_OK) {
316        op.sq.id = lduw_le_p(bytes + 14);
317        op.sq.len = ldl_le_p(bytes + 8);
318        op.status = FW_BAD_DESC;
319        s->last_fault = TOY_LAST_FAULT_BAD_DESC;
320    }
321    pf->operations++;
322    if (pf->delay_cpl_ticks && toy_fault_target(pf, vf)) {
323        op.due = toy_tick() + pf->delay_cpl_ticks;
324        s->pending = op;
325        fprintf(stderr, "TOY_PENDING vf=%u epoch=%" PRIu64 " token=%" PRIu64
326                " cq_index=%u due=%" PRIu64 "\n", vf, op.epoch, op.token, op.cq_index, op.due);
327    } else {
328        toy_complete(pf, vf, &op);
329    }
330    return true;
331}

340static void observe_kill(ToyPF *pf, unsigned vf)
341{
342    fw_status status = FW_OK;
343    enum fw_state state = pf->core.vf[vf].state;
344    if (!pf->slots[vf].killed || state == FW_FREE || state == FW_FAILED) {
345        return;
346    }
347    /* MMIO KILL 只關硬體 gate；正常運作的 firmware 另以合法事件保留 ledger。 */
348    if (state == FW_PROVISIONING) {
349        status = fw_vf_event(&pf->core, vf, FW_ABORT_FAIL);
350    } else if (state == FW_DESTROYING) {
351        status = fw_vf_event(&pf->core, vf, FW_RELEASE_FAIL);
352    } else {
353        if (state == FW_ACTIVE) {
354            /* KILL 可能未換 epoch；取得新 fence 才能送出 FENCE 事件。 */
355            status = toy_fence(pf, vf, 0, false);
356            if (status == FW_OK) {
357                status = fw_vf_event(&pf->core, vf, FW_FENCE);
358            }
359        }
360        if (status == FW_OK && pf->core.vf[vf].state == FW_QUIESCING) {
361            status = fw_vf_event(&pf->core, vf, FW_DRAIN_OK);
362        }
363        if (status == FW_OK && pf->core.vf[vf].state == FW_RESETTING) {
364            status = fw_vf_event(&pf->core, vf, FW_RESET_DONE);
365        }
366        if (status == FW_OK && pf->core.vf[vf].state == FW_RECOVERING) {
367            status = fw_vf_event(&pf->core, vf, FW_VERIFY_FAIL);
368        }
369    }
370    pf->event |= TOY_PF_EVENTS_VF_HEALTH;
371    msix_notify(PCI_DEVICE(pf), 1);
372    fprintf(stderr, "TOY_KILL_OBSERVED vf=%u status=%u state=%u\n", vf, status,
373            pf->core.vf[vf].state);
374}

376void toy_worker(void *opaque)
377{
378    ToyPF *pf = opaque;
379    fw_u32 backlog[FW_VFS] = { 0 }, service[FW_VFS] = { 0 };
380    bool more = false;
381    unsigned start = pf->core.round % FW_VFS;
382    fw_status status;
383    if (pf->fw_hang_all) {
384        return;
385    }
386    for (unsigned i = 0; i < FW_VFS; i++) {
387        ToySlot *s = &pf->slots[i];
388        if (pf->fw_hang_vf && toy_fault_target(pf, i)) {
389            continue;
390        }
391        observe_kill(pf, i);
392        if (s->pending.valid) {
393            if (s->pending.due <= toy_tick()) {
394                toy_complete(pf, i, &s->pending);
395            } else {
396                more = true;
397                continue;
398            }
399        }
400        if (s->vf && !s->fenced && !s->killed && s->config_valid && pf->core.vf[i].state == FW_ACTIVE) {
401            uint32_t pending = s->sq_tail - s->sq_head;
402            uint32_t used = s->cq_tail - s->cq_head;
403            if (pending > s->config.sq_entries || used > s->config.cq_entries) {
404                toy_fail(pf, i, TOY_LAST_FAULT_BAD_QUEUE);
405                continue;
406            }
407            backlog[i] = MIN(pending, s->config.cq_entries - used);
408            if (pf->delay_cpl_ticks && toy_fault_target(pf, i)) {
409                backlog[i] = MIN(backlog[i], 1);
410            }
411        }
412    }
413    status = fw_schedule_round(&pf->core, backlog, service);
414    if (status != FW_OK) {
415        fprintf(stderr, "TOY_SCHEDULER_ERROR status=%u\n", status);
416        for (unsigned i = 0; i < FW_VFS; i++) {
417            toy_kill(pf, i);
418        }
419        return;
420    }
421    for (unsigned round = 0; round < 64; round++) {
422        for (unsigned offset = 0; offset < FW_VFS; offset++) {
423            unsigned i = (start + offset) % FW_VFS;
424            if (round < service[i]) {
425                service_one(pf, i);
426            }
427        }
428    }
429    for (unsigned i = 0; i < FW_VFS; i++) {
430        ToySlot *s = &pf->slots[i];
431        more |= s->pending.valid || (!s->fenced && !s->killed && s->config_valid &&
432                s->sq_head != s->sq_tail && s->cq_tail - s->cq_head < s->config.cq_entries &&
433                !(pf->fw_hang_vf && toy_fault_target(pf, i)));
434        more |= pf->core.vf[i].state == FW_QUIESCING || pf->core.vf[i].state == FW_RESETTING ||
435                (s->vf && pf->core.vf[i].state == FW_RECOVERING);
436    }
437    status = fw_poll_deadlines(&pf->core);
438    if (status != FW_OK && status != FW_TIMEOUT) {
439        fprintf(stderr, "TOY_DEADLINE_ERROR status=%u\n", status);
440    }
441    toy_mbx_poll(pf);
442    if (more) {
443        toy_schedule(pf);
444    }
