4/* 先確認 VF 與硬體都可配置，再一次扣除 queue、vector 和 credit 配額。 */
5fw_status fw_create(struct fw_context *ctx, fw_u16 vf, fw_u32 reserve)
6{
7    struct fw_hw_snapshot hw;
8    fw_status rc;
9    if (!ctx || vf>=FW_VFS || !reserve || reserve>FW_CAP_CREDIT) return FW_INVAL;
10    if (ctx->vf[vf].state!=FW_FREE) return FW_STATE;
11    rc=fw_hw(ctx,vf,&hw); if (rc!=FW_OK) return rc;
12    if (!hw.fenced || hw.killed || hw.config_valid || hw.outstanding || hw.pci_present) return FW_STATE;
13    if (ctx->used_queues>=FW_VFS || ctx->used_vectors>14 || ctx->used_credit>FW_CAP_CREDIT-reserve) return FW_NOSPC;
14    ctx->vf[vf].state=FW_PROVISIONING;
15    ctx->vf[vf].reserve=reserve; ctx->vf[vf].epoch=hw.epoch;
16    ++ctx->used_queues; ctx->used_vectors+=2; ctx->used_credit+=reserve;
17    fw_note(ctx,"vf_reserved",vf,reserve);
18    return FW_OK;
19}

21/* 只允許 fenced 且沒有未完成工作的 VF 調整配額；先扣舊值再檢查新總量。 */
22fw_status fw_set_reserve(struct fw_context *ctx, fw_u16 vf, fw_u32 reserve)
23{
24    struct fw_hw_snapshot hw;
25    fw_u32 remaining;
26    fw_status rc;
27    if (!ctx || vf>=FW_VFS || !reserve || reserve>FW_CAP_CREDIT) return FW_INVAL;
28    if (ctx->vf[vf].state!=FW_ACTIVE && ctx->vf[vf].state!=FW_RECOVERING) return FW_STATE;
29    rc=fw_hw(ctx,vf,&hw); if (rc!=FW_OK) return rc;
30    if (!hw.fenced || hw.outstanding || hw.killed) return FW_STATE;
31    if (ctx->used_credit<ctx->vf[vf].reserve) return FW_STATE;
32    remaining=ctx->used_credit-ctx->vf[vf].reserve;
33    if (remaining>FW_CAP_CREDIT-reserve) return FW_NOSPC;
34    ctx->used_credit=remaining+reserve; ctx->vf[vf].reserve=reserve;
35    return FW_OK;
36}

38/* CONFIG 完成不等於允許 DMA；epoch、ready_gen、BME 與期限都符合後才開 gate。 */
39fw_status fw_rearm(struct fw_context *ctx, fw_u16 vf, fw_u64 epoch, fw_u32 ready_gen)
40{
41    struct fw_hw_snapshot hw;
42    fw_status rc;
43    if (!ctx || vf>=FW_VFS || !ready_gen) return FW_INVAL;
44    if (ctx->vf[vf].state!=FW_ACTIVE && ctx->vf[vf].state!=FW_RECOVERING) return FW_STATE;
45    rc=fw_hw(ctx,vf,&hw); if (rc!=FW_OK) return rc;
46    if (hw.epoch!=epoch || ctx->vf[vf].epoch!=epoch || !hw.config_valid || !hw.ready_pending ||
47        hw.ready_gen!=ready_gen || hw.killed || !hw.fenced || !hw.bme || hw.outstanding) return FW_STATE;
48    if (ctx->vf[vf].state==FW_RECOVERING && ctx->hal.now_tick(ctx->hal.opaque)>=ctx->vf[vf].deadline) {
49        rc=fw_transition(ctx,vf,FW_VERIFY_FAIL);
50        return rc==FW_OK?FW_TIMEOUT:rc;
51    }
52    rc=ctx->hal.set_gate(ctx->hal.opaque,vf,1);
53    if (rc!=FW_OK) return rc;
54    ctx->vf[vf].state=FW_ACTIVE; ctx->vf[vf].deadline=0;
55    fw_note(ctx,"vf_rearmed",vf,epoch);
56    return FW_OK;
57}
