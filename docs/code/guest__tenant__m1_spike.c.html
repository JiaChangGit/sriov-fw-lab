44static int host_checkpoint(const char *event)
45{
46    static const char expected[]="M1_HOST_CHECKED\n";
47    struct termios original,raw;
48    struct timespec now;
49    struct pollfd input;
50    int fd=-1,changed=0,saved=0,rc=-1;
51    int64_t deadline;
52    size_t received=0;
53    /* 使用 init 配置的協定 UART，不重新打開指向 kernel UART 的 console。
54     * 獨立開啟 stdin 的裝置以保留原 fd flags；先 raw/noecho 再發出事件。 */
55    fd=open("/proc/self/fd/0",O_RDONLY|O_CLOEXEC|O_NONBLOCK);
56    if(fd<0) goto out;
57    if(tcgetattr(fd,&original)) goto out;
58    raw=original;
59    cfmakeraw(&raw);
60    if(tcsetattr(fd,TCSANOW,&raw)) goto out;
61    changed=1;
62    if(tcflush(fd,TCIFLUSH) || clock_gettime(CLOCK_MONOTONIC,&now)) goto out;
63    deadline=(int64_t)now.tv_sec*1000+now.tv_nsec/1000000+10000;
64    printf("{\"event\":\"%s\",\"status\":\"WAITING\",\"expected_token\":\"M1_HOST_CHECKED\",\"timeout_ms\":10000}\n",event);
65    if(fflush(stdout)) goto out;
66    input=(struct pollfd){.fd=fd,.events=POLLIN};
67    while(received<sizeof(expected)-1) {
68        char byte;
69        ssize_t n;
70        int ready,remaining;
71        if(clock_gettime(CLOCK_MONOTONIC,&now)) goto out;
72        remaining=(int)(deadline-((int64_t)now.tv_sec*1000+now.tv_nsec/1000000));
73        if(remaining<=0) { errno=ETIMEDOUT; goto out; }
74        ready=poll(&input,1,remaining);
75        if(ready<0 && errno==EINTR) continue;
76        if(ready<0) goto out;
77        if(!ready) { errno=ETIMEDOUT; goto out; }
78        if(!(input.revents&POLLIN)) { errno=EIO; goto out; }
79        n=read(fd,&byte,1);
80        if(n<0 && (errno==EINTR || errno==EAGAIN)) continue;
81        if(n<0) goto out;
82        if(n!=1) { errno=EIO; goto out; }
83        if(byte!=expected[received]) { errno=EPROTO; goto out; }
84        ++received;
85    }
86    rc=0;
87out:
88    saved=errno;
89    /* 成功與失敗都恢復協定 UART，且不讓清理覆蓋原始失敗原因。 */
90    if(changed && tcsetattr(fd,TCSANOW,&original) && !rc) { saved=errno; rc=-1; }
91    if(fd>=0 && close(fd) && !rc) { saved=errno; rc=-1; }
92    if(rc) {
93        errno=saved;
94        return error(event);
95    }
96    printf("{\"event\":\"M1_HOST_CHECKPOINT_ACK\",\"checkpoint\":\"%s\",\"status\":\"PASS\"}\n",event);
97    return 0;
98}

124static int read_config(const char *bdf, unsigned char config[4096])
125{
126    char path[PATH_MAX];
127    size_t done=0;
128    int fd;
129    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/config",bdf);
130    fd=open(path,O_RDONLY|O_CLOEXEC);
131    if(fd<0) return error("open_config");
132    while(done<4096) {
133        ssize_t n=pread(fd,config+done,4096-done,(off_t)done);
134        if(n<0 && errno==EINTR) continue;
135        if(n<=0) { close(fd); errno=EIO; return error("read_extended_config"); }
136        done+=(size_t)n;
137    }
138    if(close(fd)) return error("close_config");
139    return 0;

142static int standard_cap(const unsigned char *config, unsigned id, unsigned *result)
143{
144    unsigned offset=config[PCI_CAPABILITY_LIST]&0xfcU,seen[64]={0},hops=0;
145    *result=0;
146    while(offset) {
147        if(offset<0x40 || offset>0xfc || seen[offset/4] || ++hops>48) { errno=EPROTO; return error("standard_cap_chain"); }
148        seen[offset/4]=1;
149        if(config[offset]==id) { *result=offset; return 0; }
150        offset=config[offset+1]&0xfcU;
151    }
152    errno=ENOENT; return error("required_standard_cap");
153}

155static int extended_cap(const unsigned char *config, unsigned id, unsigned *result)
156{
157    unsigned offset=0x100,seen[1024]={0},hops=0;
158    *result=0;
159    while(offset) {
160        uint32_t header;
161        if(offset<0x100 || offset>0xffc || (offset&3U) || seen[offset/4] || ++hops>960) { errno=EPROTO; return error("extended_cap_chain"); }
162        seen[offset/4]=1; header=le32(config+offset);
163        /* 即使已找到目標，仍走完鏈結，避免略過後段破損或重複的 capability。 */
164        if(!header || header==UINT32_MAX || !(header&0xffffU) || !((header>>16)&15U)) {
165            errno=EPROTO; return error("extended_cap_header");
166        }
167        if((header&0xffffU)==id) {
168            if(*result) { errno=EPROTO; return error("duplicate_extended_cap"); }
169            *result=offset;
170        }
171        offset=(header>>20)&0xfffU;
172    }
173    if(*result) return 0;
174    errno=ENOENT; return error("required_extended_cap");
175}

196static int pci_caps(const char *bdf, int is_vf, unsigned *first, unsigned *stride)
197{
198    unsigned char config[4096];
199    unsigned msix,pcie,sriov=0,ari,acs;
200    uint32_t table,pba,bar;
201    uint64_t vendor,device;
202    char path[PATH_MAX];
203    if(read_config(bdf,config)) return -1;
204    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/vendor",bdf);
205    if(lab_read_u64(path,&vendor)) return error("enumerated_vendor");
206    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/device",bdf);
207    if(lab_read_u64(path,&device)) return error("enumerated_device");
208    /* VF 的識別字來自 PF SR-IOV capability；原始 config ID 依法可為 ffff。 */
209    if(require(vendor==TOY_PCI_VENDOR && device==(is_vf?TOY_PCI_VF:TOY_PCI_PF) &&
210               (is_vf || (le16(config)==TOY_PCI_VENDOR && le16(config+2)==TOY_PCI_PF)) &&
211               config[PCI_REVISION_ID]==TOY_PCI_REVISION &&
212               (le32(config+PCI_CLASS_REVISION)>>8)==TOY_PCI_CLASS_CODE,"pci_identity")) return -1;
213    if(standard_cap(config,PCI_CAP_ID_MSIX,&msix) || standard_cap(config,PCI_CAP_ID_EXP,&pcie)) return -1;
214    if(extended_cap(config,PCI_EXT_CAP_ID_ARI,&ari) || extended_cap(config,PCI_EXT_CAP_ID_ACS,&acs)) return -1;
215    if(require(ari<=4096-8 && acs<=4096-8 && ari!=acs,"ari_acs_capability_bounds")) return -1;
216    table=le32(config+msix+PCI_MSIX_TABLE); pba=le32(config+msix+PCI_MSIX_PBA); bar=le32(config+PCI_BASE_ADDRESS_0);
217    if(require(msix<=0xf4 && pcie<=0xc4 && (le16(config+msix+PCI_MSIX_FLAGS)&PCI_MSIX_FLAGS_QSIZE)==1 &&
218               (table&7U)==2 && (table&~7U)==0 && (pba&7U)==2 && (pba&~7U)==0x1000 &&
219               (is_vf || (bar&15U)==PCI_BASE_ADDRESS_MEM_TYPE_64),"bar_msix_contract")) return -1;
220    if(is_vf && require(le32(config+pcie+PCI_EXP_DEVCAP)&PCI_EXP_DEVCAP_FLR,"vf_flr_capability")) return -1;
221    if(!is_vf) {
222        if(extended_cap(config,PCI_EXT_CAP_ID_SRIOV,&sriov)) return -1;
223        if(require(sriov<=4096-PCI_EXT_CAP_SRIOV_SIZEOF && le16(config+sriov+PCI_SRIOV_TOTAL_VF)==8 &&
224                   le16(config+sriov+PCI_SRIOV_INITIAL_VF)==8 && le16(config+sriov+PCI_SRIOV_VF_DID)==TOY_PCI_VF &&
225                   le32(config+sriov+PCI_SRIOV_SUP_PGSIZE)!=0 && le32(config+sriov+PCI_SRIOV_SYS_PGSIZE)!=0,"sriov_contract")) return -1;
226        *first=le16(config+sriov+PCI_SRIOV_VF_OFFSET); *stride=le16(config+sriov+PCI_SRIOV_VF_STRIDE);
227        if(require(*first && *stride,"vf_routing_offset_stride")) return -1;
228        printf("{\"event\":\"sriov_capability\",\"pf\":\"%s\",\"offset\":%u,\"total_vfs\":8,\"initial_vfs\":8,\"first_vf_offset\":%u,\"stride\":%u,\"supported_page_sizes\":%u,\"system_page_size\":%u}\n",bdf,sriov,*first,*stride,le32(config+sriov+PCI_SRIOV_SUP_PGSIZE),le32(config+sriov+PCI_SRIOV_SYS_PGSIZE));
229    }
230    printf("{\"event\":\"function_caps\",\"bdf\":\"%s\",\"kind\":\"%s\",\"msix_offset\":%u,\"pcie_offset\":%u,\"ari_offset\":%u,\"acs_offset\":%u",bdf,is_vf?"VF":"PF",msix,pcie,ari,acs);
231    if(!is_vf) printf(",\"sriov_offset\":%u",sriov);
232    printf(",\"vectors\":2,\"flr\":%s,\"raw_vendor\":%u,\"raw_device\":%u,\"enumerated_vendor\":%" PRIu64 ",\"enumerated_device\":%" PRIu64 "}\n",(le32(config+pcie+PCI_EXP_DEVCAP)&PCI_EXP_DEVCAP_FLR)?"true":"false",le16(config),le16(config+2),vendor,device);
233    return resources(bdf,is_vf?TOY_LIMITS_VF_BAR0_BYTES:TOY_LIMITS_PF_BAR0_BYTES);
234}

324        printf("]}");
325    }
326    puts("],\"pf_excluded\":true,\"complete_members_logged\":true}");

341static int type1_probe(int pf, char bdfs[VF_COUNT][32], const unsigned groups[VF_COUNT])
342{
343    struct lab_vfio vf[VF_COUNT];
344    struct toy_vf_info before[VF_COUNT],after;
345    unsigned i,j,opened=0;
346    int failed=0,baseline=fd_count();
347    for(i=0;i<VF_COUNT;++i) lab_vfio_init(&vf[i]);
348    for(i=0;i<VF_COUNT;++i) {
349        ++opened;
350        if(lab_vfio_open(&vf[i],bdfs[i],groups[i]) || lab_vfio_map_probe(&vf[i])) { error("vfio_open_type1_map"); failed=1; break; }
351        for(j=0;j<i;++j) if(require(vf[i].container!=vf[j].container,"separate_containers")) { failed=1; break; }
352        if(failed) break;
353    }
354    if(!failed) {
355        for(i=0;i<VF_COUNT;++i) if(get_vf(pf,i,&before[i])) { failed=1; break; }
356    }
357    if(!failed) {
358        if(ioctl(vf[0].device,VFIO_DEVICE_RESET)) { error("vfio_device_reset"); failed=1; }
359        for(i=0;i<VF_COUNT && !failed;++i) {
360            if(get_vf(pf,i,&after)) { failed=1; break; }
361            if(require((i==0?after.epoch>before[i].epoch:after.epoch==before[i].epoch) &&
362                       (after.hw_flags&TOY_HW_FLAGS_FENCED) && !(after.hw_flags&TOY_HW_FLAGS_CONFIG_VALID),"flr_epoch_fence_isolation")) failed=1;
363            printf("{\"event\":\"vfio_reset_observed\",\"vf\":\"%s\",\"target\":%s,\"before_epoch\":%llu,\"after_epoch\":%llu,\"hw_flags\":%u}\n",bdfs[i],i==0?"true":"false",(unsigned long long)before[i].epoch,(unsigned long long)after.epoch,after.hw_flags);
364        }
365    }
366    while(opened) if(lab_vfio_close(&vf[--opened])) { error("vfio_cleanup"); failed=1; }
367    if(require(fd_count()==baseline,"vfio_no_fd_leak")) failed=1;
368    if(failed) return -1;
369    puts("{\"id\":\"M1_TYPE1\",\"status\":\"PASS\",\"viable_groups\":4,\"simultaneous_containers\":4,\"maps\":4,\"exact_unmaps\":4,\"remaining_fds\":0,\"remaining_maps\":0,\"vfio_reset_epoch_checked\":true}");
370    return 0;
371}

402        if(require(vf.state==TOY_STATES_FREE && !vf.reserve && !vf.outstanding &&
403                   (vf.hw_flags&TOY_HW_FLAGS_FENCED) && !(vf.hw_flags&(TOY_HW_FLAGS_CONFIG_VALID|TOY_HW_FLAGS_PCI_PRESENT|TOY_HW_FLAGS_KILLED)),"disabled_ledger_free")) return -1;
404        total+=vf.reserve;
405        snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/virtfn%u",bdf,i);
406        if(!lstat(path,&st) || errno!=ENOENT) return require(0,"vf_link_removed");
407    }
408    printf("{\"event\":\"%s\",\"cycle\":%u,\"enabled\":0,\"ledger_credit\":%u,\"outstanding\":0,\"vf_links\":0}\n",cycle?"disable_snapshot":"warmup_disable_snapshot",cycle,total);
409    return 0;
410}

412int main(void)
413{
414    char pf_bdf[32]={0},bdfs[VF_COUNT][32],path[PATH_MAX],driver[64];
415    unsigned groups[VF_COUNT],first=0,stride=0,cycle,i;
416    int pf=-1,enabled=0,baseline=-1,failed=0;
417    struct toy_info info;
418    setvbuf(stdout,NULL,_IOLBF,0);
419    if(find_pf(pf_bdf)) goto fail;
420    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/driver",pf_bdf);
421    if(lab_link_basename(path,driver,sizeof(driver)) || require(!strcmp(driver,"toy_pf"),"pf_owned_by_toy_driver")) goto fail;
422    if(lab_ensure_char("/sys/class/misc/toy_pf0/dev","/dev/toy_pf0")) { error("pf_device_node"); goto fail; }
423    pf=open("/dev/toy_pf0",O_RDWR|O_CLOEXEC);
424    if(pf<0) { error("open_pf_control"); goto fail; }
425    if(get_info(pf,&info) || require(info.caps_vfs==8 && info.caps_queues==8 && info.caps_vectors==16 &&
426       info.caps_credits==64 && (info.features&7U)==7U && info.session && !info.enabled_vfs,"pf_capabilities") ||
427       pci_caps(pf_bdf,0,&first,&stride)) goto fail;
428    if(host_checkpoint("M1_COLD_CHECKPOINT")) goto fail;
429    baseline=fd_count();
430    if(require(baseline>=0,"fd_baseline")) goto fail;
431    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/sriov_numvfs",pf_bdf);
432    /* 首次使用會建立 vIOMMU 每個 requester 的快取；另做完整 100 次穩態循環。 */
433    for(cycle=0;cycle<=CYCLES;++cycle) {
434        if(lab_write_text(path,"4\n")) { error("enable_four_vfs"); goto fail; }
435        enabled=1;
436        if(enabled_snapshot(pf,cycle) || vf_links(pf_bdf,bdfs,groups,first,stride,cycle==0)) goto fail;
437        if(cycle==0) {
438            for(i=0;i<VF_COUNT;++i) if(pci_caps(bdfs[i],1,&first,&stride)) goto fail;
439            puts("{\"id\":\"M1_CAPS\",\"status\":\"PASS\",\"pf_vectors\":2,\"vf_vectors\":2,\"vfs\":4,\"unique_requesters\":true,\"flr_declared\":true}");
440            if(group_members(pf_bdf,bdfs,groups)) goto fail;
441            if(type1_probe(pf,bdfs,groups)) goto fail;
442        }
443        if(lab_write_text(path,"0\n")) { error("disable_four_vfs"); goto fail; }
444        enabled=0;
445        if(disabled_snapshot(pf,pf_bdf,cycle) || require(fd_count()==baseline,"cycle_fd_no_growth")) goto fail;
446        if(cycle==0 && host_checkpoint("M1_INITIAL_CHECKPOINT")) goto fail;
447    }
448    if(host_checkpoint("M1_FINAL_CHECKPOINT")) goto fail;
449    puts("{\"id\":\"T01\",\"status\":\"PASS\",\"cycles\":100,\"vfs_per_cycle\":4,\"final_ledger\":0,\"remaining_vf_functions\":0,\"remaining_vfio_fds\":0,\"remaining_vfio_maps\":0,\"fd_growth\":0}");
450    if(close(pf)) { error("close_pf"); pf=-1; goto fail; }
451    puts("LAB_RESULT=PASS");
452    return 0;
453fail:
454    failed=errno;
455    if(enabled && lab_write_text(path,"0\n")) error("failure_disable_cleanup");
456    if(pf>=0 && close(pf)) error("failure_pf_close");
457    printf("{\"id\":\"M1_FAILURE\",\"status\":\"FAIL\",\"errno\":%d}\n",failed);
458    puts("LAB_RESULT=FAIL");
459    return 1;
