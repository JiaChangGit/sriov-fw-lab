14static int exchange(int fd,struct lab_message *m,uint16_t sent,uint16_t expected,struct owner *inspect)
15{
16    struct lab_message reply;
17    m->type=sent;
18    if (lab_send(fd,m,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
19    for (;;) {
20        if (lab_recv(fd,&reply,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
21        if (reply.type!=LAB_INSPECT||!inspect) break;
22        uint32_t digests[OWNER_MAPS];
23        if (reply.request_id!=m->request_id||lab_tuple(&reply,m->vf,m->session,m->epoch,m->lease_token,m->operation)||
24            reply.ready_gen!=m->ready_gen||
25            owner_unchanged(inspect,digests)) { errno=EPROTO; return -1; }
26        reply.type=LAB_INSPECT_REPLY; reply.payload_len=16; memset(reply.payload,0,16);
27        for (unsigned i=0;i<OWNER_MAPS;i++) lab_put32(reply.payload+i*4,digests[i]);
28        if (lab_send(fd,&reply,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
29    }
30    if (reply.type!=expected||reply.request_id!=m->request_id||
31        lab_tuple(&reply,m->vf,m->session,m->epoch,m->lease_token,m->operation)||
32        (sent!=LAB_HELLO&&reply.ready_gen!=m->ready_gen)) { errno=EPROTO; return -1; }
33    *m=reply; return 0;
34}
35static int open_fds(void)
36{
37    DIR *directory=opendir("/proc/self/fd");
38    struct dirent *entry;
39    int count=0;
40    if (!directory) return -1;
41    while ((entry=readdir(directory))) {
42        if (entry->d_name[0]=='.') continue;
43        if (atoi(entry->d_name)!=dirfd(directory)) ++count;
44    }
45    closedir(directory); return count;
46}

48static int healthy_work(int fd,struct lab_message *m,struct owner *o)
49{
50    int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
51    while (lab_now_ms()<deadline) {
52        struct pollfd p={.fd=fd,.events=POLLIN};
53        struct lab_message query;
54        if (owner_workload(o,256)) return -1;
55        int rc=poll(&p,1,0);
56        if (rc<0&&errno==EINTR) continue;
57        if (rc<0) return -1;
58        if (!rc) continue;
59        if (!(p.revents&POLLIN)||lab_recv(fd,&query,deadline)||query.request_id!=m->request_id||
60            lab_tuple(&query,m->vf,m->session,m->epoch,m->lease_token,m->operation)||query.ready_gen!=m->ready_gen) { errno=EPROTO; return -1; }
61        if (query.type==LAB_STOP_REQUEST) return 0;
62        if (query.type!=LAB_INSPECT) { errno=EPROTO; return -1; }
63        uint32_t digests[OWNER_MAPS],flags;
64        if (owner_freeze(o)||owner_unchanged(o,digests)||owner_latch(o,&flags)||
65            o->epoch!=m->epoch||o->session!=m->session||(flags&3u)) { errno=EPROTO; return -1; }
66        query.type=LAB_INSPECT_REPLY; query.payload_len=16; memset(query.payload,0,16);
67        for (unsigned i=0;i<OWNER_MAPS;i++) lab_put32(query.payload+i*4,digests[i]);
68        printf("{\"event\":\"%s\",\"pid\":%d,\"epoch\":%" PRIu64 ",\"session\":%" PRIu64 ",\"completed\":%" PRIu64 ",\"flags\":%u}\n",o->recovery_case?"RECOVERY_HEALTHY_PROGRESS":"T06_HEALTHY_PROGRESS",getpid(),o->epoch,o->session,o->completed,flags);
69        if (lab_send(fd,&query,deadline)) return -1;
70    }
71    errno=ETIMEDOUT; return -1;
72}
