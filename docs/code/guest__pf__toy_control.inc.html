2static struct toy_uapi_header toy_header(size_t size)
3{
4    return (struct toy_uapi_header){ size, TOY_UAPI_MAJOR, TOY_UAPI_MINOR };

7static int toy_header_valid(const struct toy_uapi_header *hdr, size_t size)
8{
9    if (hdr->size != size)
10        return -EINVAL;
11    if (hdr->abi_major != TOY_UAPI_MAJOR || hdr->abi_minor != TOY_UAPI_MINOR)
12        return -EPROTONOSUPPORT;
13    return 0;
14}

16static bool toy_reserved_zero(const u32 *reserved, size_t count)
17{
18    size_t i;
19    for (i = 0; i < count; ++i)
20        if (reserved[i])
21            return false;
22    return true;
23}

25static void toy_snapshot_vf(struct toy_pf *pf, struct toy_file *ctx,
26                            unsigned int vf, struct toy_vf_info *out)
27{
28    u32 off = TOY_VF_VIEW(vf, 0);
29    memset(out, 0, sizeof(*out));
30    out->hdr = toy_header(sizeof(*out));
31    out->vf = vf;
32    out->session = toy_read64(pf, TOY_PF_SESSION_LO);
33    out->epoch = toy_read64(pf, off + TOY_PF_VF_VIEW_EPOCH_LO);
34    out->state = toy_read(pf, off + TOY_PF_VF_VIEW_STATE);
35    out->hw_flags = toy_read(pf, off + TOY_PF_VF_VIEW_HW_STATUS);
36    out->reserve = toy_read(pf, off + TOY_PF_VF_VIEW_RESERVE);
37    out->outstanding = toy_read(pf, off + TOY_PF_VF_VIEW_OUTSTANDING);
38    out->ready_gen = toy_read(pf, off + TOY_PF_VF_VIEW_READY_GEN);
39    if (ctx && pf->leases[vf].owner == ctx)
40        out->lease_token = pf->leases[vf].token;
41}

43static bool toy_has_leases(struct toy_pf *pf)
44{
45    unsigned int i;
46    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i)

52/* fanout 只在 admin_mutex 下執行；IRQ 不接觸 context list。 */
53static void toy_publish(struct toy_pf *pf, u32 type, unsigned int vf, u32 detail,
54                        const struct toy_vf_info *snapshot)
55{
56    struct toy_file *ctx;
57    list_for_each_entry(ctx, &pf->files, link) {
58        struct toy_event event = { .hdr = toy_header(sizeof(event)), .type = type,
59                                   .vf = vf, .detail = detail };
60        unsigned long flags;
61        if (snapshot) {
62            event.session = snapshot->session;
63            event.epoch = snapshot->epoch;
64            event.ready_gen = snapshot->ready_gen;
65            event.state = snapshot->state;
66            event.hw_flags = snapshot->hw_flags;
67        }
68        if (vf < TOY_PCI_TOTAL_VFS && pf->leases[vf].owner == ctx)
69            event.lease_token = pf->leases[vf].token;
70        spin_lock_irqsave(&ctx->event_lock, flags);
71        if (ctx->count == TOY_EVENT_CAPACITY) {
72            ctx->head = (ctx->head + 1) % TOY_EVENT_CAPACITY;
73            --ctx->count;
74            if (ctx->lost != U32_MAX)
75                ++ctx->lost;
76        }
77        ctx->events[(ctx->head + ctx->count) % TOY_EVENT_CAPACITY] = event;
78        ++ctx->count;
79        spin_unlock_irqrestore(&ctx->event_lock, flags);
80        wake_up_interruptible(&ctx->event_wait);
81    }
82}

84static void toy_notify_work(struct work_struct *work)
85{
86    struct toy_pf *pf = container_of(work, struct toy_pf, notify_work);
87    unsigned int i;
88    mutex_lock(&pf->admin_mutex);
89    if (pf->dying)
90        goto out;
91    /* 一次 coherent snapshot 覆蓋合併 IRQ；只報實際觀察到的 tuple 變化。 */
92    atomic_xchg(&pf->pending_events, 0);
93    toy_write(pf, TOY_PF_SNAPSHOT, 1);
94    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i) {
95        struct toy_vf_info now;
96        u32 fault;
97        toy_snapshot_vf(pf, NULL, i, &now);
98        fault = toy_read(pf, TOY_VF_VIEW(i, TOY_PF_VF_VIEW_LAST_FAULT));
99        if (pf->leases[i].token) {
100            if (now.epoch != pf->observed[i].epoch ||
101                now.session != pf->observed[i].session) {
102                pf->leases[i].fence_valid = false;
103                toy_publish(pf, TOY_EVENT_FLR, i, 0, &now);
104            }
105            if ((now.state == TOY_STATES_FAILED && pf->observed[i].state != now.state) ||
106                fault != pf->observed_fault[i] ||
107                ((now.hw_flags ^ pf->observed[i].hw_flags) & TOY_HW_FLAGS_KILLED))
108                toy_publish(pf, TOY_EVENT_FAULT, i, fault, &now);
109        }
110        pf->observed[i] = now;
111        pf->observed_fault[i] = fault;
112    }
113out:
114    mutex_unlock(&pf->admin_mutex);
115}

117/* 一次 open 建立一個 context；dup 共用同一 struct file，不會建立第二個 lease owner。 */
118static int toy_open(struct inode *inode, struct file *file)
119{
120    struct miscdevice *misc = file->private_data;
121    struct toy_pf *pf = container_of(misc, struct toy_pf, misc);
122    struct toy_file *ctx;
123    int ret = 0;
124    ctx = kzalloc(sizeof(*ctx), GFP_KERNEL);
125    if (!ctx)
126        return -ENOMEM;
127    ctx->pf = pf;
128    spin_lock_init(&ctx->event_lock);
129    init_waitqueue_head(&ctx->event_wait);
130    mutex_init(&ctx->read_mutex);
131    mutex_lock(&pf->admin_mutex);
132    if (!pf->accepting || pf->dying)
133        ret = -ENODEV;
134    else {
135        kref_get(&pf->ref);
136        list_add_tail(&ctx->link, &pf->files);
137        file->private_data = ctx;
138        nonseekable_open(inode, file);
139    }
140    mutex_unlock(&pf->admin_mutex);
141    if (ret)
142        kfree(ctx);
143    return ret;
144}

146/* 控制 fd 關閉不代表 VFIO owner 已退出；遺留 lease 標為 orphaned，不能直接當作已釋放。 */
147static int toy_release_file(struct inode *inode, struct file *file)
148{
149    struct toy_file *ctx = file->private_data;
150    struct toy_pf *pf = ctx->pf;
151    unsigned int i;
152    mutex_lock(&pf->admin_mutex);
153    list_del(&ctx->link);
154    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i)
155        if (pf->leases[i].owner == ctx) {
156            if (!pf->dying && toy_fence(pf, i))
157                toy_kill(pf, i);
158            pf->leases[i].owner = NULL;
159            pf->leases[i].orphaned = true;
160            pf->leases[i].fence_valid = false;
161            pr_err("TOY_LEASE_ORPHAN vf=%u token=%llu\n", i, pf->leases[i].token);
162        }
163    mutex_unlock(&pf->admin_mutex);
164    kfree(ctx);
165    kref_put(&pf->ref, toy_release_ref);
166    return 0;
167}

169static void toy_ctl_snapshot(struct toy_pf *pf, struct toy_file *ctx,
170                              struct toy_vf_ctl *ctl)
171{
172    struct toy_vf_info snapshot;
173    toy_write(pf, TOY_PF_SNAPSHOT, 1);
174    ctl->session = toy_read64(pf, TOY_PF_SESSION_LO);
175    if (ctl->vf == TOY_CTL_GLOBAL)
176        return;
177    toy_snapshot_vf(pf, ctx, ctl->vf, &snapshot);
178    ctl->epoch = snapshot.epoch;
179    ctl->lease_token = snapshot.lease_token;
180    ctl->ready_gen = snapshot.ready_gen;
181    ctl->state = snapshot.state;
182    ctl->hw_flags = snapshot.hw_flags;
183    ctl->outstanding = snapshot.outstanding;
184}

186static int toy_ctl_shape(unsigned int cmd, const struct toy_vf_ctl *ctl)
187{
188    int ret = toy_header_valid(&ctl->hdr, sizeof(*ctl));
189    if (ret)
190        return ret;
191    if (!ctl->operation || !toy_reserved_zero(ctl->reserved, 3) || ctl->state ||
192        ctl->hw_flags || ctl->outstanding || ctl->fw_status ||
193        (cmd != TOY_REARM_VF && ctl->ready_gen))
194        return -EINVAL;
195    if (ctl->flags != (cmd == TOY_LEASE_RELEASE ? TOY_CTL_OWNER_CLOSED : 0))
196        return -EINVAL;
197    if (cmd == TOY_GLOBAL_RECOVER)
198        return ctl->vf == TOY_CTL_GLOBAL && !ctl->epoch && !ctl->lease_token ? 0 : -EINVAL;
199    if (ctl->vf >= TOY_PCI_TOTAL_VFS || (cmd == TOY_LEASE_ACQUIRE && ctl->lease_token))
200        return -EINVAL;
201    return 0;
202}

204static int toy_global_recover(struct toy_pf *pf, u32 *fw_status)
205{
206    const u32 args[4] = { TOY_PROTOCOL_MAJOR, TOY_PROTOCOL_MINOR, TOY_PROTOCOL_MINOR, 0 };
207    struct toy_vf_info before[TOY_PCI_TOTAL_VFS], after;
208    u32 reply[4];
209    unsigned int i;
210    int ret;
211    if (toy_has_leases(pf) || pf->enabled || pci_num_vf(pf->pdev))
212        return -EBUSY;
213    toy_write(pf, TOY_PF_SNAPSHOT, 1);
214    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i)
215        toy_snapshot_vf(pf, NULL, i, &before[i]);
216    toy_fence_all(pf);
217    toy_write(pf, TOY_PF_CONTROL, TOY_PF_CONTROL_GLOBAL_RESET);
218    *fw_status = toy_read(pf, TOY_PF_HW_RESULT);
219    ret = toy_status_errno(*fw_status);
220    toy_write(pf, TOY_PF_SNAPSHOT, 1);
221    if (ret)
222        return ret;
223    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i) {
224        toy_snapshot_vf(pf, NULL, i, &after);
225        if (after.session <= before[i].session || after.epoch <= before[i].epoch ||
226            after.state != TOY_STATES_FREE || after.reserve || after.outstanding ||
227            after.hw_flags != TOY_HW_FLAGS_FENCED) {
228            toy_fence_all(pf);
229            return -EIO;
230        }
231        pf->observed[i] = after;
232        pf->observed_fault[i] = 0;
233    }
234    /* 新 session 的 mailbox seq 可重起；host lease/fence/operation 都不歸零。 */
235    pf->seq = 0;
236    ret = toy_mailbox_status(pf, TOY_OPCODES_VERSION_NEG, TOY_GLOBAL, args, reply, fw_status);
237    return !ret && reply[0] != TOY_ABI_VERSION ? -EPROTO : ret;
238}

240static int toy_execute_ctl(struct toy_file *ctx, unsigned int cmd,
241                           const struct toy_vf_ctl *input, struct toy_vf_ctl *result)
242{
243    struct toy_pf *pf = ctx->pf;
244    struct toy_lease *lease;
245    struct toy_vf_info now;
246    u32 args[4];
247    int ret;
248    *result = *input;
249    toy_ctl_snapshot(pf, ctx, result);
250    if (pf->transaction)
251        return -EBUSY;
252    if (input->session != result->session || input->epoch != result->epoch)
253        return -ESTALE;
254    if (cmd == TOY_GLOBAL_RECOVER) {
255        ret = toy_global_recover(pf, &result->fw_status);
256        toy_ctl_snapshot(pf, ctx, result);
257        return ret;
258    }
259    lease = &pf->leases[input->vf];
260    if (cmd == TOY_LEASE_ACQUIRE) {
261        if (lease->token || input->vf >= pf->enabled ||
262            (result->hw_flags & (TOY_HW_FLAGS_FENCED | TOY_HW_FLAGS_PCI_PRESENT |
263                                TOY_HW_FLAGS_KILLED)) !=
264                (TOY_HW_FLAGS_FENCED | TOY_HW_FLAGS_PCI_PRESENT) ||
265            result->state == TOY_STATES_FREE || result->state == TOY_STATES_FAILED)
266            return -EBUSY;
267        if (pf->next_lease == U64_MAX) {
268            toy_kill(pf, input->vf);
269            return -EOVERFLOW;
270        }
271        lease->owner = ctx;
272        lease->token = ++pf->next_lease;
273        lease->orphaned = false;
274        lease->fence_valid = false;
275        result->lease_token = lease->token;
276        toy_snapshot_vf(pf, NULL, input->vf, &pf->observed[input->vf]);
277        pf->observed_fault[input->vf] = toy_read(pf,
278            TOY_VF_VIEW(input->vf, TOY_PF_VF_VIEW_LAST_FAULT));
279        return 0;
280    }
281    if (lease->owner != ctx || !input->lease_token || input->lease_token != lease->token)
282        return -EPERM;
283    switch (cmd) {
284    case TOY_LEASE_RELEASE:
285        if (!(result->hw_flags & TOY_HW_FLAGS_FENCED) || result->outstanding)
286            return -EBUSY;
287        memset(lease, 0, sizeof(*lease));
288        result->lease_token = 0;
289        return 0;
290    case TOY_FENCE_VF:
291        lease->fence_valid = false;
292        ret = toy_fence_status(pf, input->vf, &result->fw_status);
293        toy_ctl_snapshot(pf, ctx, result);
294        if (!ret && (result->epoch <= input->epoch || result->session != input->session))
295            ret = -EIO;
296        if (!ret) {
297            lease->fence_valid = true;
298            lease->fence_epoch = result->epoch;
299            lease->fence_session = result->session;
300        }
301        break;
302    case TOY_RESET_VF:
303        if (!lease->fence_valid || lease->fence_epoch != input->epoch ||
304            lease->fence_session != input->session)
305            return -ESTALE;
306        if (result->state != TOY_STATES_ACTIVE || !(result->hw_flags & TOY_HW_FLAGS_FENCED))
307            return -EBUSY;
308        /* 即使 ACK 遺失也消耗 fence；重送由原請求 cache 處理。 */
309        lease->fence_valid = false;
310        ret = toy_mailbox_status(pf, TOY_OPCODES_QUIESCE, input->vf, NULL, NULL,
311                                 &result->fw_status);
312        if (!ret)
313            ret = toy_mailbox_status(pf, TOY_OPCODES_RESET, input->vf, NULL, NULL,
314                                     &result->fw_status);
315        if (ret)
316            toy_kill(pf, input->vf);
317        toy_ctl_snapshot(pf, ctx, result);
318        if (!ret && (result->state != TOY_STATES_RECOVERING || result->outstanding ||
319            !(result->hw_flags & TOY_HW_FLAGS_FENCED) ||
320            (result->hw_flags & (TOY_HW_FLAGS_CONFIG_VALID | TOY_HW_FLAGS_KILLED)) ||
321            result->session != input->session || result->epoch != input->epoch))
322            ret = -EIO;
323        toy_snapshot_vf(pf, NULL, input->vf, &now);
324        toy_publish(pf, ret ? TOY_EVENT_FAULT : TOY_EVENT_RESET, input->vf, -ret, &now);
325        break;
326    case TOY_REARM_VF:
327        if (!input->ready_gen || input->ready_gen != result->ready_gen)
328            return -ESTALE;
329        if (result->outstanding ||
330            (result->state != TOY_STATES_ACTIVE && result->state != TOY_STATES_RECOVERING) ||
331            (result->hw_flags & (TOY_HW_FLAGS_READY_PENDING | TOY_HW_FLAGS_CONFIG_VALID |
332             TOY_HW_FLAGS_BME | TOY_HW_FLAGS_FENCED | TOY_HW_FLAGS_KILLED)) !=
333            (TOY_HW_FLAGS_READY_PENDING | TOY_HW_FLAGS_CONFIG_VALID |
334             TOY_HW_FLAGS_BME | TOY_HW_FLAGS_FENCED))
335            return -EBUSY;
336        args[0] = input->epoch;
337        args[1] = input->epoch >> 32;
338        args[2] = input->ready_gen;
339        args[3] = 0;
340        ret = toy_mailbox_status(pf, TOY_OPCODES_REARM, input->vf, args, NULL,
341                                 &result->fw_status);
342        toy_ctl_snapshot(pf, ctx, result);
343        if (!ret && (result->state != TOY_STATES_ACTIVE ||
344            (result->hw_flags & (TOY_HW_FLAGS_FENCED | TOY_HW_FLAGS_KILLED)) ||
345            !(result->hw_flags & TOY_HW_FLAGS_CONFIG_VALID) ||
346            result->epoch != input->epoch || result->session != input->session)) {
347            toy_kill(pf, input->vf);
348            toy_ctl_snapshot(pf, ctx, result);
349            ret = -EIO;
350        }
351        lease->fence_valid = false;
352        break;
353    default:
354        return -ENOTTY;
355    }
356    toy_snapshot_vf(pf, NULL, input->vf, &pf->observed[input->vf]);
357    return ret;
358}

360/* 先比對 operation/request 的重送，再執行 live-state 檢查；重試可取回已產生副作用的結果。 */
361static long toy_control_ioctl(struct toy_file *ctx, unsigned int cmd, void __user *user)
362{
363    struct toy_vf_ctl input;
364    int ret;
365    if (copy_from_user(&input, user, sizeof(input)))
366        return -EFAULT;
367    ret = toy_ctl_shape(cmd, &input);
368    if (ret)
369        return ret;
370    if (ctx->replay_valid && input.operation <= ctx->request.operation) {
371        if (input.operation < ctx->request.operation)
372            return -ESTALE;
373        if (cmd != ctx->replay_cmd || memcmp(&input, &ctx->request, sizeof(input)))
374            return -EPROTO;
375    } else {
376        ctx->request = input;
377        ctx->replay_cmd = cmd;
378        ctx->result_errno = toy_execute_ctl(ctx, cmd, &input, &ctx->result);
379        ctx->replay_valid = true;
380        pr_info("TOY_CTL operation=%llu cmd=%u vf=%u errno=%d fw_status=%u session=%llu epoch=%llu flags=%#x\n",
381            input.operation, _IOC_NR(cmd), input.vf, ctx->result_errno,
382            ctx->result.fw_status, ctx->result.session, ctx->result.epoch, ctx->result.hw_flags);
383    }
384    /* 負 errno 也複製完整結果；EFAULT 不可取消已執行操作或清掉 cache。 */
385    if (copy_to_user(user, &ctx->result, sizeof(ctx->result)))
386        return -EFAULT;
387    return ctx->result_errno;
388}

390static long toy_ioctl(struct file *file, unsigned int cmd, unsigned long arg)
391{
392    struct toy_file *ctx = file->private_data;
393    struct toy_pf *pf = ctx->pf;
394    void __user *user = (void __user *)arg;
395    struct toy_info info;
396    struct toy_vf_info vf;
397    unsigned int index;
398    int ret = 0;
399    mutex_lock(&pf->admin_mutex);
400    if (pf->dying) {
401        ret = -ENODEV;
402        goto out;
403    }
404    switch (cmd) {
405    case TOY_GET_INFO:
406        if (copy_from_user(&info, user, sizeof(info))) { ret = -EFAULT; break; }
407        ret = toy_header_valid(&info.hdr, sizeof(info));
408        if (ret || !toy_reserved_zero(info.reserved, 4)) { ret = ret ?: -EINVAL; break; }
409        memset(&info, 0, sizeof(info));
410        info.hdr = toy_header(sizeof(info));
411        toy_write(pf, TOY_PF_SNAPSHOT, 1);
412        info.session = toy_read64(pf, TOY_PF_SESSION_LO);
413        info.caps_vfs = toy_read(pf, TOY_PF_CAP_VFS);
414        info.caps_queues = toy_read(pf, TOY_PF_CAP_QUEUES);
415        info.caps_vectors = toy_read(pf, TOY_PF_CAP_VECTORS);
416        info.caps_credits = toy_read(pf, TOY_PF_CAP_CREDIT);
417        info.features = toy_read(pf, TOY_PF_CAP_FEATURES);
418        info.enabled_vfs = pf->enabled;
419        if (copy_to_user(user, &info, sizeof(info))) ret = -EFAULT;
420        break;
421    case TOY_GET_VF:
422        if (copy_from_user(&vf, user, sizeof(vf))) { ret = -EFAULT; break; }
423        ret = toy_header_valid(&vf.hdr, sizeof(vf));
424        if (ret || !toy_reserved_zero(vf.reserved, 4) || vf.vf >= TOY_PCI_TOTAL_VFS) {
425            ret = ret ?: -EINVAL; break;
426        }
427        index = vf.vf;
428        toy_write(pf, TOY_PF_SNAPSHOT, 1);
429        toy_snapshot_vf(pf, ctx, index, &vf);
430        if (copy_to_user(user, &vf, sizeof(vf))) ret = -EFAULT;
431        break;
432    case TOY_LEASE_ACQUIRE: case TOY_LEASE_RELEASE: case TOY_FENCE_VF:
433    case TOY_RESET_VF: case TOY_REARM_VF: case TOY_GLOBAL_RECOVER:
434        ret = toy_control_ioctl(ctx, cmd, user);
435        break;
436    default:
437        ret = -ENOTTY;
438    }
439out:
440    mutex_unlock(&pf->admin_mutex);
441    return ret;
442}

444static bool toy_event_available(struct toy_file *ctx)
445{
446    return READ_ONCE(ctx->stash_valid) || READ_ONCE(ctx->lost) || READ_ONCE(ctx->count);
447}

449/* read_mutex 序列化讀者；event_lock 只保護 queue/stash，copy_to_user 必須在 spinlock 外執行。 */
450static ssize_t toy_read_file(struct file *file, char __user *buffer,
451                             size_t length, loff_t *offset)
452{
453    struct toy_file *ctx = file->private_data;
454    unsigned long flags;
455    int ret;
456    if (READ_ONCE(ctx->pf->dying))
457        return -ENODEV;
458    if (length < sizeof(struct toy_event))
459        return -EINVAL;
460    ret = mutex_lock_interruptible(&ctx->read_mutex);
461    if (ret)
462        return ret;
463    while (!toy_event_available(ctx)) {
464        if (READ_ONCE(ctx->pf->dying)) { ret = -ENODEV; goto out; }
465        if (file->f_flags & O_NONBLOCK) { ret = -EAGAIN; goto out; }
466        ret = wait_event_interruptible(ctx->event_wait,
467            READ_ONCE(ctx->pf->dying) || toy_event_available(ctx));
468        if (ret)
469            goto out;
470    }
471    if (READ_ONCE(ctx->pf->dying)) { ret = -ENODEV; goto out; }
472    spin_lock_irqsave(&ctx->event_lock, flags);
473    if (!ctx->stash_valid) {
474        if (ctx->event_seq == U64_MAX) {
475            spin_unlock_irqrestore(&ctx->event_lock, flags);
476            ret = -EOVERFLOW;
477            goto out;
478        }
479        if (ctx->lost) {
480            memset(&ctx->stash, 0, sizeof(ctx->stash));
481            ctx->stash.hdr = toy_header(sizeof(ctx->stash));
482            ctx->stash.type = TOY_EVENT_LOST_EVENTS;
483            ctx->stash.vf = TOY_CTL_GLOBAL;
484            ctx->stash.detail = ctx->lost;
485            ctx->lost = 0;
486        } else {
487            ctx->stash = ctx->events[ctx->head];
488            ctx->head = (ctx->head + 1) % TOY_EVENT_CAPACITY;
489            --ctx->count;
490        }
491        ctx->stash.event_seq = ++ctx->event_seq;
492        ctx->stash_valid = true;
493    }
494    spin_unlock_irqrestore(&ctx->event_lock, flags);
495    ret = copy_to_user(buffer, &ctx->stash, sizeof(ctx->stash)) ? -EFAULT : sizeof(ctx->stash);
496    if (ret > 0) {
497        spin_lock_irqsave(&ctx->event_lock, flags);
498        ctx->stash_valid = false;
499        spin_unlock_irqrestore(&ctx->event_lock, flags);
500    }
501    if (READ_ONCE(ctx->pf->dying))
502        ret = -ENODEV;
503out:
504    mutex_unlock(&ctx->read_mutex);
505    return ret;
506}

508static __poll_t toy_poll(struct file *file, poll_table *wait)
509{
510    struct toy_file *ctx = file->private_data;
511    poll_wait(file, &ctx->event_wait, wait);
512    if (READ_ONCE(ctx->pf->dying))
513        return EPOLLHUP | EPOLLERR;
514    return toy_event_available(ctx) ? EPOLLIN | EPOLLRDNORM : 0;
515}
