10int owner_reset_clear(struct owner *o,const char *cause)
11{
12    static const uint32_t registers[]={TOY_VF_SQ_BASE_LO,TOY_VF_SQ_BASE_HI,TOY_VF_SQ_ENTRIES,
13        TOY_VF_CQ_BASE_LO,TOY_VF_CQ_BASE_HI,TOY_VF_CQ_ENTRIES,TOY_VF_SQ_HEAD,TOY_VF_SQ_TAIL,
14        TOY_VF_CQ_HEAD,TOY_VF_CQ_TAIL,TOY_VF_CONFIG_EPOCH_LO,TOY_VF_CONFIG_EPOCH_HI,
15        TOY_VF_CONFIG_SESSION_LO,TOY_VF_CONFIG_SESSION_HI,TOY_VF_QUEUE_STATUS,TOY_VF_QUEUE_ERROR};
16    uint32_t flags,value;
17    if (owner_latch(o,&flags)) return -1;
18    if (!(flags&TOY_HW_FLAGS_FENCED)||(flags&(TOY_HW_FLAGS_CONFIG_VALID|TOY_HW_FLAGS_READY_PENDING|TOY_HW_FLAGS_KILLED))) { errno=EPROTO; return -1; }
19    for (unsigned i=0;i<sizeof(registers)/sizeof(registers[0]);i++) {
20        if (owner_read32(o,registers[i],&value)) return -1;
21        if (value) { errno=EPROTO; return -1; }
22    }
23    printf("{\"event\":\"%s\",\"pid\":%d,\"bdf\":\"%s\",\"cause\":\"%s\",\"session\":%" PRIu64 ",\"epoch\":%" PRIu64 ",\"flags\":%u,\"zero_registers\":16}\n",o->recovery_case?"RECOVERY_CLEAR":"T06_CLEAR",getpid(),o->vfio.bdf,cause,o->session,o->epoch,flags);
24    return 0;
25}

27int owner_vfio_reset(struct owner *o)
28{
29    uint64_t old_epoch=o->epoch,old_session=o->session,count;
30    uint32_t status,sq,cq,digests[OWNER_MAPS];
31    if (!(o->vfio.info.flags&VFIO_DEVICE_FLAGS_RESET)||owner_freeze(o)||
32        owner_read32(o,TOY_VF_QUEUE_STATUS,&status)||status!=TOY_QUEUE_STATUS_RUNNING||
33        owner_read32(o,TOY_VF_SQ_HEAD,&sq)||owner_read32(o,TOY_VF_CQ_TAIL,&cq)||!sq||sq!=cq) { errno=EPROTO; return -1; }
34    /* 真正走 VFIO ioctl 與 PCI FLR，不以 PF mailbox 代替。 */
35    if (ioctl(o->vfio.device,VFIO_DEVICE_RESET)||owner_reset_clear(o,"VFIO_DEVICE_RESET")||
36        o->session!=old_session||o->epoch<=old_epoch||owner_write32(o,TOY_VF_DOORBELL,1)||
37        owner_unchanged(o,digests)) { if (!errno) errno=EPROTO; return -1; }
38    errno=0;
39    if (read(o->eventfds[0],&count,sizeof(count))!=-1||errno!=EAGAIN) { errno=EPROTO; return -1; }
40    printf("{\"event\":\"T06_IOCTL_RESET\",\"pid\":%d,\"old_epoch\":%" PRIu64 ",\"new_epoch\":%" PRIu64 ",\"old_sq_head\":%u,\"old_cq_tail\":%u,\"completed\":%" PRIu64 ",\"ioctl_errno\":0,\"doorbell\":true,\"cq_irq\":0,\"old_buffers_unchanged\":true}\n",getpid(),old_epoch,o->epoch,sq,cq,o->completed);
41    return 0;
