114/* 所有呼叫點皆由 probe 序列化或持有 admin_mutex；雙槽獨立計次。 */
115static bool toy_fail(struct toy_pf *pf, const char *name)
116{
117    unsigned int i;
118    bool hit = false;
119    if (!strcmp(name, forward_failpoint) && atomic_inc_return(&forward_seen) == forward_hit)
120        hit = true;
121    if (!strcmp(name, unwind_failpoint) && atomic_inc_return(&unwind_seen) == unwind_hit)
122        hit = true;
123    if (pf)
124        for (i = 0; i < ARRAY_SIZE(pf->failpoints); ++i) {
125            struct toy_failpoint *fp = &pf->failpoints[i];
126            if (!fp->fired && !strcmp(fp->name, name) && ++fp->seen == fp->hit) {
127                fp->fired = true;
128                hit = true;
129            }
130        }
131    if (hit)
132        pr_info("TOY_FAILPOINT_HIT name=%s\n", name);
133    return hit;
134}

136static int toy_status_errno(u32 status)
137{
138    switch (status) {
139    case TOY_STATUSES_OK: return 0;
140    case TOY_STATUSES_INVAL: return -EINVAL;
141    case TOY_STATUSES_NOSPC: return -ENOSPC;
142    case TOY_STATUSES_STATE: return -EBUSY;
143    case TOY_STATUSES_FENCED: return -EACCES;
144    case TOY_STATUSES_DMA: return -EIO;
145    case TOY_STATUSES_TIMEOUT: return -ETIMEDOUT;
146    case TOY_STATUSES_UNSUP: return -EOPNOTSUPP;
147    case TOY_STATUSES_NO_NEG: return -EPROTO;
148    case TOY_STATUSES_SEQ_CONFLICT: return -EPROTO;
149    case TOY_STATUSES_STALE_SEQ: return -ESTALE;
150    case TOY_STATUSES_BUSY: return -EBUSY;
151    case TOY_STATUSES_OVERFLOW: return -EOVERFLOW;
152    case TOY_STATUSES_BAD_DESC: return -EINVAL;
153    case TOY_STATUSES_BAD_QUEUE: return -EINVAL;
154    case TOY_STATUSES_IO: return -EIO;
155    default: return -EPROTO;
156    }

169static u64 toy_read64(struct toy_pf *pf, u32 low)
170{
171    /* 呼叫者必須已在 admin_mutex 下完成 SNAPSHOT。 */
172    u32 lo = toy_read(pf, low);
173    u32 hi = toy_read(pf, low + 4);
174    return ((u64)hi << 32) | lo;

177static void toy_fence_all(struct toy_pf *pf)
178{
179    unsigned int vf;
180    toy_write(pf, TOY_PF_CONTROL, TOY_PF_CONTROL_FENCE_ALL);
181    (void)toy_read(pf, TOY_PF_HW_RESULT);
182    toy_write(pf, TOY_PF_SNAPSHOT, 1);
183    for (vf = 0; vf < TOY_PCI_TOTAL_VFS; ++vf) {
184        u32 flags = toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_HW_STATUS));
185        if (!(flags & TOY_HW_FLAGS_FENCED))
186            dev_err(&pf->pdev->dev, "FENCE_ALL_READBACK vf=%u flags=%#x\n", vf, flags);
187    }

192static int toy_fence_status(struct toy_pf *pf, unsigned int vf, u32 *fw_status)
193{
194    u32 result, flags;
195    if (pf->fence_token == U32_MAX) {
196        toy_kill(pf, vf);
197        *fw_status = TOY_STATUSES_OVERFLOW;
198        return -EOVERFLOW;
199    }
200    toy_write(pf, TOY_PF_FENCE_TOKEN, ++pf->fence_token);
201    toy_write(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_HW_CTL), TOY_VF_HW_CONTROL_FENCE_BEGIN);
202    result = toy_read(pf, TOY_PF_HW_RESULT);
203    *fw_status = result;
204    toy_write(pf, TOY_PF_SNAPSHOT, 1);
205    flags = toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_HW_STATUS));
206    if (result)
207        return toy_status_errno(result);
208    return flags & TOY_HW_FLAGS_FENCED ? 0 : -EIO;
209}

218{

226/* 呼叫者持有 admin_mutex，再取得 mbx_mutex。IRQ 只完成通知，不拿這兩把 mutex。 */
227static int toy_mailbox_status(struct toy_pf *pf, u32 opcode, u32 target,
228                             const u32 args[4], u32 reply[4], u32 *fw_status)
229{
230    unsigned long deadline;
231    u32 seq, status;
232    unsigned int i, retries = 0;
233    int ret = -ETIMEDOUT;
234    /* IRQ completion 不取得 admin/mbx 鎖，poll 是掉 IRQ 時的必要後備。 */
235    mutex_lock(&pf->mbx_mutex);
236    *fw_status = TOY_STATUSES_TIMEOUT;
237    if (pf->seq == U32_MAX) {
238        ret = -EOVERFLOW;
239        *fw_status = TOY_STATUSES_OVERFLOW;
240        toy_fence_all(pf);
241        goto out;
242    }
243    status = toy_read(pf, TOY_PF_MBX_STATUS);
244    if (status == TOY_MBX_STATUS_BUSY) {
245        ret = -EBUSY;
246        *fw_status = TOY_STATUSES_BUSY;
247        goto out;
248    }
249    if (status == TOY_MBX_STATUS_DONE)
250        toy_write(pf, TOY_PF_MBX_ACK, 1);
251    seq = ++pf->seq;
252    reinit_completion(&pf->mbx_done);
253    toy_write(pf, TOY_PF_MBX_SEQ, seq);
254    toy_write(pf, TOY_PF_MBX_OPCODE, opcode);
255    toy_write(pf, TOY_PF_MBX_TARGET, target);
256    for (i = 0; i < 4; ++i)
257        toy_write(pf, TOY_PF_MBX_ARG0 + 4 * i, args ? args[i] : 0);
258    toy_write(pf, TOY_PF_MBX_SUBMIT, 1);
259    deadline = jiffies + msecs_to_jiffies(TOY_MBX_TIMEOUT_MS);
260    do {
261        status = toy_read(pf, TOY_PF_MBX_STATUS);
262        if (status == TOY_MBX_STATUS_DONE && toy_read(pf, TOY_PF_MBX_ACK_SEQ) == seq) {
263            u32 fw_result = toy_read(pf, TOY_PF_MBX_RESULT);
264            *fw_status = fw_result;
265            if (reply)
266                for (i = 0; i < 4; ++i)
267                    reply[i] = toy_read(pf, TOY_PF_MBX_REPLY0 + 4 * i);
268            toy_write(pf, TOY_PF_MBX_ACK, 1);
269            ret = toy_status_errno(fw_result);
270            break;
271        }
272        /* ACK 遺失且裝置已回 IDLE 時，重送完全相同的 request，不重做 fence。 */
273        if (status == TOY_MBX_STATUS_IDLE && retries < 2) {
274            ++retries;
275            toy_write(pf, TOY_PF_MBX_SUBMIT, 1);
276            dev_info(&pf->pdev->dev, "MAILBOX_RETRY seq=%u attempt=%u\n", seq, retries);
277        }
278        wait_for_completion_timeout(&pf->mbx_done, msecs_to_jiffies(1));
279    } while (time_before(jiffies, deadline));
280    if (ret)
281        dev_err(&pf->pdev->dev, "MAILBOX_ERROR seq=%u opcode=%u target=%u errno=%d\n",
282                seq, opcode, target, ret);
283out:
284    mutex_unlock(&pf->mbx_mutex);
285    return ret;
286}

295/* 中斷脈絡只 ACK、喚醒 mailbox 等待者及排 work；需要管理鎖的通知整理留給 workqueue。 */
296static irqreturn_t toy_irq(int irq, void *data)
297{
298    struct toy_pf *pf = data;
299    u32 events = toy_read(pf, TOY_PF_EVENT);
300    if (!events)
301        return IRQ_NONE;
302    toy_write(pf, TOY_PF_EVENT, events);
303    if (events & TOY_PF_EVENTS_MBX_DONE)
304        complete(&pf->mbx_done);
305    if (events & (TOY_PF_EVENTS_FLR_EVENT | TOY_PF_EVENTS_VF_HEALTH)) {
306        atomic_or(events, &pf->pending_events);
307        if (!READ_ONCE(pf->dying))
308            schedule_work(&pf->notify_work);
309    }
310    return IRQ_HANDLED;
311}

313static int toy_verify_free(struct toy_pf *pf, unsigned int vf)
314{
315    u32 offset = TOY_VF_VIEW(vf, 0);
316    u32 state, reserve, outstanding, flags;
317    toy_write(pf, TOY_PF_SNAPSHOT, 1);
318    state = toy_read(pf, offset + TOY_PF_VF_VIEW_STATE);
319    reserve = toy_read(pf, offset + TOY_PF_VF_VIEW_RESERVE);
320    outstanding = toy_read(pf, offset + TOY_PF_VF_VIEW_OUTSTANDING);
321    flags = toy_read(pf, offset + TOY_PF_VF_VIEW_HW_STATUS);
322    if (state != TOY_STATES_FREE || reserve || outstanding ||
323        (flags & (TOY_HW_FLAGS_CONFIG_VALID | TOY_HW_FLAGS_PCI_PRESENT)) ||
324        !(flags & TOY_HW_FLAGS_FENCED)) {
325        dev_err(&pf->pdev->dev,
326                "FREE_VERIFY_FAILED vf=%u state=%u reserve=%u outstanding=%u flags=%#x\n",
327                vf, state, reserve, outstanding, flags);
328        return -EIO;
329    }
330    dev_info(&pf->pdev->dev, "FREE_VERIFIED vf=%u reserve=0 outstanding=0 flags=%#x\n", vf, flags);
331    return 0;
332}

334static int toy_destroy_slot(struct toy_pf *pf, unsigned int vf)
335{
336    int ret = toy_fence(pf, vf);
337    if (!ret && toy_fail(pf, "unwind.destroy"))
338        ret = -EIO;
339    if (!ret)
340        ret = toy_mailbox(pf, TOY_OPCODES_DESTROY, vf, NULL, NULL);
341    if (!ret)
342        ret = toy_verify_free(pf, vf);
343    if (ret) {
344        unsigned long deadline = jiffies + msecs_to_jiffies(TOY_MBX_TIMEOUT_MS);
345        toy_kill(pf, vf);
346        /* firmware 存活時會觀察 KILL 並以合法事件收斂；逾時仍保留配額。 */
347        do {
348            toy_write(pf, TOY_PF_SNAPSHOT, 1);
349            if (toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_STATE)) == TOY_STATES_FAILED)
350                break;
351            usleep_range(1000, 2000);
352        } while (time_before(jiffies, deadline));
353        dev_err(&pf->pdev->dev, "DESTROY_RETAINED vf=%u errno=%d\n", vf, ret);
354    }
355    return ret;
356}

358static int toy_sriov_configure(struct pci_dev *pdev, int count)
359{
360    struct toy_pf *pf = pci_get_drvdata(pdev);
361    const u32 args[4] = { 8, 0, 0, 0 };
362    unsigned int i, provisioned = 0;
363    int ret = 0, cleanup_ret;
364    if (!pf || count < 0 || count > TOY_PCI_TOTAL_VFS)
365        return -EINVAL;
366    mutex_lock(&pf->admin_mutex);
367    if (!pf->accepting || pf->dying) {
368        ret = -ENODEV;
369        goto out;
370    }
371    if (pf->transaction) {
372        ret = -EBUSY;
373        goto out;
374    }
375    if (count && pf->enabled) {
376        ret = count == pf->enabled ? count : -EBUSY;
377        goto out;
378    }
379    if (!count) {
380        for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i)
381            if (pf->leases[i].token) {
382                ret = -EBUSY;
383                goto out;
384            }
385        if (pci_vfs_assigned(pdev)) {
386            ret = -EBUSY;
387            goto out;
388        }
389        for (i = 0; i < pf->enabled; ++i) {
390            ret = toy_fence(pf, i);
391            if (ret)
392                goto out;
393        }
394        pf->transaction = true;
395        reinit_completion(&pf->transaction_done);
396        /* PCI/VFIO 可能等待 owner；不能持有 worker 或控制 fd 需要的鎖。 */
397        mutex_unlock(&pf->admin_mutex);
398        pci_disable_sriov(pdev);
399        mutex_lock(&pf->admin_mutex);
400        if (pf->dying) {
401            ret = -ENODEV;
402            goto transaction_done;
403        }
404        for (i = pf->enabled; i > 0; --i) {
405            cleanup_ret = toy_destroy_slot(pf, i - 1);
406            if (cleanup_ret && !ret)
407                ret = cleanup_ret;
408        }
409        pf->enabled = 0;
410        goto transaction_done;
411    }
412    pf->transaction = true;
413    reinit_completion(&pf->transaction_done);
414    for (i = 0; i < count; ++i) {
415        ret = toy_mailbox(pf, TOY_OPCODES_PROVISION, i, args, NULL);
416        if (ret)
417            goto unwind;
418        ++provisioned;
419        if (toy_fail(pf, "provision.host_record")) {
420            ret = -ENOMEM;
421            goto unwind;
422        }
423    }
424    if (toy_fail(pf, "provision.pci_enable")) {
425        ret = -EIO;
426        goto unwind;
427    }
428    mutex_unlock(&pf->admin_mutex);
429    ret = pci_enable_sriov(pdev, count);
430    mutex_lock(&pf->admin_mutex);
431    if (ret)
432        goto unwind;
433    if (pf->dying || toy_fail(pf, "provision.after_pci_enable")) {
434        ret = pf->dying ? -ENODEV : -EIO;
435        mutex_unlock(&pf->admin_mutex);
436        pci_disable_sriov(pdev);
437        mutex_lock(&pf->admin_mutex);
438        goto unwind;
439    }
440    pf->enabled = count;
441    ret = count;
442    goto transaction_done;
443unwind:
444    /* PCI core 的失敗返回已完成枚舉 unwind；ledger 依反向次序撤銷。 */
445    while (provisioned) {
446        cleanup_ret = toy_destroy_slot(pf, --provisioned);
447        if (cleanup_ret)
448            dev_err(&pdev->dev, "PROVISION_UNWIND_RETAINED vf=%u errno=%d\n",
449                    provisioned, cleanup_ret);
450    }
451transaction_done:
452    pf->transaction = false;
453    complete_all(&pf->transaction_done);
454out:
455    mutex_unlock(&pf->admin_mutex);
456    return ret;
457}

494static ssize_t toy_debug_read(struct file *file, char __user *buffer,
495                              size_t length, loff_t *offset)
496{
497    struct toy_pf *pf = file->private_data;
498    struct toy_failpoint *fp = &pf->failpoints[toy_debug_slot(file)];
499    char record[160];
500    int size;
501    mutex_lock(&pf->admin_mutex);
502    if (pf->dying) {
503        mutex_unlock(&pf->admin_mutex);
504        return -ENODEV;
505    }
506    size = scnprintf(record, sizeof(record), "name=%s hit=%u seen=%u fired=%u\n",
507                      fp->name[0] ? fp->name : "off", fp->hit, fp->seen, fp->fired);
508    mutex_unlock(&pf->admin_mutex);
509    return simple_read_from_buffer(buffer, length, offset, record, size);

512static ssize_t toy_debug_write(struct file *file, const char __user *buffer,
513                               size_t length, loff_t *offset)
514{
515    struct toy_pf *pf = file->private_data;
516    struct toy_failpoint *fp = &pf->failpoints[toy_debug_slot(file)];
517    char record[96], name[64], trailing;
518    unsigned int hit;
519    int ret = length;
520    if (!length || length >= sizeof(record))
521        return -EINVAL;
522    if (copy_from_user(record, buffer, length))
523        return -EFAULT;
524    if (memchr(record, 0, length))
525        return -EINVAL;
526    record[length] = 0;
527    if (sscanf(record, "%63s %u %c", name, &hit, &trailing) != 2 || !hit ||
528        (strcmp(name, "off") && !toy_fail_name_valid(name)))
529        return -EINVAL;
530    mutex_lock(&pf->admin_mutex);
531    if (pf->dying)
532        ret = -ENODEV;
533    else if (pf->transaction)
534        ret = -EBUSY;
535    else {
536        memset(fp, 0, sizeof(*fp));
537        if (strcmp(name, "off"))
538            strscpy(fp->name, name, sizeof(fp->name));
539        fp->hit = hit;
540        dev_info(&pf->pdev->dev, "FAILPOINT_ARM slot=%u name=%s hit=%u\n",
541                 toy_debug_slot(file), name, hit);
542    }
543    mutex_unlock(&pf->admin_mutex);
544    return ret;
545}

552/* 只讀生命週期觀測；不清空 queue 或等待 worker，避免讓量測改變結果。 */
553static ssize_t toy_lifecycle_read(struct file *file, char __user *buffer,
554                                  size_t length, loff_t *offset)
555{
556    struct toy_pf *pf = file->private_data;
557    struct toy_file *ctx;
558    unsigned int files = 0, leases = 0, orphaned = 0, events = 0, i;
559    char record[256];
560    int size;
561    mutex_lock(&pf->admin_mutex);
562    if (pf->dying) {
563        mutex_unlock(&pf->admin_mutex);
564        return -ENODEV;
565    }
566    list_for_each_entry(ctx, &pf->files, link) {
567        unsigned long flags;
568        ++files;
569        spin_lock_irqsave(&ctx->event_lock, flags);
570        events += ctx->count + !!ctx->lost + ctx->stash_valid;
571        spin_unlock_irqrestore(&ctx->event_lock, flags);
572    }
573    for (i = 0; i < TOY_PCI_TOTAL_VFS; ++i) {
574        leases += !!pf->leases[i].token;
575        orphaned += pf->leases[i].orphaned;
576    }
577    size = scnprintf(record, sizeof(record),
578        "contexts=%u leases=%u orphaned=%u events=%u work_busy=%u pending_irqs=%u enabled=%u transaction=%u\n",
579        files, leases, orphaned, events, work_busy(&pf->notify_work),
580        (unsigned int)atomic_read(&pf->pending_events), pf->enabled, pf->transaction);
581    mutex_unlock(&pf->admin_mutex);
582    return simple_read_from_buffer(buffer, length, offset, record, size);
583}

590static ssize_t toy_stats_read(struct file *file, char __user *buffer,
591                              size_t length, loff_t *offset)
592{
593    struct toy_pf *pf = file->private_data;
594    char record[1536];
595    int used = 0;
596    unsigned int vf;
597    mutex_lock(&pf->admin_mutex);
598    if (pf->dying) {
599        mutex_unlock(&pf->admin_mutex);
600        return -ENODEV;
601    }
602    /* 同一個硬體 snapshot 提供所有 VF 的唯讀健康計數。 */
603    toy_write(pf, TOY_PF_SNAPSHOT, 1);
604    for (vf = 0; vf < TOY_PCI_TOTAL_VFS; ++vf)
605        used += scnprintf(record + used, sizeof(record) - used,
606            "vf=%u stale_desc=%u stale_cpl=%u dma_fault=%u fence_blocked=%u irq_suppressed=%u\n",
607            vf, toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_STALE_DESC)),
608            toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_STALE_CPL)),
609            toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_DMA_FAULT)),
610            toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_FENCE_BLOCKED)),
611            toy_read(pf, TOY_VF_VIEW(vf, TOY_PF_VF_VIEW_IRQ_SUPPRESSED)));
612    mutex_unlock(&pf->admin_mutex);
613    return simple_read_from_buffer(buffer, length, offset, record, used);
614}

621/* 依序取得 PCI、BAR、IRQ 與控制節點；每個 goto 都只釋放此前已成功取得的資源。 */
622static int toy_probe(struct pci_dev *pdev, const struct pci_device_id *id)
623{
624    struct toy_pf *pf;
625    const u32 version_args[4] = { TOY_PROTOCOL_MAJOR, TOY_PROTOCOL_MINOR, TOY_PROTOCOL_MINOR, 0 };
626    u32 version_reply[4];
627    int ret, i;
628    if (!forward_hit || !unwind_hit ||
629        (forward_failpoint[0] && !toy_fail_name_valid(forward_failpoint)) ||
630        (unwind_failpoint[0] && !toy_fail_name_valid(unwind_failpoint)))
631        return -EINVAL;
632    if (toy_fail(NULL, "probe.alloc_ctx"))
633        return -ENOMEM;
634    pf = kzalloc(sizeof(*pf), GFP_KERNEL);
635    if (!pf)
636        return -ENOMEM;
637    pf->pdev = pdev;
638    mutex_init(&pf->admin_mutex);
639    mutex_init(&pf->mbx_mutex);
640    init_completion(&pf->mbx_done);
641    init_completion(&pf->transaction_done);
642    complete_all(&pf->transaction_done);
643    INIT_LIST_HEAD(&pf->files);
644    INIT_WORK(&pf->notify_work, toy_notify_work);
645    atomic_set(&pf->pending_events, 0);
646    pf->failpoints[0].hit = pf->failpoints[1].hit = 1;
647    kref_init(&pf->ref);
648    ret = toy_fail(pf, "probe.enable_mem") ? -EIO : pci_enable_device_mem(pdev);
649    if (ret)
650        goto free_context;
651    ret = toy_fail(pf, "probe.request_regions") ? -EBUSY : pci_request_regions(pdev, "toy_pf");
652    if (ret)
653        goto disable_device;
654    if (pci_resource_len(pdev, 0) != TOY_LIMITS_PF_BAR0_BYTES ||
655        !(pci_resource_flags(pdev, 0) & IORESOURCE_MEM)) {
656        ret = -ENODEV;
657        goto release_regions;
658    }
659    if (!toy_fail(pf, "probe.map_bar"))
660        pf->bar = pci_iomap(pdev, 0, 0);
661    if (!pf->bar) {
662        ret = -ENOMEM;
663        goto release_regions;
664    }
665    if (toy_fail(pf, "probe.validate_caps") ||
666        toy_read(pf, TOY_PF_MAGIC) != TOY_PF_MAGIC_VALUE ||
667        toy_read(pf, TOY_PF_ABI) != TOY_ABI_VERSION ||
668        toy_read(pf, TOY_PF_CAP_VFS) != TOY_PCI_TOTAL_VFS ||
669        toy_read(pf, TOY_PF_CAP_QUEUES) != TOY_LIMITS_QUEUES ||
670        toy_read(pf, TOY_PF_CAP_VECTORS) != TOY_LIMITS_VF_VECTORS ||
671        toy_read(pf, TOY_PF_CAP_CREDIT) != TOY_LIMITS_CREDITS) {
672        ret = -EPROTO;
673        goto unmap;
674    }
675    /* 同一 machine 的 driver rebind 不可從零重用去重序號。 */
676    pf->seq = toy_read(pf, TOY_PF_MBX_ACK_SEQ);
677    pf->fence_token = toy_read(pf, TOY_PF_FENCE_TOKEN);
678    ret = toy_fail(pf, "probe.alloc_vectors") ? -ENOMEM :
679          pci_alloc_irq_vectors(pdev, 2, 2, PCI_IRQ_MSIX);
680    if (ret < 0)
681        goto unmap;
682    for (i = 0; i < 2; ++i) {
683        ret = toy_fail(pf, i ? "probe.request_irq1" : "probe.request_irq0") ? -EIO :
684              request_irq(pci_irq_vector(pdev, i), toy_irq, 0, "toy_pf", pf);
685        if (ret)
686            goto free_irqs;
687        ++pf->irq_count;
688    }
689    pci_set_master(pdev);
690    pci_set_drvdata(pdev, pf);
691    mutex_lock(&pf->admin_mutex);
692    ret = toy_fail(pf, "probe.version_neg") ? -EIO :
693          toy_mailbox(pf, TOY_OPCODES_VERSION_NEG, TOY_GLOBAL, version_args, version_reply);
694    if (!ret && version_reply[0] != TOY_ABI_VERSION)
695        ret = -EPROTO;
696    mutex_unlock(&pf->admin_mutex);
697    if (ret)
698        goto clear_master;
699    pf->misc.minor = MISC_DYNAMIC_MINOR;
700    pf->misc.name = "toy_pf0";
701    pf->misc.fops = &toy_fops;
702    pf->misc.mode = 0600;
703    pf->misc.parent = &pdev->dev;
704    pf->accepting = true;
705    ret = toy_fail(pf, "probe.misc_register") ? -EIO : misc_register(&pf->misc);
706    if (ret)
707        goto clear_master;
708    pf->debugfs = debugfs_create_dir("toy_pf0", NULL);
709    debugfs_create_file("forward", 0600, pf->debugfs, pf, &toy_debug_fops);
710    debugfs_create_file("unwind", 0600, pf->debugfs, pf, &toy_debug_fops);
711    debugfs_create_file("lifecycle", 0400, pf->debugfs, pf, &toy_lifecycle_fops);
712    debugfs_create_file("vf-stats", 0400, pf->debugfs, pf, &toy_stats_fops);
713    dev_info(&pdev->dev, "TOY_PF_READY abi=1.0 vectors=2 total_vfs=8\n");
714    return 0;
715clear_master:
716    pf->accepting = false;
717    toy_fence_all(pf);
718    pci_clear_master(pdev);
719    pci_set_drvdata(pdev, NULL);
720free_irqs:
721    WRITE_ONCE(pf->dying, true);
722    while (pf->irq_count) {
723        --pf->irq_count;
724        free_irq(pci_irq_vector(pdev, pf->irq_count), pf);
725    }
726    pci_free_irq_vectors(pdev);
727    cancel_work_sync(&pf->notify_work);
728unmap:
729    pci_iounmap(pdev, pf->bar);
730release_regions:
731    pci_release_regions(pdev);
732disable_device:
733    pci_disable_device(pdev);
734free_context:
735    kref_put(&pf->ref, toy_release_ref);
736    return ret;
737}

739/* 先拒絕新操作並 fence，再同步 worker/IRQ 與解除 BAR；舊 fd 持有的 context 由 kref 延後釋放。 */
740static void toy_remove(struct pci_dev *pdev)
741{
742    struct toy_pf *pf = pci_get_drvdata(pdev);
743    unsigned int i;
744    if (!pf)
745        return;
746    mutex_lock(&pf->admin_mutex);
747    pf->accepting = false;
748    WRITE_ONCE(pf->dying, true);
749    toy_fence_all(pf);
750    toy_publish(pf, TOY_EVENT_REMOVE, TOY_CTL_GLOBAL, 0, NULL);
751    dev_info(&pdev->dev, "TOY_REMOVE_FENCED enabled=%u\n", pf->enabled);
752    mutex_unlock(&pf->admin_mutex);
753    misc_deregister(&pf->misc);
754    debugfs_remove_recursive(pf->debugfs);
755    cancel_work_sync(&pf->notify_work);
756    /* PCI core 的 device lock 序列化 sriov_configure/remove；owner 自行關 fd。 */
757    pci_disable_sriov(pdev);
758    mutex_lock(&pf->admin_mutex);
759    for (i = pf->enabled; i > 0; --i)
760        toy_destroy_slot(pf, i - 1);
761    pf->enabled = 0;
762    mutex_unlock(&pf->admin_mutex);
763    for (i = 0; i < pf->irq_count; ++i)
764        disable_irq(pci_irq_vector(pdev, i));
765    while (pf->irq_count) {
766        --pf->irq_count;
767        synchronize_irq(pci_irq_vector(pdev, pf->irq_count));
768        free_irq(pci_irq_vector(pdev, pf->irq_count), pf);
769    }
770    pci_free_irq_vectors(pdev);
771    pci_clear_master(pdev);
772    pci_iounmap(pdev, pf->bar);
773    pf->bar = NULL;
774    pci_release_regions(pdev);
775    pci_disable_device(pdev);
776    pci_set_drvdata(pdev, NULL);
777    dev_info(&pdev->dev, "TOY_PF_REMOVED\n");
778    kref_put(&pf->ref, toy_release_ref);
779}
