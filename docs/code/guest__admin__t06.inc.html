2static int reset_snapshot(struct admin *a,const char *phase,unsigned state,unsigned config)
3{
4    if (snapshot(a)||a->info.state!=state||a->info.reserve!=8||a->info.outstanding||
5        a->info.lease_token!=a->lease||(a->info.hw_flags&TOY_HW_FLAGS_KILLED)||
6        !!(a->info.hw_flags&TOY_HW_FLAGS_CONFIG_VALID)!=config) { errno=EPROTO; return -1; }
7    printf("{\"event\":\"%s_SNAPSHOT\",\"phase\":\"%s\",\"pid\":%d,\"vf\":%u,\"state\":%u,\"epoch\":%" PRIu64 ",\"session\":%" PRIu64 ",\"ready_gen\":%u,\"flags\":%u,\"reserve\":%u,\"outstanding\":%u,\"lease\":%" PRIu64 "}\n",a->scenario,phase,a->child,a->vf,a->info.state,(uint64_t)a->info.epoch,(uint64_t)a->info.session,a->info.ready_gen,a->info.hw_flags,a->info.reserve,a->info.outstanding,(uint64_t)a->info.lease_token);
8    return 0;
9}
10static int healthy_checkpoint(struct admin *b,const char *phase)
11{
12    if (owner_inspect(b)||snapshot(b)||b->info.state!=TOY_STATES_ACTIVE||b->info.hw_flags!=28||
13        b->info.epoch!=b->original.epoch||b->info.session!=b->original.session||
14        b->info.reserve!=8||b->info.lease_token!=b->lease) { errno=EPROTO; return -1; }
15    printf("{\"event\":\"%s_HEALTHY_CHECK\",\"phase\":\"%s\",\"pid\":%d,\"epoch\":%" PRIu64 ",\"session\":%" PRIu64 ",\"state\":%u,\"flags\":%u,\"reserve\":%u,\"lease\":%" PRIu64 "}\n",b->scenario,phase,b->child,(uint64_t)b->info.epoch,(uint64_t)b->info.session,b->info.state,b->info.hw_flags,b->info.reserve,(uint64_t)b->info.lease_token);
16    return 0;
17}
18static int reset_handshake(struct admin *a,struct admin *b)
19{
20    struct lab_message m=a->current;
21    struct toy_vf_ctl ctl;
22    uint64_t old_epoch=m.epoch;
23    uint32_t old_gen=m.ready_gen;
24    if (reset_snapshot(a,"before_ioctl",TOY_STATES_ACTIVE,1)||healthy_checkpoint(b,"before_ioctl")) return -1;
25    m.type=LAB_RESET_NOTICE; m.payload_len=16; memset(m.payload,0,16); lab_put32(m.payload,1);
26    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_HELLO,4)||
27        reset_snapshot(a,"after_ioctl",TOY_STATES_RECOVERING,0)) return -1;
28    if (m.epoch<=old_epoch||m.epoch!=a->info.epoch||m.session!=a->original.session||m.ready_gen!=old_gen||
29        m.operation!=a->original.operation||a->info.ready_gen!=old_gen||!(a->info.hw_flags&TOY_HW_FLAGS_FENCED)) { errno=ESTALE; return -1; }
30    a->original=a->current=m;
31    /* 舊 generation 的 REARM 必須遭拒；不可先放行再補 CONFIG。 */
32    errno=0;
33    int rc=control(a,TOY_REARM_VF,old_gen,&ctl),error=errno;
34    if (!rc||error!=EBUSY||reset_snapshot(a,"rearm_rejected",TOY_STATES_RECOVERING,0)||
35        owner_inspect(a)||healthy_checkpoint(b,"unconfigured")) { errno=EPROTO; return -1; }
36    printf("{\"event\":\"T06_REARM_REJECT\",\"errno\":%d,\"epoch\":%" PRIu64 ",\"ready_gen\":%u}\n",error,m.epoch,old_gen);
37    if (old_gen==UINT32_MAX) { errno=EOVERFLOW; return -1; }
38    m.type=LAB_HELLO_REPLY; m.ready_gen=old_gen+1; a->original=m;
39    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_READY,5)||
40        reset_snapshot(a,"fresh_config",TOY_STATES_RECOVERING,1)) return -1;
41    if (lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
42        m.ready_gen!=a->original.ready_gen||a->info.ready_gen!=m.ready_gen||a->info.hw_flags!=61||
43        a->info.epoch!=m.epoch) { errno=ESTALE; return -1; }
44    a->current=m;
45    if (healthy_checkpoint(b,"configured_fenced")||owner_inspect(a)||control(a,TOY_REARM_VF,m.ready_gen,&ctl)||
46        reset_snapshot(a,"rearmed",TOY_STATES_ACTIVE,1)||owner_run(a)||healthy_checkpoint(b,"fresh_io")) return -1;
47    return 0;
48}
49static int vfio_reset_case(const char *pf,const char *device,const char *tenant)
50{
51    struct admin owners[2];
52    char count_path[PATH_MAX];
53    bool enabled=false;
54    int ret=1,saved;
55    snprintf(count_path,sizeof(count_path),"/sys/bus/pci/devices/%s/sriov_numvfs",pf);
56    for (unsigned i=0;i<2;i++) owners[i]=(struct admin){.pf=-1,.listen=-1,.connection=-1,.pidfd=-1,.management=-1,
57        .vf=i,.failpoint="",.fail_hit=1,.scenario="T06",.count_path=count_path,.vfio_reset=!i,.healthy=i};
58    if (lab_write_text(count_path,"4\n")) goto out;
59    enabled=true;
60    for (unsigned i=0;i<2;i++) if (isolated_owner_start(&owners[i],pf,device,tenant,16,4096)) goto out;
61    struct admin *a=&owners[0],*b=&owners[1];
62    struct lab_message m=b->current;
63    m.type=LAB_RUN;
64    if (lab_send(b->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||healthy_checkpoint(b,"started")||
65        owner_run(a)||reset_handshake(a,b)||owner_stop(a)||reset_snapshot(a,"closed",TOY_STATES_RECOVERING,0)||
66        healthy_checkpoint(b,"closed")) goto out;
67    uint64_t closed_epoch=a->info.epoch,session=a->info.session;
68    pid_t old_pid=a->child;
69    if (release_owner(a)) goto out;
70    isolated_owner_cleanup(a);
71    *a=(struct admin){.pf=-1,.listen=-1,.connection=-1,.pidfd=-1,.management=-1,
72        .vf=0,.failpoint="",.fail_hit=1,.scenario="T06",.count_path=count_path};
73    if (isolated_owner_start(a,pf,device,tenant,16,4096)||a->child==old_pid||
74        reset_snapshot(a,"reopened_rearmed",TOY_STATES_ACTIVE,1)||a->info.epoch<=closed_epoch||a->info.session!=session||
75        healthy_checkpoint(b,"reopened")||owner_run(a)||healthy_checkpoint(b,"reopened_io")||owner_stop(a)||
76        reset_snapshot(a,"reclosed",TOY_STATES_RECOVERING,0)||release_owner(a)) goto out;
77    m=b->current; m.type=LAB_STOP_REQUEST; m.payload_len=16; memset(m.payload,0,16);
78    if (lab_send(b->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(b,&m,LAB_STOPPED,3)||
79        lab_tuple(&m,b->original.vf,b->original.session,b->original.epoch,b->original.lease_token,b->original.operation)||
80        m.ready_gen!=b->original.ready_gen) goto out;
81    b->current=m;
82    if (owner_stop(b)||release_owner(b)||lab_write_text(count_path,"0\n")) goto out;
83    enabled=false;
84    uint64_t count;
85    if (lab_read_u64(count_path,&count)||count) { errno=EBUSY; goto out; }
86    for (unsigned i=0;i<4;i++) {
87        char path[PATH_MAX]; struct stat st;
88        snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/virtfn%u",pf,i);
89        if (!lstat(path,&st)||errno!=ENOENT) { errno=EBUSY; goto out; }
90    }
91    for (unsigned i=0;i<8;i++) {
92        struct toy_vf_info info={.hdr=header(sizeof(info)),.vf=i};
93        if (ioctl(a->pf,TOY_GET_VF,&info)||info.state||info.reserve||info.outstanding||info.lease_token||info.hw_flags!=1) { errno=EIO; goto out; }
94        printf("{\"event\":\"T06_FINAL\",\"vf\":%u,\"state\":%u,\"reserve\":%u,\"outstanding\":%u,\"flags\":%u,\"lease\":%" PRIu64 "}\n",i,info.state,info.reserve,info.outstanding,info.hw_flags,(uint64_t)info.lease_token);
95    }
96    puts("{\"event\":\"M3_CASE\",\"id\":\"T06\",\"status\":\"PASS\",\"ioctl_resets\":1,\"vf_a_owners\":2,\"all_free\":true,\"owner_closed\":true}");
97    puts("LAB_RESULT=PASS"); ret=0;
98out:
99    saved=errno;
100    for (unsigned i=0;i<2;i++) isolated_owner_cleanup(&owners[i]);
101    if (enabled&&lab_write_text(count_path,"0\n")) fprintf(stderr,"LABD_DISABLE_FAILED errno=%d\n",errno);
102    if (ret) {
103        fprintf(stderr,"LABD_ERROR scenario=T06 errno=%d message=%s\n",saved,strerror(saved));
104        printf("{\"event\":\"M3_CASE\",\"id\":\"T06\",\"status\":\"FAIL\",\"errno\":%d}\n",saved);
105        puts("LAB_RESULT=FAIL");
106    }
107    return ret;
108}
