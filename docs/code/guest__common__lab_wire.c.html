16uint16_t lab_get16(const void *p) { const unsigned char *b=p; return (uint16_t)b[0]|((uint16_t)b[1]<<8); }
17uint32_t lab_get32(const void *p) { const unsigned char *b=p; return (uint32_t)lab_get16(b)|((uint32_t)lab_get16(b+2)<<16); }
18uint64_t lab_get64(const void *p) { const unsigned char *b=p; return (uint64_t)lab_get32(b)|((uint64_t)lab_get32(b+4)<<32); }
19void lab_put16(void *p,uint16_t v) { unsigned char *b=p; b[0]=(unsigned char)v; b[1]=(unsigned char)(v>>8); }
20void lab_put32(void *p,uint32_t v) { unsigned char *b=p; lab_put16(b,(uint16_t)v); lab_put16(b+2,(uint16_t)(v>>16)); }

22int64_t lab_now_ms(void) { struct timespec ts; if(clock_gettime(CLOCK_MONOTONIC,&ts)) return -1; return (int64_t)ts.tv_sec*1000+ts.tv_nsec/1000000; }
23int lab_wait(int fd,short events,int64_t deadline)
24{
25    struct pollfd p={.fd=fd,.events=events};
26    for(;;) {
27        int64_t remain=deadline-lab_now_ms(); int r;
28        if(remain<=0) { errno=ETIMEDOUT; return -1; }
29        r=poll(&p,1,remain>INT_MAX?INT_MAX:(int)remain);
30        if(r<0 && errno==EINTR) continue;
31        if(r<0) return -1;
32        if(!r) { errno=ETIMEDOUT; return -1; }
33        if(p.revents&events) return 0;
34        if(p.revents&(POLLERR|POLLNVAL|POLLHUP)) { errno=ECONNRESET; return -1; }
35    }
36}
37static int shape(uint16_t type,uint32_t n,const unsigned char *p)
38{
39    switch(type) {
40    case LAB_HELLO: case LAB_HELLO_REPLY: case LAB_READY: case LAB_STOPPED: case LAB_RUN: case LAB_INSPECT: return n==0?0:-1;
41    case LAB_INSPECT_REPLY: return n==16 && lab_get32(p+12)==0?0:-1;
42    case LAB_STOP_REQUEST: case LAB_RESET_NOTICE: case LAB_STOP: return n==16 && lab_get32(p+12)==0?0:-1;
43    case LAB_EXIT_ACK: return n==16?0:-1;
44    case LAB_ERROR: return n==8 && lab_get32(p)>0 && lab_get32(p)<=4095?0:-1;
45    default: return -1;
46    }
47}
48int lab_encode(unsigned char *b,size_t cap,const struct lab_message *m)
49{
50    size_t n=LAB_HEADER_SIZE+(size_t)m->payload_len;
51    if(m->payload_len>sizeof(m->payload) || n>cap || !m->request_id || shape(m->type,m->payload_len,m->payload)) { errno=EPROTO; return -1; }
52    memset(b,0,n); memcpy(b,"TLAB",4); lab_put16(b+4,1); lab_put16(b+6,m->type);
53    lab_put64(b+8,m->request_id); lab_put64(b+16,m->session); lab_put64(b+24,m->epoch);
54    lab_put64(b+32,m->lease_token); lab_put64(b+40,m->operation); lab_put32(b+48,m->vf);
55    lab_put32(b+52,m->ready_gen); lab_put32(b+60,m->payload_len); memcpy(b+64,m->payload,m->payload_len);
56    return (int)n;
57}
58int lab_decode(struct lab_message *m,const unsigned char *b,size_t n)
59{
60    uint32_t bytes;
61    if(n<64 || n>LAB_FRAME_MAX || memcmp(b,"TLAB",4) || lab_get16(b+4)!=1 || lab_get32(b+56)) { errno=EPROTO; return -1; }
62    bytes=lab_get32(b+60);
63    if(bytes!=n-64 || shape(lab_get16(b+6),bytes,b+64) || !lab_get64(b+8)) { errno=EPROTO; return -1; }
64    memset(m,0,sizeof(*m)); m->type=lab_get16(b+6); m->request_id=lab_get64(b+8);
65    m->session=lab_get64(b+16); m->epoch=lab_get64(b+24); m->lease_token=lab_get64(b+32);
66    m->operation=lab_get64(b+40); m->vf=lab_get32(b+48); m->ready_gen=lab_get32(b+52);
67    m->payload_len=bytes; memcpy(m->payload,b+64,bytes); return 0;
68}
69/* Partial I/O 與 EINTR 只推進已完成的部分；重試沿用原 deadline，避免不斷延長等待。 */
70static int transfer(int fd,void *buf,size_t length,int64_t deadline,bool sending)
71{
72    unsigned char *p=buf;
73    while(length) {
74        ssize_t n;
75        if(lab_wait(fd,sending?POLLOUT:POLLIN,deadline)) return -1;
76        if(sending) n=send(fd,p,length,MSG_DONTWAIT|MSG_NOSIGNAL);
77        else {
78            unsigned char control[CMSG_SPACE(16*sizeof(int))];
79            struct iovec iov={.iov_base=p,.iov_len=length};
80            struct msghdr msg={.msg_iov=&iov,.msg_iovlen=1,.msg_control=control,.msg_controllen=sizeof(control)};
81            bool invalid=false;
82            n=recvmsg(fd,&msg,MSG_DONTWAIT|MSG_CMSG_CLOEXEC);
83            /* 收到任何 ancillary 都拒絕；已送入本程序的 fd 必須先關閉。 */
84            for(struct cmsghdr *c=CMSG_FIRSTHDR(&msg);n>=0 && c;c=CMSG_NXTHDR(&msg,c)) {
85                invalid=true;
86                if(c->cmsg_level==SOL_SOCKET && c->cmsg_type==SCM_RIGHTS && c->cmsg_len>=CMSG_LEN(0)) {
87                    size_t count=(c->cmsg_len-CMSG_LEN(0))/sizeof(int); int *fds=(void *)CMSG_DATA(c);
88                    for(size_t i=0;i<count;i++) close(fds[i]);
89                }
90            }
91            if(n>=0 && (invalid || (msg.msg_flags&(MSG_CTRUNC|MSG_TRUNC)))) { errno=EPROTO; return -1; }
92        }
93        if(n<0 && (errno==EINTR || errno==EAGAIN || errno==EWOULDBLOCK)) continue;
94        if(n<0) return -1;
95        if(!n) { errno=ECONNRESET; return -1; }
96        p+=(size_t)n; length-=(size_t)n;
97    }
98    return 0;
99}
100int lab_send(int fd,const struct lab_message *m,int64_t deadline)
101{
102    unsigned char frame[4+LAB_FRAME_MAX]; int n=lab_encode(frame+4,LAB_FRAME_MAX,m);
103    if(n<0) return -1;
104    lab_put32(frame,(uint32_t)n); return transfer(fd,frame,(size_t)n+4,deadline,true);
105}
106int lab_recv(int fd,struct lab_message *m,int64_t deadline)
107{
108    unsigned char frame[LAB_FRAME_MAX],prefix[4]; uint32_t n;
109    if(transfer(fd,prefix,4,deadline,false)) return -1;
110    n=lab_get32(prefix);
111    if(n<64 || n>LAB_FRAME_MAX) { errno=EMSGSIZE; return -1; }
112    if(transfer(fd,frame,n,deadline,false)) return -1;
113    return lab_decode(m,frame,n);
114}
115/* 核對 socket 的 peer credentials；訊息內聲稱的 VF/session 仍要另經 lab_tuple 驗證。 */
116int lab_peer(int fd,uid_t uid,pid_t pid)
117{
118    struct ucred c; socklen_t n=sizeof(c);
119    if(getsockopt(fd,SOL_SOCKET,SO_PEERCRED,&c,&n)) return -1;
120    if(n!=sizeof(c) || c.uid!=uid || c.pid!=pid) { errno=EACCES; return -1; }
121    return 0;
122}

133int lab_listen(const char *path)
134{
135    struct sockaddr_un a; int fd,saved;
136    if(addr(&a,path)) return -1;
137    fd=socket(AF_UNIX,SOCK_STREAM|SOCK_CLOEXEC|SOCK_NONBLOCK,0); if(fd<0) return -1;
138    if(bind(fd,(struct sockaddr *)&a,sizeof(a)) || chmod(path,0600) || listen(fd,8)) { saved=errno; close(fd); errno=saved; return -1; }
139    return fd;
140}
141int lab_connect(const char *path,int64_t deadline)
142{
143    struct sockaddr_un a; int fd,saved,error=0; socklen_t n=sizeof(error);
144    if(addr(&a,path)) return -1;
145    fd=socket(AF_UNIX,SOCK_STREAM|SOCK_CLOEXEC|SOCK_NONBLOCK,0); if(fd<0) return -1;
146    if(!connect(fd,(struct sockaddr *)&a,sizeof(a))) return fd;
147    if(errno!=EINPROGRESS || lab_wait(fd,POLLOUT,deadline) || getsockopt(fd,SOL_SOCKET,SO_ERROR,&error,&n) || error) {
148        saved=error?error:errno; close(fd); errno=saved; return -1;
149    }
150    return fd;
151}
152uint32_t lab_crc32(const void *p,size_t n)
153{
154    const unsigned char *b=p; uint32_t c=UINT32_MAX;
155    for(size_t i=0;i<n;i++) { c^=b[i]; for(unsigned j=0;j<8;j++) c=(c>>1)^((0u-(c&1u))&0xedb88320u); }
156    return c^UINT32_MAX;
