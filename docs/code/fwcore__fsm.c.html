28fw_status fw_transition(struct fw_context *ctx, fw_u16 vf, enum fw_event event)
29{
30    fw_size i;
31    enum fw_state to=FW_STATE_COUNT;
32    struct fw_hw_snapshot hw;
33    fw_u64 deadline=0;
34    fw_status rc;
35    if (!ctx || vf>=FW_VFS) return FW_INVAL;
36    /* 不合法的邊在任何 HAL 呼叫前拒絕，包含 context padding 都不更動。 */
37    for (i=0; i<sizeof(edges)/sizeof(edges[0]); ++i) {
38        if (edges[i].from==ctx->vf[vf].state && edges[i].event==event) { to=edges[i].to; break; }
39    }
40    if (to==FW_STATE_COUNT) return FW_STATE;
41    if (event==FW_CREATE) return fw_create(ctx,vf,8);
42    rc=fw_hw(ctx,vf,&hw); if (rc!=FW_OK) return rc;
43    if (!hw.fenced) return FW_STATE;
44    if (event==FW_FENCE || event==FW_DRAIN_OK || event==FW_FLR || event==FW_RESET_DONE) {
45        rc=fw_deadline(ctx,(event==FW_FLR || event==FW_RESET_DONE)?1000:100,&deadline);
46        if (rc!=FW_OK) return rc;
47    }
48    switch (event) {
49    case FW_COMMIT:
50        if (!ctx->vf[vf].reserve || hw.killed || hw.config_valid || hw.outstanding) return FW_STATE;
51        break;
52    case FW_ABORT_OK:
53    case FW_RELEASE_OK:
54        if (hw.pci_present || hw.config_valid || hw.outstanding ||
55            !ctx->hal.owners_closed(ctx->hal.opaque,vf) || !ctx->vf[vf].reserve ||
56            ctx->used_credit<ctx->vf[vf].reserve || !ctx->used_queues || ctx->used_vectors<2) return FW_STATE;
57        ctx->used_credit-=ctx->vf[vf].reserve; --ctx->used_queues; ctx->used_vectors-=2;
58        ctx->vf[vf].reserve=0;
59        break;
60    case FW_EV_TIMEOUT:
61        if (ctx->hal.now_tick(ctx->hal.opaque)<ctx->vf[vf].deadline) return FW_STATE;
62        /* fall through */
63    case FW_ABORT_FAIL:
64    case FW_VERIFY_FAIL:
65    case FW_RELEASE_FAIL:
66        rc=ctx->hal.kill(ctx->hal.opaque,vf); if (rc!=FW_OK) return rc;
67        break;
68    case FW_FENCE:
69        if (hw.epoch<=ctx->vf[vf].epoch) return FW_STATE;
70        rc=ctx->hal.cancel_io(ctx->hal.opaque,vf); if (rc!=FW_OK) return rc;
71        break;
72    case FW_DRAIN_OK:
73        if (hw.outstanding) return FW_STATE;
74        break;
75    case FW_RESET_DONE:
76        if (hw.outstanding) return FW_STATE;
77        rc=ctx->hal.clear_queue(ctx->hal.opaque,vf); if (rc!=FW_OK) return rc;
78        break;
79    case FW_FLR:
80        if (hw.epoch<=ctx->vf[vf].epoch || hw.outstanding || hw.config_valid) return FW_STATE;
81        break;
82    case FW_DESTROY:
83        if (!ctx->hal.owners_closed(ctx->hal.opaque,vf) || hw.outstanding ||
84            (ctx->vf[vf].state!=FW_FAILED && hw.epoch<=ctx->vf[vf].epoch)) return FW_STATE;
85        rc=ctx->hal.clear_queue(ctx->hal.opaque,vf); if (rc!=FW_OK) return rc;
86        break;
87    default:
88        return FW_STATE;
89    }
90    /* HAL 操作成功後才更新 core 狀態；失敗不能只改 state，假裝回收已完成。 */
91    ctx->vf[vf].epoch=hw.epoch; ctx->vf[vf].deadline=deadline; ctx->vf[vf].state=to;
92    fw_note(ctx,"vf_transition",vf,(fw_u64)to);
93    return FW_OK;
94}

96fw_status fw_vf_event(struct fw_context *ctx, fw_u16 vf, enum fw_event event)
97{
98    struct fw_hw_snapshot hw;
99    fw_status rc;
100    /* READY 路徑與一般 transition 分層，timeout 收斂不再形成遞迴。 */
101    if (event==FW_READY_REARM && ctx && vf<FW_VFS && ctx->vf[vf].state==FW_RECOVERING) {
102        rc=fw_hw(ctx,vf,&hw);
103        if (rc!=FW_OK) return rc;
104        return fw_rearm(ctx,vf,hw.epoch,hw.ready_gen);
105    }
106    return fw_transition(ctx,vf,event);
107}
