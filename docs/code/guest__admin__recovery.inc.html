3/* 測試同步點以 token 配對 host ACK；QMP 操作完成後才繼續，不用固定 sleep 猜進度。 */
4static int recovery_barrier(struct recovery_run *run,const char *fields)
5{
6    struct termios old,raw;
7    char token[64],expected[96],line[96];
8    size_t used=0;
9    int ret=-1,saved;
10    if (tcgetattr(STDIN_FILENO,&old)) return -1;
11    raw=old; cfmakeraw(&raw);
12    if (tcsetattr(STDIN_FILENO,TCSANOW,&raw)) return -1;
13    snprintf(token,sizeof(token),"recovery-%u",++run->serial);
14    snprintf(expected,sizeof(expected),"M3_HOST_ACK %s\n",token);
15    printf("{\"event\":\"M3_BARRIER\",\"scenario\":\"%s\",\"phase\":\"request\",\"barrier\":\"%s\",%s}\n",run->scenario,token,fields);
16    fflush(stdout);
17    int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
18    while (used<sizeof(line)-1) {
19        if (lab_wait(STDIN_FILENO,POLLIN,deadline)) goto out;
20        ssize_t n=read(STDIN_FILENO,line+used,1);
21        if (n<0&&errno==EINTR) continue;
22        if (n!=1) goto out;
23        if (line[used++]=='\n') {
24            line[used]=0;
25            if (strcmp(line,expected)) { errno=EPROTO; goto out; }
26            ret=0; goto out;
27        }
28    }
29    errno=EOVERFLOW;
30out:
31    saved=errno;
32    if (tcsetattr(STDIN_FILENO,TCSANOW,&old)) return -1;
33    errno=saved; return ret;
34}
35static int recovery_counters(struct admin *a,const char *phase,unsigned expected)
36{
37    char record[1536],*line;
38    if (lab_read_text("/sys/kernel/debug/toy_pf0/vf-stats",record,sizeof(record))) return -1;
39    for (line=record;line&&*line;line=strchr(line,'\n'),line=line?line+1:NULL) {
40        unsigned vf,desc,cpl,dma,blocked,irq;
41        if (sscanf(line,"vf=%u stale_desc=%u stale_cpl=%u dma_fault=%u fence_blocked=%u irq_suppressed=%u",&vf,&desc,&cpl,&dma,&blocked,&irq)!=6) { errno=EPROTO; return -1; }
42        if (vf!=a->vf) continue;
43        if (desc||cpl!=expected||dma||irq) { errno=EPROTO; return -1; }
44        printf("{\"event\":\"RECOVERY_COUNTERS\",\"phase\":\"%s\",\"vf\":%u,\"stale_desc\":%u,\"stale_cpl\":%u,\"dma_fault\":%u,\"fence_blocked\":%u,\"irq_suppressed\":%u}\n",phase,vf,desc,cpl,dma,blocked,irq);
45        return 0;
46    }
47    errno=ENOENT; return -1;
48}
49/* 保留一筆工作後執行 fence/reset，等待新設定與 READY，再 REARM；B 的進度另行核對。 */
50static int recovery_handshake(struct admin *a,struct admin *b,struct recovery_run *run)
51{
52    struct toy_vf_ctl ctl;
53    struct lab_message m;
54    if (recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"fault-vf\",\"value\":0")||
55        recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"fault-after\",\"value\":0")||
56        recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"delay-cpl-ticks\",\"value\":60000")||
57        owner_run(a)||snapshot(a)||a->info.outstanding!=1||a->info.state!=TOY_STATES_ACTIVE||
58        recovery_barrier(run,"\"action\":\"pending-check\",\"expected\":1")) return -1;
59    printf("{\"event\":\"RECOVERY_LOADED\",\"vf\":0,\"epoch\":%" PRIu64 ",\"outstanding\":%u,\"state\":%u,\"flags\":%u}\n",(uint64_t)a->info.epoch,a->info.outstanding,a->info.state,a->info.hw_flags);
60    uint64_t old_epoch=a->info.epoch;
61    m=a->current; m.type=LAB_STOP_REQUEST; m.payload_len=16; memset(m.payload,0,16);
62    printf("{\"event\":\"RECOVERY_STOP_REQUEST\",\"vf\":0,\"epoch\":%" PRIu64 "}\n",m.epoch);
63    if (healthy_checkpoint(b,"pending")||lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||
64        control(a,TOY_FENCE_VF,0,&ctl)||tenant_message(a,&m,LAB_STOPPED,4)||
65        lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
66        m.ready_gen!=a->original.ready_gen||snapshot(a)||
67        control(a,TOY_RESET_VF,0,&ctl)||reset_snapshot(a,"recovering",TOY_STATES_RECOVERING,0)||
68        a->info.epoch<=old_epoch||recovery_counters(a,"reset",0)||
69        recovery_barrier(run,"\"action\":\"pending-check\",\"expected\":0")||
70        recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"delay-cpl-ticks\",\"value\":0")) return -1;
71    m=a->current; m.type=LAB_RESET_NOTICE; m.epoch=a->info.epoch; m.payload_len=16;
72    memset(m.payload,0,16); lab_put32(m.payload+4,a->info.hw_flags); lab_put32(m.payload+8,a->info.outstanding);
73    a->original=m;
74    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_HELLO,5)||snapshot(a)||
75        lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
76        m.ready_gen!=a->original.ready_gen||a->info.epoch!=m.epoch) { errno=ESTALE; return -1; }
77    a->current=m;
78    if (recovery_barrier(run,"\"action\":\"virtual-delay\",\"ticks\":200")||healthy_checkpoint(b,"slow_ready")||
79        owner_inspect(a)||reset_snapshot(a,"slow_ready",TOY_STATES_RECOVERING,0)) return -1;
80    if (m.ready_gen==UINT32_MAX) { errno=EOVERFLOW; return -1; }
81    m.type=LAB_HELLO_REPLY; ++m.ready_gen; a->original=m;
82    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_READY,6)||
83        reset_snapshot(a,"fresh_config",TOY_STATES_RECOVERING,1)||
84        lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
85        m.ready_gen!=a->original.ready_gen||a->info.ready_gen!=m.ready_gen||a->info.hw_flags!=61) { errno=ESTALE; return -1; }
86    a->current=m;
87    if (a->recovery_case==8) {
88        if (recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"try-stale-cpl\",\"value\":true")||
89            recovery_counters(a,"callback_fenced",1)) return -1;
90    } else if (recovery_barrier(run,"\"action\":\"virtual-delay\",\"ticks\":50")) return -1;
91    if (owner_inspect(a)||healthy_checkpoint(b,"configured_fenced")||control(a,TOY_REARM_VF,m.ready_gen,&ctl)||
92        reset_snapshot(a,"rearmed",TOY_STATES_ACTIVE,1)) return -1;
93    if (a->recovery_case==8) {
94        if (recovery_barrier(run,"\"action\":\"qmp-set\",\"property\":\"try-stale-cpl\",\"value\":true")||
95            recovery_counters(a,"callback_rearmed",2)||owner_inspect(a)) return -1;
96    }
97    if (owner_run(a)||recovery_counters(a,"finished",a->recovery_case==8?2:0)||
98        healthy_checkpoint(b,"fresh_io")||owner_inspect(a)) return -1;
99    return 0;
100}
101static int recovery_case(const char *pf,const char *device,const char *tenant,unsigned number)
102{
103    const char *scenario=number==7?"T07":"T08";
104    struct recovery_run run={.scenario=scenario};
105    struct admin owners[2];
106    char count_path[PATH_MAX];
107    bool enabled=false;
108    int ret=1,saved;
109    snprintf(count_path,sizeof(count_path),"/sys/bus/pci/devices/%s/sriov_numvfs",pf);
110    for (unsigned i=0;i<2;i++) owners[i]=(struct admin){.pf=-1,.listen=-1,.connection=-1,.pidfd=-1,.management=-1,
111        .vf=i,.failpoint="",.fail_hit=1,.scenario=scenario,.count_path=count_path,.healthy=i,.recovery_case=number};
112    if (lab_write_text(count_path,"4\n")) goto out;
113    enabled=true;
114    for (unsigned i=0;i<2;i++) if (isolated_owner_start(&owners[i],pf,device,tenant,16,4096)) goto out;
115    struct admin *a=&owners[0],*b=&owners[1];
116    struct lab_message m=b->current;
117    m.type=LAB_RUN;
118    if (lab_send(b->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||owner_run(a)||
119        recovery_handshake(a,b,&run)) goto out;
120    m=b->current; m.type=LAB_STOP_REQUEST; m.payload_len=16; memset(m.payload,0,16);
121    if (lab_send(b->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(b,&m,LAB_STOPPED,3)||
122        lab_tuple(&m,b->original.vf,b->original.session,b->original.epoch,b->original.lease_token,b->original.operation)||
123        m.ready_gen!=b->original.ready_gen) goto out;
124    b->current=m;
125    for (unsigned i=0;i<2;i++) if (owner_stop_begin(&owners[i])) goto out;
126    for (unsigned i=0;i<2;i++) if (owner_stop_finish(&owners[i])||release_owner(&owners[i])) goto out;
127    if (lab_write_text(count_path,"0\n")) goto out;
128    enabled=false;
129    uint64_t count;
130    if (lab_read_u64(count_path,&count)||count) { errno=EBUSY; goto out; }
131    for (unsigned i=0;i<4;i++) {
132        char path[PATH_MAX]; struct stat st;
133        snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/virtfn%u",pf,i);
134        if (!lstat(path,&st)||errno!=ENOENT) { errno=EBUSY; goto out; }
135    }
136    for (unsigned i=0;i<8;i++) {
137        struct toy_vf_info info={.hdr=header(sizeof(info)),.vf=i};
138        if (ioctl(a->pf,TOY_GET_VF,&info)||info.state||info.reserve||info.outstanding||info.lease_token||info.hw_flags!=1) { errno=EIO; goto out; }
139        printf("{\"event\":\"RECOVERY_FINAL\",\"vf\":%u,\"state\":%u,\"reserve\":%u,\"outstanding\":%u,\"flags\":%u,\"lease\":%" PRIu64 "}\n",i,info.state,info.reserve,info.outstanding,info.hw_flags,(uint64_t)info.lease_token);
140    }
141    printf("{\"event\":\"M3_CASE\",\"id\":\"%s\",\"status\":\"PASS\",\"pending_at_reset\":1,\"callbacks\":%u,\"all_free\":true,\"owner_closed\":true}\n",scenario,number==8?2:0);
142    puts("LAB_RESULT=PASS"); ret=0;
143out:
144    saved=errno;
145    for (unsigned i=0;i<2;i++) isolated_owner_cleanup(&owners[i]);
146    if (enabled&&lab_write_text(count_path,"0\n")) fprintf(stderr,"LABD_DISABLE_FAILED errno=%d\n",errno);
147    if (ret) {
148        fprintf(stderr,"LABD_ERROR scenario=%s errno=%d message=%s\n",scenario,saved,strerror(saved));
149        printf("{\"event\":\"M3_CASE\",\"id\":\"%s\",\"status\":\"FAIL\",\"errno\":%d}\n",scenario,saved);
150        puts("LAB_RESULT=FAIL");
151    }
152    return ret;
