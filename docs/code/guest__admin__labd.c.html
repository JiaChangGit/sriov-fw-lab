44static int snapshot(struct admin *a)
45{
46    memset(&a->info,0,sizeof(a->info)); a->info.hdr=header(sizeof(a->info)); a->info.vf=a->vf;
47    return ioctl(a->pf,TOY_GET_VF,&a->info);

49static int control(struct admin *a,unsigned long cmd,uint32_t generation,struct toy_vf_ctl *result)
50{
51    if (a->operation==UINT64_MAX) { errno=EOVERFLOW; return -1; }
52    memset(result,0,sizeof(*result)); result->hdr=header(sizeof(*result));
53    result->vf=a->vf; result->session=a->info.session; result->epoch=a->info.epoch;
54    result->lease_token=a->lease; result->operation=++a->operation; result->ready_gen=generation;
55    if (cmd==TOY_LEASE_RELEASE) result->flags=TOY_CTL_OWNER_CLOSED;
56    int rc=ioctl(a->pf,cmd,result),saved=errno;
57    printf("{\"event\":\"M3_CONTROL\",\"vf\":%u,\"command\":%lu,\"operation\":%" PRIu64 ",\"errno\":%d,\"session\":%" PRIu64 ",\"epoch\":%" PRIu64 ",\"state\":%u,\"flags\":%u,\"outstanding\":%u,\"fw_status\":%u}\n",a->vf,_IOC_NR(cmd),a->operation,rc?saved:0,(uint64_t)result->session,(uint64_t)result->epoch,result->state,result->hw_flags,result->outstanding,result->fw_status);
58    errno=saved; return rc;
59}
60static int child_wait(struct admin *a,int timeout)
61{
62    int status;
63    if (a->child_reaped||a->child<=0) return 0;
64    if (lab_wait(a->pidfd,POLLIN,lab_now_ms()+timeout)) return -1;
65    pid_t got;
66    do { got=waitpid(a->child,&status,0); } while (got<0&&errno==EINTR);
67    if (got!=a->child) return -1;
68    a->child_reaped=true;
69    if (!WIFEXITED(status)||WEXITSTATUS(status)) { errno=ECHILD; return -1; }
70    printf("{\"event\":\"M3_OWNER_REAP\",\"vf\":%u,\"pid\":%d,\"exit_code\":0,\"pidfd\":true}\n",a->vf,a->child);
71    return 0;
72}
73static int tenant_message(struct admin *a,struct lab_message *m,uint16_t type,uint64_t request)
74{
75    int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
76    for (;;) {
77        struct pollfd fds[3]={{a->connection,POLLIN,0},{a->pidfd,POLLIN,0},{a->management,POLLIN,0}};
78        int64_t remain=deadline-lab_now_ms();
79        if (remain<=0) { errno=ETIMEDOUT; return -1; }
80        int count=poll(fds,3,(int)remain);
81        if (count<0&&errno==EINTR) continue;
82        if (count<0) return -1;
83        if (!count) { errno=ETIMEDOUT; return -1; }
84        if (fds[2].revents&POLLIN) {
85            int fd=accept4(a->management,NULL,NULL,SOCK_CLOEXEC|SOCK_NONBLOCK);
86            if (fd>=0) {
87                struct ucred cred; socklen_t size=sizeof(cred);
88                struct lab_message query;
89                if (!getsockopt(fd,SOL_SOCKET,SO_PEERCRED,&cred,&size)&&size==sizeof(cred)&&cred.uid==getuid()&&
90                    !lab_recv(fd,&query,lab_now_ms()+1000)&&query.type==LAB_HELLO&&!snapshot(a)) {
91                    query.type=LAB_HELLO_REPLY; query.vf=a->vf; query.session=a->info.session;
92                    query.epoch=a->info.epoch; query.lease_token=a->lease;
93                    query.ready_gen=a->info.ready_gen; query.operation=a->operation;
94                    (void)lab_send(fd,&query,lab_now_ms()+1000);
95                }
96                close(fd);
97            }
98        }
99        if (fds[0].revents&POLLIN) break;
100        if ((fds[0].revents&(POLLHUP|POLLERR))||(fds[1].revents&POLLIN)) { errno=ECHILD; return -1; }
101    }
102    if (lab_recv(a->connection,m,deadline)) return -1;
103    if (m->type==LAB_ERROR) {
104        a->owner_error=true; a->failure=*m;
105        fprintf(stderr,"CHILD_REPORTED_ERROR errno=%u stage=%u\n",lab_get32(m->payload),lab_get32(m->payload+4));
106        errno=EIO; return -1;
107    }
108    if (m->type!=type||m->request_id!=request||m->vf!=a->vf||m->lease_token!=a->lease) { errno=EPROTO; return -1; }
109    return 0;
110}

116    snprintf(lease_text,sizeof(lease_text),"%" PRIu64,a->lease); snprintf(operation_text,sizeof(operation_text),"%" PRIu64,a->operation);

132              "--vfio-reset",a->vfio_reset?"1":"0","--healthy",a->healthy?"1":"0",
133              "--inspect",(!strcmp(a->scenario,"T04")||!strcmp(a->scenario,"T05")||!strcmp(a->scenario,"T06")||a->recovery_case)?"1":"0",(char *)NULL);
134        _exit(127);
135    }
136    a->pidfd=(int)syscall(SYS_pidfd_open,a->child,0); if (a->pidfd<0) return -1;
137    if (lab_wait(a->listen,POLLIN,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
138    a->connection=accept4(a->listen,NULL,NULL,SOCK_CLOEXEC|SOCK_NONBLOCK);
139    if (a->connection<0||lab_peer(a->connection,getuid(),a->child)) return -1;
140    printf("{\"event\":\"M3_PEER_VERIFIED\",\"vf\":%u,\"pid\":%d,\"uid\":%u,\"pidfd\":true}\n",a->vf,a->child,getuid());
141    return 0;

143static int busy_operations(struct admin *a)
144{
145    struct toy_vf_info before[8],after;
146    char links[4][32],path[PATH_MAX],actual[32];
147    uint64_t count;
148    const char *values[]={"0\n","2\n"};
149    for (unsigned i=0;i<8;i++) {
150        before[i]=(struct toy_vf_info){.hdr=header(sizeof(before[i])),.vf=i};
151        if (ioctl(a->pf,TOY_GET_VF,&before[i])||before[i].reserve!=(i<4?8u:0u)) return -1;
152        if (i<4) {
153            /* 使用 PF 的 count 路徑組出每一個 virtfn link。 */
154            snprintf(path,sizeof(path),"%.*svirtfn%u",(int)(strlen(a->count_path)-strlen("sriov_numvfs")),a->count_path,i);
155            if (lab_link_basename(path,links[i],sizeof(links[i]))) return -1;
156        }
157    }
158    if (!before[a->vf].lease_token) { errno=EPROTO; return -1; }
159    for (unsigned op=0;op<2;op++) {
160        errno=0;
161        int rc=lab_write_text(a->count_path,values[op]),error=errno;
162        if (!rc||(error!=EBUSY&&error!=EPERM)||lab_read_u64(a->count_path,&count)||count!=4) { errno=EPROTO; return -1; }
163        for (unsigned i=0;i<8;i++) {
164            after=(struct toy_vf_info){.hdr=header(sizeof(after)),.vf=i};
165            if (ioctl(a->pf,TOY_GET_VF,&after)||memcmp(&before[i],&after,sizeof(after))) { errno=EPROTO; return -1; }
166            if (i<4) {
167                snprintf(path,sizeof(path),"%.*svirtfn%u",(int)(strlen(a->count_path)-strlen("sriov_numvfs")),a->count_path,i);
168                if (lab_link_basename(path,actual,sizeof(actual))||strcmp(actual,links[i])) { errno=EPROTO; return -1; }
169            }
170            printf("{\"event\":\"T10_SNAPSHOT\",\"attempt\":%u,\"vf\":%u,\"state\":%u,\"reserve\":%u,\"flags\":%u,\"outstanding\":%u,\"epoch\":%" PRIu64 ",\"lease\":%" PRIu64 ",\"unchanged\":true}\n",op,i,after.state,after.reserve,after.hw_flags,after.outstanding,(uint64_t)after.epoch,(uint64_t)after.lease_token);
171        }
172        printf("{\"event\":\"T10_BUSY\",\"attempt\":%u,\"requested\":%u,\"errno\":%d,\"enabled\":%" PRIu64 ",\"topology_unchanged\":true,\"ledger_unchanged\":true,\"owner_pid\":%d}\n",op,op?2:0,error,count,a->child);
173    }
174    return 0;
175}
176static int owner_prepare(struct admin *a)
177{
178    struct lab_message m;
179    struct toy_vf_ctl ctl;
180    if (tenant_message(a,&m,LAB_HELLO,1)||snapshot(a)) return -1;
181    if (lab_tuple(&m,a->vf,a->info.session,a->info.epoch,a->lease,a->operation)||a->info.ready_gen==UINT32_MAX) { errno=ESTALE; return -1; }
182    m.type=LAB_HELLO_REPLY; m.ready_gen=a->info.ready_gen+1; a->original=m;
183    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_READY,2)||snapshot(a)) return -1;
184    if (lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
185        m.ready_gen!=a->original.ready_gen||a->info.ready_gen!=m.ready_gen||a->info.epoch!=m.epoch||a->info.session!=m.session) { errno=ESTALE; return -1; }
186    if (control(a,TOY_REARM_VF,m.ready_gen,&ctl)) return -1;
187    a->current=m;
188    return 0;
189}
190static int owner_run(struct admin *a)
191{
192    struct lab_message m=a->current;
193    if (!strcmp(a->scenario,"T10")&&busy_operations(a)) return -1;
194    m.type=LAB_RUN;
195    if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)||tenant_message(a,&m,LAB_STOPPED,a->current.request_id+1)) return -1;
196    if (lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
197        m.ready_gen!=a->original.ready_gen) { errno=ESTALE; return -1; }
198    a->current=m;
199    return 0;
200}
201/* 停止拆成發送與等待兩段，讓多 VF teardown 能先通知全部 owner，再逐一收尾。 */
202static int owner_stop_begin(struct admin *a)
203{
204    struct lab_message m=a->current;
205    struct toy_vf_ctl ctl;
206    if (snapshot(a)) return -1;
207    if (control(a,TOY_FENCE_VF,0,&ctl)||ctl.outstanding||!(ctl.hw_flags&TOY_HW_FLAGS_FENCED)) return -1;
208    m.type=LAB_STOP; m.payload_len=16; memset(m.payload,0,16); lab_put32(m.payload+4,ctl.hw_flags);
209    return lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS);
210}
211static int owner_stop_finish(struct admin *a)
212{
213    struct lab_message m;
214    if (tenant_message(a,&m,LAB_EXIT_ACK,a->current.request_id)) return -1;
215    if (lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
216        m.ready_gen!=a->original.ready_gen) { errno=ESTALE; return -1; }
217    for (unsigned i=0;i<16;i++) if (m.payload[i]) { errno=EBUSY; return -1; }
218    return child_wait(a,LAB_TIMEOUT_MS);
219}
220static int owner_stop(struct admin *a)
221{ return owner_stop_begin(a)||owner_stop_finish(a)?-1:0; }
222static int descriptor_counters(struct admin *a,const char *phase)
223{
224    char record[1536],*line;
225    if (lab_read_text("/sys/kernel/debug/toy_pf0/vf-stats",record,sizeof(record))) return -1;
226    for (line=record;line&&*line;line=strchr(line,'\n'),line=line?line+1:NULL) {
227        unsigned vf,desc,cpl,dma,blocked,irq;
228        if (sscanf(line,"vf=%u stale_desc=%u stale_cpl=%u dma_fault=%u fence_blocked=%u irq_suppressed=%u",&vf,&desc,&cpl,&dma,&blocked,&irq)!=6) { errno=EPROTO; return -1; }
229        if (vf!=a->vf) continue;
230        if (desc!=(!strcmp(phase,"after")?1u:0u)||cpl||dma||blocked||irq) { errno=EPROTO; return -1; }
231        printf("{\"event\":\"T09_COUNTERS\",\"phase\":\"%s\",\"vf\":%u,\"stale_desc\":%u,\"stale_cpl\":%u,\"dma_fault\":%u,\"fence_blocked\":%u,\"irq_suppressed\":%u}\n",phase,vf,desc,cpl,dma,blocked,irq);
232        return 0;
233    }
234    errno=ENOENT; return -1;
235}

269static int failed_owner(struct admin *a)
270{
271    struct lab_message m=a->failure;
272    struct toy_vf_ctl ctl;
273    unsigned stage=lab_get32(m.payload+4);
274    if (!a->owner_error||!*a->failpoint||m.vf!=a->vf||m.lease_token!=a->lease||
275        lab_get32(m.payload)!=EIO||(stage!=1&&stage!=3)||m.request_id!=(stage==1?2u:4u)||
276        lab_tuple(&m,a->original.vf,a->original.session,a->original.epoch,a->original.lease_token,a->original.operation)||
277        m.ready_gen!=a->original.ready_gen) { errno=EPROTO; return -1; }
278    if (stage==1) {
279        if (snapshot(a)||lab_tuple(&m,a->vf,a->info.session,a->info.epoch,a->lease,a->operation)||
280            control(a,TOY_FENCE_VF,0,&ctl)||ctl.outstanding||!(ctl.hw_flags&TOY_HW_FLAGS_FENCED)) return -1;
281        m.type=LAB_STOP; m.payload_len=16; memset(m.payload,0,16); lab_put32(m.payload+4,ctl.hw_flags);
282        if (lab_send(a->connection,&m,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
283    }
284    if (lab_wait(a->pidfd,POLLIN,lab_now_ms()+LAB_TIMEOUT_MS)) return -1;
285    int status; pid_t got;
286    do { got=waitpid(a->child,&status,0); } while (got<0&&errno==EINTR);
287    if (got!=a->child) return -1;
288    a->child_reaped=true;
289    if (!WIFEXITED(status)||WEXITSTATUS(status)!=42) { errno=ECHILD; return -1; }
290    printf("{\"event\":\"T02_OWNER_REAP\",\"pid\":%d,\"exit_code\":42,\"stage\":%u}\n",a->child,stage);
291    return 0;
292}
293static int owner_fault_matrix(const char *program)
294{
295    static const struct { const char *name; unsigned hits; } points[]={
296        {"owner.alloc_buffer",3},{"owner.map_sq",1},{"owner.map_cq",1},{"owner.map_payload",1},
297        {"owner.eventfd",2},{"owner.irq_register",1},{"owner.unmap",3},{"owner.short_unmap",3}};
298    unsigned cases=0;
299    for (unsigned p=0;p<sizeof(points)/sizeof(points[0]);p++) for (unsigned hit=1;hit<=points[p].hits;hit++) {
300        ++cases;
301        for (unsigned clean=0;clean<2;clean++) {
302            char hit_text[16]; snprintf(hit_text,sizeof(hit_text),"%u",clean?1:hit);
303            printf("{\"event\":\"T02_OWNER_START\",\"case_index\":%u,\"name\":\"%s\",\"hit\":%u,\"clean_retry\":%s}\n",cases,points[p].name,hit,clean?"true":"false");
304            fflush(stdout);
305            pid_t child=fork(); if (child<0) return 1;
306            if (!child) {
307                if (syscall(SYS_close_range,3u,UINT_MAX,0u)) _exit(125);

309                      "--fail-hit",hit_text,(char *)NULL);
310                _exit(127);
311            }
312            int status; pid_t got;
313            do { got=waitpid(child,&status,0); } while (got<0&&errno==EINTR);
314            if (got!=child||!WIFEXITED(status)||WEXITSTATUS(status)) { puts("LAB_RESULT=FAIL"); return 1; }
315        }
316    }
317    printf("{\"event\":\"M3_CASE\",\"id\":\"T02_OWNER\",\"status\":\"PASS\",\"variants\":%u,\"clean_retries\":%u,\"partial_pci_enumeration\":\"NOT_RUN\"}\n",cases,cases);
318    puts("LAB_RESULT=PASS"); return 0;
319}
320static int isolated_group(struct admin *a)
321{
322    char path[PATH_MAX];
323    snprintf(path,sizeof(path),"/sys/kernel/iommu_groups/%u/devices",a->group);
324    DIR *directory=opendir(path);
325    if (!directory) return -1;
326    struct dirent *entry;
327    unsigned members=0;
328    while ((entry=readdir(directory))) {
329        if (entry->d_name[0]=='.') continue;
330        /* 本拓樸要求群組只有自己的 VF，PF 與其他 VF 一律拒絕。 */
331        if (strcmp(entry->d_name,a->bdf)) { closedir(directory); errno=EXDEV; return -1; }
332        ++members;
333    }
334    closedir(directory);
335    if (members!=1) { errno=EPROTO; return -1; }
336    printf("{\"event\":\"%s_GROUP\",\"vf\":%u,\"bdf\":\"%s\",\"group\":%u,\"members\":%u}\n",a->scenario,a->vf,a->bdf,a->group,members);
337    return 0;
338}
339static int isolated_owner_start(struct admin *a,const char *pf,const char *device,const char *tenant,unsigned entries,unsigned requests)
340{
341    char path[PATH_MAX],group_text[32],*end;
342    struct toy_vf_ctl ctl;
343    snprintf(a->directory,sizeof(a->directory),"/run/lab-%d-vf%u",getpid(),a->vf);

351    if (lab_link_basename(path,a->bdf,sizeof(a->bdf))||snapshot(a)||control(a,TOY_LEASE_ACQUIRE,0,&ctl)) return -1;
352    a->lease=ctl.lease_token; a->leased=true;

364static void isolated_owner_cleanup(struct admin *a)
365{
366    struct toy_vf_ctl ctl;
367    if (a->leased&&a->pf>=0&&!a->child_reaped&&!snapshot(a)) (void)control(a,TOY_FENCE_VF,0,&ctl);
368    if (a->child>0&&!a->child_reaped) {
369        if (a->pidfd>=0) (void)syscall(SYS_pidfd_send_signal,a->pidfd,SIGKILL,NULL,0);
370        else (void)kill(a->child,SIGKILL);
371        int status; while (waitpid(a->child,&status,0)<0&&errno==EINTR) {}
372        a->child_reaped=true;
373    }
374    if (a->pf>=0&&release_owner(a)) fprintf(stderr,"LABD_QUARANTINE vf=%u cleanup_errno=%d\n",a->vf,errno);
375    if (a->connection>=0) close(a->connection);
376    if (a->listen>=0) close(a->listen);
377    if (a->management>=0) close(a->management);
378    if (a->pidfd>=0) close(a->pidfd);
379    if (a->pf>=0) close(a->pf);
380    if (a->socket[0]) unlink(a->socket);
381    if (a->management_path[0]) unlink(a->management_path);
382    if (a->directory[0]) rmdir(a->directory);
383}

505static int directory_entries(const char *path)
506{

519static int lifecycle_quiet(int fd,unsigned cycle,int baseline_fds,int baseline_tasks)
520{
521    unsigned contexts,leases,orphaned,events,work,pending,enabled,transaction,drained=0;
522    char line[256];
523    struct toy_event event;
524    int64_t deadline=lab_now_ms()+5000;
525    for (;;) {
526        ssize_t got;
527        do {
528            got=read(fd,&event,sizeof(event));
529            if (got==(ssize_t)sizeof(event)) {
530                if (event.hdr.size!=sizeof(event)||event.type==TOY_EVENT_LOST_EVENTS||++drained>256) { errno=EPROTO; return -1; }
531            } else if (got>=0) { errno=EPROTO; return -1; }
532        } while (got>0||(got<0&&errno==EINTR));
533        if (errno!=EAGAIN) return -1;
534        if (lab_read_text("/sys/kernel/debug/toy_pf0/lifecycle",line,sizeof(line))) return -1;
535        if (sscanf(line,"contexts=%u leases=%u orphaned=%u events=%u work_busy=%u pending_irqs=%u enabled=%u transaction=%u",
536                   &contexts,&leases,&orphaned,&events,&work,&pending,&enabled,&transaction)!=8) { errno=EPROTO; return -1; }
537        if (contexts!=1||leases||orphaned||enabled||transaction) {
538            fprintf(stderr,"T01_DRIVER_REJECT %s",line); errno=EBUSY; return -1;
539        }
540        if (!events&&!work&&!pending) break;
541        if (lab_now_ms()>=deadline) { errno=ETIMEDOUT; return -1; }
542        usleep(1000);
543    }
544    int fds=directory_entries("/proc/self/fd"),tasks=directory_entries("/proc/self/task");
545    /* waitpid 的 ECHILD 同時確認沒有仍存活或尚未 reap 的子程序，不依賴選配 proc children。 */
546    int child_status;
547    pid_t child;
548    do { child=waitpid(-1,&child_status,WNOHANG); } while (child<0&&errno==EINTR);
549    if (child!=-1||errno!=ECHILD||fds!=baseline_fds||tasks!=baseline_tasks) {
550        fprintf(stderr,"T01_PROCESS_REJECT fds=%d baseline_fds=%d tasks=%d baseline_tasks=%d waitpid=%d errno=%d\n",fds,baseline_fds,tasks,baseline_tasks,child,errno);
551        errno=EBUSY; return -1;
552    }
553    printf("{\"event\":\"T01_LIFECYCLE\",\"cycle\":%u,\"contexts\":%u,\"leases\":%u,\"orphaned\":%u,\"events\":%u,\"work_busy\":%u,\"pending_irqs\":%u,\"drained\":%u,\"fds\":%d,\"baseline_fds\":%d,\"tasks\":%d,\"baseline_tasks\":%d,\"children\":0}\n",cycle,contexts,leases,orphaned,events,work,pending,drained,fds,baseline_fds,tasks,baseline_tasks);
554    return 0;
555}

574static int lifecycle_case(const char *pf,const char *device,const char *tenant)
575{
576    char count_path[PATH_MAX];
577    bool enabled=false;
578    int ret=1,saved,fd=open(device,O_RDWR|O_NONBLOCK|O_CLOEXEC);
579    snprintf(count_path,sizeof(count_path),"/sys/bus/pci/devices/%s/sriov_numvfs",pf);
580    if (fd<0) goto out;
581    int baseline_fds=directory_entries("/proc/self/fd"),baseline_tasks=directory_entries("/proc/self/task");
582    if (baseline_fds<4||baseline_tasks!=1||lab_inventory_checkpoint("T01",0,"cold")) goto out;
583    /* 暖機也走完整 owner 生命週期，之後另外量測一百次。 */
584    for (unsigned cycle=0;cycle<=100;cycle++) {
585        char bdfs[4][32];
586        struct admin owners[4];
587        int failed=0,error=0;
588        for (unsigned vf=0;vf<4;vf++) owners[vf]=(struct admin){.pf=-1,.listen=-1,.connection=-1,.pidfd=-1,.management=-1,
589            .vf=vf,.failpoint="",.fail_hit=1,.scenario="T01",.count_path=count_path};

596            if (isolated_owner_start(&owners[vf],pf,device,tenant,16,0)||owner_run(&owners[vf])) { failed=1; break; }
597            strcpy(bdfs[vf],owners[vf].bdf);
598        }

608        }
609        error=errno;
610        for (unsigned vf=0;vf<4;vf++) isolated_owner_cleanup(&owners[vf]);
611        if (failed) { errno=error; goto out; }
612        for (unsigned vf=0;vf<4;vf++) {
613            char path[PATH_MAX]; struct stat st;
614            snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s",bdfs[vf]);
615            if (!lstat(path,&st)||errno!=ENOENT) { errno=EBUSY; goto out; }
616        }
617        if (cycle_ledger(fd,pf,cycle,false)||lifecycle_quiet(fd,cycle,baseline_fds,baseline_tasks)) goto out;
618        printf("{\"event\":\"T01_CYCLE_END\",\"cycle\":%u,\"owners\":4,\"pci_devices_absent\":4,\"status\":\"PASS\"}\n",cycle);
619        if (!cycle&&lab_inventory_checkpoint("T01",1,"initial")) goto out;
620    }
621    if (lab_inventory_checkpoint("T01",2,"final")) goto out;
622    puts("{\"event\":\"M3_CASE\",\"id\":\"T01\",\"status\":\"PASS\",\"cycles\":100,\"warmup_cycles\":1,\"owners_per_cycle\":4,\"remaining_fds\":0,\"remaining_maps\":0,\"remaining_work\":0}");
623    puts("LAB_RESULT=PASS"); ret=0;
624out:
625    saved=errno;
626    if (enabled&&lab_write_text(count_path,"0\n")) fprintf(stderr,"LABD_DISABLE_FAILED errno=%d\n",errno);
627    if (fd>=0) close(fd);
628    if (ret) {
629        fprintf(stderr,"LABD_ERROR scenario=T01 errno=%d message=%s\n",saved,strerror(saved));
630        printf("{\"event\":\"M3_CASE\",\"id\":\"T01\",\"status\":\"FAIL\",\"errno\":%d}\n",saved);
631        puts("LAB_RESULT=FAIL");
632    }
633    return ret;
634}

678    snprintf(a.management_path,sizeof(a.management_path),"%s/control.sock",a.directory);
679    a.management=lab_listen(a.management_path); if (a.management<0) goto out;
680    printf("{\"event\":\"M3_ADMIN_SOCKET\",\"path\":\"%s\"}\n",a.management_path);
681    snprintf(count_path,sizeof(count_path),"/sys/bus/pci/devices/%s/sriov_numvfs",pf);
682    a.pf=open(device,O_RDWR|O_CLOEXEC|O_NONBLOCK); if (a.pf<0) goto out;
683    if (!strcmp(scenario,"T02-provision")) {
684        if (!lab_t02_provision(pf,a.pf)) ret=0;
685        goto out;
686    }
687    if (!strcmp(scenario,"T02-pci")) {
688        if (!lab_t02_pci(pf,a.pf)) ret=0;
689        goto out;
690    }
691    if (lab_write_text(count_path,"4\n")) goto out;
692    enabled=true;
693    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/virtfn0",pf);
694    if (lab_link_basename(path,a.bdf,sizeof(a.bdf))||snapshot(&a)||control(&a,TOY_LEASE_ACQUIRE,0,&ctl)) goto out;
695    a.lease=ctl.lease_token; a.leased=true;
696    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/iommu_group",a.bdf);
697    if (lab_link_basename(path,group_text,sizeof(group_text))) goto out;
698    char *end; errno=0; unsigned long group=strtoul(group_text,&end,10);
699    if (errno||*end||group>UINT_MAX) { errno=EPROTO; goto out; }
700    snprintf(path,sizeof(path),"/sys/bus/pci/devices/%s/driver_override",a.bdf);
701    if (lab_write_text(path,"vfio-pci\n")||lab_write_text("/sys/bus/pci/drivers_probe",a.bdf)) goto out;
702    a.bound=true;
703    if (start_owner(&a,tenant,(unsigned)group,entries,requests)) goto out;
704    int session_ret=owner_session(&a);
705    if (*a.failpoint) {
706        if (!session_ret||failed_owner(&a)) goto out;
707    } else if (session_ret) goto out;
708    if (release_owner(&a)) goto out;
709    if (lab_write_text(count_path,"0\n")) goto out;
710    enabled=false;
711    for (a.vf=0;a.vf<8;a.vf++) {
712        if (snapshot(&a)||a.info.state!=TOY_STATES_FREE||a.info.reserve||a.info.outstanding||
713            (a.info.hw_flags&(TOY_HW_FLAGS_PCI_PRESENT|TOY_HW_FLAGS_CONFIG_VALID))||!(a.info.hw_flags&TOY_HW_FLAGS_FENCED)) { errno=EIO; goto out; }
714    }
715    if (!strcmp(scenario,"T02-owner-child")) {
716        printf("{\"event\":\"T02_OWNER_CASE\",\"name\":\"%s\",\"hit\":%u,\"pid\":%d,\"owner_closed\":true,\"all_free\":true,\"status\":\"PASS\"}\n",a.failpoint,a.fail_hit,a.child);
717    } else {
718        printf("{\"event\":\"M3_CASE\",\"id\":\"%s\",\"scenario\":\"%s\",\"variant\":\"entries-%u\",\"status\":\"PASS\",\"checks\":{\"entries\":%u,\"requests\":%u,\"seed\":%u,\"owner_closed\":true,\"all_free\":true}}\n",scenario,scenario,entries,entries,requests,seed);
719        puts("LAB_RESULT=PASS");
720    }
721    ret=0;
722out:
723    saved=errno;
724    if (ret) fprintf(stderr,"LABD_ERROR scenario=%s errno=%d message=%s\n",scenario,saved,strerror(saved));
725    if (a.leased&&a.vf<8&&a.pf>=0&&!a.child_reaped) {
726        if (!snapshot(&a)) (void)control(&a,TOY_FENCE_VF,0,&ctl);
727    }
728    if (a.child>0&&!a.child_reaped) {
729        if (a.pidfd>=0) (void)syscall(SYS_pidfd_send_signal,a.pidfd,SIGKILL,NULL,0);
730        else (void)kill(a.child,SIGKILL);
731        int status; while (waitpid(a.child,&status,0)<0&&errno==EINTR) {}
732        a.child_reaped=true;
733    }
734    if (ret&&a.pf>=0) {
735        if (release_owner(&a)) fprintf(stderr,"LABD_QUARANTINE cleanup_errno=%d\n",errno);
736        else if (enabled&&lab_write_text(count_path,"0\n")) fprintf(stderr,"LABD_DISABLE_FAILED errno=%d\n",errno);
737    }
738    if (a.connection>=0) close(a.connection);
739    if (a.listen>=0) close(a.listen);
740    if (a.management>=0) close(a.management);
741    if (a.pidfd>=0) close(a.pidfd);
742    if (a.pf>=0) close(a.pf);
743    if (a.socket[0]) unlink(a.socket);
744    if (a.management_path[0]) unlink(a.management_path);
745    if (a.directory[0]) rmdir(a.directory);
746    if (ret) { printf("{\"event\":\"M3_CASE\",\"id\":\"%s\",\"status\":\"FAIL\",\"errno\":%d}\n",scenario,saved); puts("LAB_RESULT=FAIL"); }
747    return ret;
