12/* SQ_HEAD 必須前進而 CQ_TAIL 不變，才能證明 descriptor 已被取走且尚未完成。 */
13int owner_submit_pending(struct owner *o)
14{
15    if (o->pending_request) { errno=EBUSY; return -1; }
16    uint32_t index=lab_ring_index(o->sq_tail,o->entries),head,tail,digests[OWNER_MAPS];
17    unsigned char *sq=(unsigned char *)o->maps[OWNER_SQ].memory+index*32;
18    if (owner_freeze(o)) return -1;
19    memset(sq,0,32); lab_put64(sq,o->maps[OWNER_PAYLOAD].iova);
20    lab_put32(sq+8,8); sq[12]=TOY_SQ_OPS_FILL; lab_put16(sq+14,0x708);
21    lab_put64(sq+16,o->epoch); lab_put64(sq+24,0x79);
22    char hex[65]; static const char digits[]="0123456789abcdef";
23    for (unsigned i=0;i<32;i++) { hex[2*i]=digits[sq[i]>>4]; hex[2*i+1]=digits[sq[i]&15]; }
24    hex[64]=0; ++o->sq_tail;
25    if (owner_write32(o,TOY_VF_SQ_TAIL,o->sq_tail)||owner_write32(o,TOY_VF_DOORBELL,1)) return -1;
26    int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
27    for (;;) {
28        if (owner_read32(o,TOY_VF_SQ_HEAD,&head)||owner_read32(o,TOY_VF_CQ_TAIL,&tail)) return -1;
29        if (tail!=o->cq_head) { errno=EPROTO; return -1; }
30        if (head==o->sq_tail) break;
31        if (lab_now_ms()>=deadline) { errno=ETIMEDOUT; return -1; }
32        usleep(1000);
33    }
34    for (unsigned i=OWNER_CQ;i<OWNER_MAPS;i++)
35        if (memcmp(o->held[i],o->maps[i].memory,o->maps[i].size)) { errno=EILSEQ; return -1; }
36    if (owner_freeze(o)||owner_unchanged(o,digests)) return -1;
37    o->pending_request=true; o->pending_epoch=o->epoch;
38    printf("{\"event\":\"RECOVERY_PENDING\",\"case\":%u,\"pid\":%d,\"epoch\":%" PRIu64 ",\"session\":%" PRIu64 ",\"sq_head\":%u,\"cq_tail\":%u,\"completed\":%" PRIu64 ",\"descriptor_hex\":\"%s\",\"payload_unchanged\":true,\"cq_unchanged\":true}\n",o->recovery_case,getpid(),o->epoch,o->session,head,tail,o->completed,hex);
39    return 0;
40}

42/* 管理端已完成 fence/reset 握手才可呼叫；keep_unmapped 只保留 CPU 記憶體，不保留 IOMMU mapping。 */
43int owner_retire(struct owner *o)
44{
45    for (unsigned i=0;i<OWNER_MAPS;i++)
46        if (o->retired[i].memory||!o->maps[i].memory||!o->held[i]) { errno=EBUSY; return -1; }
47    /* 舊 IOVA 精確解除後仍保留 CPU allocations，供真實 stale callback 前後比對。 */
48    o->keep_unmapped=true;
49    int rc=owner_unmap(o,true);
50    o->keep_unmapped=false;
51    if (rc) return -1;
52    for (unsigned i=0;i<OWNER_MAPS;i++) {
53        o->retired[i]=o->maps[i]; o->retired_guard[i]=o->held[i];
54        memset(&o->maps[i],0,sizeof(o->maps[i])); o->held[i]=NULL;
55        printf("{\"event\":\"OWNER_RETIRED\",\"pid\":%d,\"kind\":%u,\"iova\":%" PRIu64 ",\"bytes\":%zu,\"vaddr\":%" PRIuPTR ",\"mapped\":false}\n",getpid(),i,o->retired[i].iova,o->retired[i].size,(uintptr_t)o->retired[i].memory);
56    }
57    return owner_retired_unchanged(o);
58}

60int owner_retired_unchanged(struct owner *o)
61{
62    for (unsigned i=0;i<OWNER_MAPS;i++) {
63        const struct owner_map *old=&o->retired[i];
64        if (!old->memory) continue;
65        if (old->mapped||!o->retired_guard[i]||memcmp(old->memory,o->retired_guard[i],old->size)||
66            old->memory==o->maps[i].memory) { errno=EILSEQ; return -1; }
67        printf("{\"event\":\"OWNER_RETIRED_CHECK\",\"pid\":%d,\"kind\":%u,\"bytes\":%zu,\"crc\":%u,\"bytewise_unchanged\":true,\"mapped\":false}\n",getpid(),i,old->size,lab_crc32(old->memory,old->size));
68    }
69    return 0;
