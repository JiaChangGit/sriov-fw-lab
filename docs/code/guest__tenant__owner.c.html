20static int fail(struct owner *o, const char *name)
21{
22    if (!o->fail_fired && !strcmp(o->failpoint,name) && ++o->fail_seen==o->fail_hit) {
23        o->fail_fired=true;
24        printf("{\"event\":\"OWNER_FAILPOINT_HIT\",\"name\":\"%s\",\"hit\":%u}\n",name,o->fail_seen);
25        errno=EIO; return 1;
26    }
27    return 0;
28}
29void owner_init(struct owner *o)
30{
31    memset(o,0,sizeof(*o)); lab_vfio_init(&o->vfio);
32    o->eventfds[0]=o->eventfds[1]=-1; o->fail_hit=1;
33}
34int owner_read32(struct owner *o,uint32_t reg,uint32_t *value)
35{
36    unsigned char b[4];
37    if ((reg&3u) || reg>o->vfio.bar0.size-4) { errno=EINVAL; return -1; }
38    if (pread(o->vfio.device,b,4,(off_t)(o->vfio.bar0.offset+reg))!=4) return -1;
39    *value=lab_get32(b); return 0;
40}
41int owner_write32(struct owner *o,uint32_t reg,uint32_t value)
42{
43    unsigned char b[4]; lab_put32(b,value);
44    if ((reg&3u) || reg>o->vfio.bar0.size-4) { errno=EINVAL; return -1; }
45    /* 先執行 C11 thread fence，再經 VFIO pwrite 寫 MMIO；本版只驗證 x86_64 QEMU 路徑。
46     * 這裡沒有替 non-coherent 實體平台做 cache maintenance。 */
47    atomic_thread_fence(memory_order_seq_cst);
48    return pwrite(o->vfio.device,b,4,(off_t)(o->vfio.bar0.offset+reg))==4?0:-1;
49}
50static int read64(struct owner *o,uint32_t reg,uint64_t *value)
51{
52    uint32_t lo,hi;
53    if (owner_read32(o,reg,&lo)||owner_read32(o,reg+4,&hi)) return -1;
54    *value=(uint64_t)lo|((uint64_t)hi<<32); return 0;
55}
56static int write64(struct owner *o,uint32_t reg,uint64_t value)
57{
58    return owner_write32(o,reg,(uint32_t)value)||owner_write32(o,reg+4,(uint32_t)(value>>32))?-1:0;
59}
60int owner_latch(struct owner *o,uint32_t *flags)
61{
62    return owner_write32(o,TOY_VF_EPOCH_LATCH,1)||
63        read64(o,TOY_VF_EPOCH_LO,&o->epoch)||read64(o,TOY_VF_SESSION_LO,&o->session)||
64        owner_read32(o,TOY_VF_STATUS,flags)?-1:0;
65}

67/* 可變長 capability 必須重查；僅使用 pread/pwrite，不 mmap MSI-X 保留頁。 */
68static int region_caps(struct owner *o,unsigned index)
69{
70    struct vfio_region_info first={.argsz=sizeof(first),.index=index};
71    unsigned char *bytes;
72    uint32_t seen[64],count=0,offset;
73    int ret=-1;
74    if (ioctl(o->vfio.device,VFIO_DEVICE_GET_REGION_INFO,&first)) return -1;
75    if (first.argsz<sizeof(first)||first.argsz>65536) { errno=EPROTO; return -1; }
76    bytes=calloc(1,first.argsz); if (!bytes) return -1;
77    struct vfio_region_info *r=(void *)bytes;
78    r->argsz=first.argsz; r->index=index;
79    if (ioctl(o->vfio.device,VFIO_DEVICE_GET_REGION_INFO,r)) goto out;
80    if (r->argsz>first.argsz || r->argsz<sizeof(*r) ||
81        (r->flags&(VFIO_REGION_INFO_FLAG_READ|VFIO_REGION_INFO_FLAG_WRITE))!=
82         (VFIO_REGION_INFO_FLAG_READ|VFIO_REGION_INFO_FLAG_WRITE)) goto invalid;
83    offset=(r->flags&VFIO_REGION_INFO_FLAG_CAPS)?r->cap_offset:0;
84    while (offset) {
85        if (count==64 || (offset&3u) || offset<sizeof(*r) || offset>r->argsz-sizeof(struct vfio_info_cap_header)) goto invalid;
86        for (unsigned i=0;i<count;i++) if (seen[i]==offset) goto invalid;
87        seen[count++]=offset;
88        const struct vfio_info_cap_header *h=(const void *)(bytes+offset);
89        if (h->id==VFIO_REGION_INFO_CAP_SPARSE_MMAP) {
90            const struct vfio_region_info_cap_sparse_mmap *s=(const void *)h;
91            size_t available=r->argsz-offset;
92            if (h->version!=1 || available<sizeof(*s) || s->nr_areas>(available-sizeof(*s))/sizeof(s->areas[0])) goto invalid;
93            for (unsigned i=0;i<s->nr_areas;i++) {
94                uint64_t off=s->areas[i].offset,size=s->areas[i].size;
95                if (!size || off>r->size || size>r->size-off || ((off|size)&(o->vfio.bytes-1))) goto invalid;
96                printf("{\"event\":\"OWNER_SPARSE_AREA\",\"bdf\":\"%s\",\"region\":%u,\"offset\":%" PRIu64 ",\"size\":%" PRIu64 "}\n",o->vfio.bdf,index,off,size);
97            }
98        }
99        offset=h->next;
100    }
101    printf("{\"event\":\"OWNER_REGION\",\"bdf\":\"%s\",\"index\":%u,\"argsz\":%u,\"caps\":%u,\"access\":\"pread_pwrite\"}\n",o->vfio.bdf,index,r->argsz,count);
102    ret=0; goto out;
103invalid: errno=EPROTO;
104out: free(bytes); return ret;
105}
106int owner_open(struct owner *o,const char *bdf,unsigned group)
107{
108    uint32_t flags;
109    if (lab_vfio_open(&o->vfio,bdf,group)) return -1;
110    for (unsigned i=0;i<3;i++)
111        if (region_caps(o,i==0?VFIO_PCI_BAR0_REGION_INDEX:i==1?VFIO_PCI_BAR2_REGION_INDEX:VFIO_PCI_CONFIG_REGION_INDEX)) return -1;
112    if (owner_latch(o,&flags)) return -1;
113    return o->reset_test?owner_reset_clear(o,"open"):0;
114}

116static int iova_allowed(struct owner *o,uint64_t base,uint64_t size)
117{
118    unsigned char bytes[16384] __attribute__((aligned(8)))={0};
119    struct vfio_iommu_type1_info *info=(void *)bytes;
120    uint32_t offset,seen[64],count=0;
121    bool allowed=false;
122    info->argsz=sizeof(bytes);
123    if (ioctl(o->vfio.container,VFIO_IOMMU_GET_INFO,info)) return -1;
124    if (info->argsz<sizeof(*info)||info->argsz>sizeof(bytes)||
125        !(info->flags&VFIO_IOMMU_INFO_PGSIZES)||!(info->iova_pgsizes&o->vfio.bytes)||
126        !(info->flags&VFIO_IOMMU_INFO_CAPS)) goto invalid;
127    offset=info->cap_offset;
128    while (offset) {
129        if (count==64 || (offset&3u) || offset<sizeof(*info) || offset>info->argsz-sizeof(struct vfio_info_cap_header)) goto invalid;
130        for (unsigned i=0;i<count;i++) if (seen[i]==offset) goto invalid;
131        seen[count++]=offset;
132        const struct vfio_info_cap_header *h=(const void *)(bytes+offset);
133        if (h->id==VFIO_IOMMU_TYPE1_INFO_CAP_IOVA_RANGE) {
134            const struct vfio_iommu_type1_info_cap_iova_range *r=(const void *)h;
135            size_t n=info->argsz-offset;
136            if (h->version!=1 || n<sizeof(*r)||r->nr_iovas>(n-sizeof(*r))/sizeof(r->iova_ranges[0])) goto invalid;
137            for (unsigned i=0;i<r->nr_iovas;i++) {
138                uint64_t start=r->iova_ranges[i].start,end=r->iova_ranges[i].end;
139                if (start>end) goto invalid;
140                if (base>=start && base<=end && size && size-1<=end-base) allowed=true;
141            }
142        }
143        offset=h->next;
144    }
145    if (!allowed) { errno=ERANGE; return -1; } return 0;
146invalid: errno=EPROTO; return -1;
147}
148/* vaddr 是 CPU pointer，iova 是裝置位址；READ/WRITE 權限以裝置 DMA 的方向解讀。 */
149static int map_buffers(struct owner *o,uint32_t entries)
150{
151    static const char *const names[]={"owner.map_sq","owner.map_cq","owner.map_payload"};
152    size_t page=o->vfio.bytes;
153    for (unsigned i=0;i<OWNER_MAPS;i++) {
154        struct owner_map *m=&o->maps[i];
155        size_t wanted=i==OWNER_PAYLOAD?(size_t)entries*8192:(size_t)entries*32;
156        struct vfio_iommu_type1_dma_map map={.argsz=sizeof(map)};
157        if (m->memory) { errno=EBUSY; return -1; }
158        m->size=(wanted+page-1)&~(page-1);
159        m->iova=i==OWNER_SQ?0x100000:i==OWNER_CQ?0x200000:0x400000;
160        m->permissions=i==OWNER_SQ?VFIO_DMA_MAP_FLAG_READ:i==OWNER_CQ?VFIO_DMA_MAP_FLAG_WRITE:
161                       VFIO_DMA_MAP_FLAG_READ|VFIO_DMA_MAP_FLAG_WRITE;
162        m->dma_size=m->size;
163        if (i==OWNER_PAYLOAD) {
164            if (o->dma_fault==1) m->dma_size=0;
165            if (o->dma_fault==2) m->dma_size=page;
166            if (o->dma_fault==3) m->permissions=VFIO_DMA_MAP_FLAG_READ;
167            if (o->dma_fault==4) m->permissions=VFIO_DMA_MAP_FLAG_WRITE;
168        }
169        if (i==OWNER_SQ&&o->dma_fault==5) m->permissions=VFIO_DMA_MAP_FLAG_WRITE;
170        if (i==OWNER_CQ&&o->dma_fault==6) m->permissions=VFIO_DMA_MAP_FLAG_READ;
171        if (iova_allowed(o,m->iova,m->size)||fail(o,"owner.alloc_buffer")) return -1;
172        m->memory=mmap(NULL,m->size,PROT_READ|PROT_WRITE,MAP_PRIVATE|MAP_ANONYMOUS,-1,0);
173        if (m->memory==MAP_FAILED) { m->memory=NULL; return -1; }
174        memset(m->memory,i==OWNER_PAYLOAD?(0xa5^o->pattern):0,m->size);
175        if (!m->dma_size) {
176            printf("{\"event\":\"T05_MAP_OMITTED\",\"pid\":%d,\"kind\":%u,\"iova\":%" PRIu64 ",\"size\":%zu}\n",getpid(),i,m->iova,m->size);
177            continue;
178        }
179        map.vaddr=(uintptr_t)m->memory; map.iova=m->iova; map.size=m->dma_size; map.flags=m->permissions;
180        if (fail(o,names[i])||ioctl(o->vfio.container,VFIO_IOMMU_MAP_DMA,&map)) return -1;
181        m->mapped=true;
182        printf("{\"event\":\"OWNER_MAP\",\"pid\":%d,\"bdf\":\"%s\",\"kind\":%u,\"iova\":%" PRIu64 ",\"size\":%zu,\"flags\":%u,\"vaddr\":%" PRIuPTR "}\n",getpid(),o->vfio.bdf,i,m->iova,m->dma_size,m->permissions,(uintptr_t)m->memory);
183    }
184    return 0;
185}
186/* 將兩個 MSI-X vector 接到 eventfd；計數是通知數，完成項目仍須逐筆讀 CQ 驗證。 */
187static int enable_irq(struct owner *o)
188{
189    unsigned char bytes[sizeof(struct vfio_irq_set)+2*sizeof(int)] __attribute__((aligned(8)))={0};
190    struct vfio_irq_set *irq=(void *)bytes;
191    if (o->irq_enabled) return 0;
192    for (unsigned i=0;i<2;i++) {
193        if (fail(o,"owner.eventfd")) return -1;
194        o->eventfds[i]=eventfd(0,EFD_CLOEXEC|EFD_NONBLOCK);
195        if (o->eventfds[i]<0) return -1;
196    }
197    irq->argsz=sizeof(bytes); irq->flags=VFIO_IRQ_SET_DATA_EVENTFD|VFIO_IRQ_SET_ACTION_TRIGGER;
198    irq->index=VFIO_PCI_MSIX_IRQ_INDEX; irq->count=2;
199    memcpy(irq->data,o->eventfds,sizeof(o->eventfds));
200    if (fail(o,"owner.irq_register")||ioctl(o->vfio.device,VFIO_DEVICE_SET_IRQS,irq)) return -1;
201    o->irq_enabled=true; return 0;
202}
203/* 先準備 mapping/IRQ，再提交 queue 設定；此函式只確認 READY_PENDING，不自行執行 PF REARM。 */
204int owner_configure(struct owner *o,uint32_t entries)
205{
206    unsigned char command[2]; uint32_t status,error,flags;
207    if (entries<16||entries>256||(entries&(entries-1))) { errno=EINVAL; return -1; }
208    if (map_buffers(o,entries)||enable_irq(o)) return -1;
209    if (pread(o->vfio.device,command,2,(off_t)o->vfio.config.offset+4)!=2) return -1;
210    lab_put16(command,lab_get16(command)|6u);
211    if (pwrite(o->vfio.device,command,2,(off_t)o->vfio.config.offset+4)!=2 ||
212        pread(o->vfio.device,command,2,(off_t)o->vfio.config.offset+4)!=2 ||
213        (lab_get16(command)&6)!=6) { errno=EIO; return -1; }
214    o->entries=entries; o->sq_tail=o->cq_head=0;
215    if (write64(o,TOY_VF_SQ_BASE_LO,o->maps[OWNER_SQ].iova)||write64(o,TOY_VF_CQ_BASE_LO,o->maps[OWNER_CQ].iova)||
216        owner_write32(o,TOY_VF_SQ_ENTRIES,entries)||owner_write32(o,TOY_VF_CQ_ENTRIES,entries)||
217        write64(o,TOY_VF_CONFIG_EPOCH_LO,o->epoch)||write64(o,TOY_VF_CONFIG_SESSION_LO,o->session)||
218        owner_write32(o,TOY_VF_CONFIG_COMMIT,1)||owner_read32(o,TOY_VF_QUEUE_STATUS,&status)||
219        owner_read32(o,TOY_VF_QUEUE_ERROR,&error)||owner_latch(o,&flags)) return -1;
220    if (error||status!=TOY_QUEUE_STATUS_READY_PENDING||
221        (flags&(TOY_HW_FLAGS_FENCED|TOY_HW_FLAGS_CONFIG_VALID|TOY_HW_FLAGS_READY_PENDING|TOY_HW_FLAGS_BME|TOY_HW_FLAGS_KILLED))!=
222        (TOY_HW_FLAGS_FENCED|TOY_HW_FLAGS_CONFIG_VALID|TOY_HW_FLAGS_READY_PENDING|TOY_HW_FLAGS_BME)) { errno=EPROTO; return -1; }
223    return 0;
224}

228    static const unsigned sizes[]={1,9,4095,4096}; return sizes[(n/2)%4];

230/* 每個 outstanding id 有不同 payload 區；確認完成內容後才移交 CQ_HEAD。 */
231int owner_workload(struct owner *o,unsigned requests)
232{
233    if (o->pending_request) { errno=EBUSY; return -1; }
234    unsigned done=0;
235    uint32_t status;
236    if (owner_read32(o,TOY_VF_QUEUE_STATUS,&status)||status!=TOY_QUEUE_STATUS_RUNNING) { errno=EACCES; return -1; }
237    while (done<requests) {
238        unsigned batch=requests-done<o->entries?requests-done:o->entries;
239        uint32_t start=o->sq_tail;
240        for (unsigned j=0;j<batch;j++) {
241            uint64_t n=o->completed+j;
242            uint32_t index=lab_ring_index(start+j,o->entries);
243            unsigned length=request_length(n);
244            unsigned char *sq=(unsigned char *)o->maps[OWNER_SQ].memory+index*32;
245            unsigned char *payload=(unsigned char *)o->maps[OWNER_PAYLOAD].memory+index*8192;
246            memset(payload,0xa5^o->pattern,8192); memset(sq,0,32);
247            if (n&1) for (unsigned k=0;k<length;k++) payload[k]=(unsigned char)(n+o->pattern+k);
248            if ((n&1)&&length==9) memcpy(payload,"123456789",9);
249            lab_put64(sq,o->maps[OWNER_PAYLOAD].iova+(uint64_t)index*8192);
250            lab_put32(sq+8,length); sq[12]=(n&1)?TOY_SQ_OPS_CRC32:TOY_SQ_OPS_FILL;
251            lab_put16(sq+14,(uint16_t)(start+j)); lab_put64(sq+16,o->epoch); lab_put64(sq+24,(unsigned char)(n+o->pattern));
252        }
253        atomic_thread_fence(memory_order_release);
254        o->sq_tail+=batch;
255        if (owner_write32(o,TOY_VF_SQ_TAIL,o->sq_tail)||owner_write32(o,TOY_VF_DOORBELL,1)) return -1;
256        unsigned received=0;
257        int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
258        while (received<batch) {
259            uint64_t count;
260            if (lab_wait(o->eventfds[0],POLLIN,deadline)) return -1;
261            if (read(o->eventfds[0],&count,sizeof(count))!=sizeof(count)) return -1;
262            o->irq_count+=count; ++o->irq_wakes;
263            uint32_t cause;
264            if (owner_read32(o,TOY_VF_IRQ_CAUSE,&cause)||owner_write32(o,TOY_VF_IRQ_CAUSE,cause)) return -1;
265            while (received<batch) {
266                uint32_t index=lab_ring_index(o->cq_head,o->entries);
267                unsigned char *cq=(unsigned char *)o->maps[OWNER_CQ].memory+index*32;
268                unsigned char phase=__atomic_load_n(cq+24,__ATOMIC_ACQUIRE);
269                unsigned char expected=lab_ring_phase(o->cq_head,o->entries);
270                if (phase!=expected) break;
271                atomic_thread_fence(memory_order_acquire);
272                uint64_t n=o->completed+received;
273                unsigned length=request_length(n);
274                /* stale SQ 不產生 CQ，兩個 ring 的 cursor 可能不同；payload
275                 * 屬於原始 submission slot，不能由 completion slot 推算。 */
276                uint32_t payload_index=lab_ring_index(start+received,o->entries);
277                unsigned char *payload=(unsigned char *)o->maps[OWNER_PAYLOAD].memory+payload_index*8192;
278                uint32_t crc=(n&1)?lab_crc32(payload,length):0;
279                if (lab_get16(cq)!=(uint16_t)(start+received)||lab_get16(cq+2)!=TOY_STATUSES_OK||
280                    lab_get32(cq+4)!=length||lab_get64(cq+8)!=o->epoch||lab_get64(cq+16)!=crc) { errno=EBADMSG; return -1; }
281                if ((n&1)&&length==9&&crc!=0xcbf43926u) { errno=EBADMSG; return -1; }
282                for (unsigned k=0;k<8192;k++) {
283                    unsigned char expected_byte=0xa5^o->pattern;
284                    if (k<length) expected_byte=(n&1)?(length==9?(unsigned char)"123456789"[k]:(unsigned char)(n+o->pattern+k)):(unsigned char)(n+o->pattern);
285                    if (payload[k]!=expected_byte) { errno=EILSEQ; return -1; }
286                }
287                for (unsigned k=25;k<32;k++) if (cq[k]) { errno=EBADMSG; return -1; }
288                ++received; ++o->cq_head;
289            }
290            atomic_thread_fence(memory_order_release);
291            if (owner_write32(o,TOY_VF_CQ_HEAD,o->cq_head)) return -1;
292        }
293        o->completed+=batch; done+=batch;
294    }
295    printf("{\"event\":\"OWNER_WORKLOAD\",\"pid\":%d,\"bdf\":\"%s\",\"entries\":%u,\"completed\":%" PRIu64 ",\"sq_tail\":%u,\"cq_head\":%u,\"irq_wakes\":%" PRIu64 ",\"irq_count\":%" PRIu64 ",\"payload_crc\":%u,\"fixed_crc\":%u}\n",getpid(),o->vfio.bdf,o->entries,o->completed,o->sq_tail,o->cq_head,o->irq_wakes,o->irq_count,lab_crc32(o->maps[OWNER_PAYLOAD].memory,o->maps[OWNER_PAYLOAD].size),lab_crc32("123456789",9));
296    return 0;
297}

299int owner_pattern_pair(struct owner *o)
300{
301    unsigned char expected[4096],*payload=o->maps[OWNER_PAYLOAD].memory;
302    unsigned char pattern=(unsigned char)(0x31+o->pattern);
303    memset(expected,pattern,sizeof(expected));
304    memset(payload,0xa5^o->pattern,8192);
305    uint32_t expected_crc=lab_crc32(expected,sizeof(expected));
306    /* 先讓裝置 FILL，再讀同一 IOVA 算 CRC；中間不以 CPU 重寫 payload。 */
307    for (unsigned step=0;step<2;step++) {
308        uint32_t index=lab_ring_index(o->sq_tail,o->entries),cause;
309        unsigned char *sq=(unsigned char *)o->maps[OWNER_SQ].memory+index*32;
310        unsigned char *cq=(unsigned char *)o->maps[OWNER_CQ].memory+lab_ring_index(o->cq_head,o->entries)*32;
311        memset(sq,0,32); lab_put64(sq,o->maps[OWNER_PAYLOAD].iova); lab_put32(sq+8,sizeof(expected));
312        sq[12]=step?TOY_SQ_OPS_CRC32:TOY_SQ_OPS_FILL; lab_put16(sq+14,(uint16_t)o->sq_tail);
313        lab_put64(sq+16,o->epoch); lab_put64(sq+24,pattern);
314        atomic_thread_fence(memory_order_release);
315        ++o->sq_tail;
316        if (owner_write32(o,TOY_VF_SQ_TAIL,o->sq_tail)||owner_write32(o,TOY_VF_DOORBELL,1)) return -1;
317        int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
318        do {
319            uint64_t count;
320            if (lab_wait(o->eventfds[0],POLLIN,deadline)||read(o->eventfds[0],&count,sizeof(count))!=sizeof(count)) return -1;
321            o->irq_count+=count; ++o->irq_wakes;
322            if (owner_read32(o,TOY_VF_IRQ_CAUSE,&cause)||owner_write32(o,TOY_VF_IRQ_CAUSE,cause)) return -1;
323        } while (__atomic_load_n(cq+24,__ATOMIC_ACQUIRE)!=lab_ring_phase(o->cq_head,o->entries));
324        atomic_thread_fence(memory_order_acquire);
325        if (lab_get16(cq)!=(uint16_t)(o->sq_tail-1)||lab_get16(cq+2)!=TOY_STATUSES_OK||
326            lab_get32(cq+4)!=sizeof(expected)||lab_get64(cq+8)!=o->epoch||lab_get64(cq+16)!=(step?expected_crc:0)||
327            memcmp(payload,expected,sizeof(expected))) { errno=EBADMSG; return -1; }
328        for (unsigned k=sizeof(expected);k<8192;k++) if (payload[k]!=(unsigned char)(0xa5^o->pattern)) { errno=EILSEQ; return -1; }
329        for (unsigned k=25;k<32;k++) if (cq[k]) { errno=EBADMSG; return -1; }
330        ++o->cq_head; ++o->completed;
331        if (owner_write32(o,TOY_VF_CQ_HEAD,o->cq_head)) return -1;
332        printf("{\"event\":\"OWNER_PATTERN_COMMAND\",\"pid\":%d,\"step\":%u,\"iova\":%" PRIu64 ",\"bytes\":%zu,\"pattern\":%u,\"result\":%" PRIu64 ",\"expected_crc\":%u}\n",getpid(),step,o->maps[OWNER_PAYLOAD].iova,sizeof(expected),pattern,lab_get64(cq+16),expected_crc);
333    }
334    printf("{\"event\":\"OWNER_PATTERN_PAIR\",\"pid\":%d,\"bdf\":\"%s\",\"completed\":%" PRIu64 ",\"payload_crc\":%u,\"sq_tail\":%u,\"cq_head\":%u}\n",getpid(),o->vfio.bdf,o->completed,lab_crc32(payload,o->maps[OWNER_PAYLOAD].size),o->sq_tail,o->cq_head);
335    return 0;
336}

338int owner_descriptor_vectors(struct owner *o)
339{
340    static const char *const names[]={"old-epoch","unknown-opcode","zero-length","length-over-4096","address-overflow","nonzero-flags"};
341    for (unsigned variant=0;variant<6;variant++) {
342        unsigned char *sq=(unsigned char *)o->maps[OWNER_SQ].memory+lab_ring_index(o->sq_tail,o->entries)*32;
343        unsigned char *cq=(unsigned char *)o->maps[OWNER_CQ].memory+lab_ring_index(o->cq_head,o->entries)*32;
344        uint32_t old_cq_tail,head,tail,cause;
345        uint32_t id=o->sq_tail;
346        char hex[65],completion_hex[65]={0};
347        if (owner_read32(o,TOY_VF_CQ_TAIL,&old_cq_tail)||old_cq_tail!=o->cq_head) { errno=EPROTO; return -1; }
348        memset(sq,0,32); lab_put64(sq,o->maps[OWNER_PAYLOAD].iova); lab_put32(sq+8,8);
349        sq[12]=TOY_SQ_OPS_FILL; lab_put16(sq+14,(uint16_t)id); lab_put64(sq+16,o->epoch); lab_put64(sq+24,0x73);
350        switch (variant) {
351        case 0: lab_put64(sq+16,o->epoch-1); break;
352        case 1: sq[12]=0xff; break;
353        case 2: lab_put32(sq+8,0); break;
354        case 3: lab_put32(sq+8,4097); break;
355        case 4: lab_put64(sq,UINT64_MAX-1024); lab_put32(sq+8,2048); break;
356        case 5: sq[13]=1; break;
357        }
358        for (unsigned i=0;i<32;i++) snprintf(hex+i*2,3,"%02x",sq[i]);
359        if (owner_freeze(o)) return -1;
360        uint64_t before_irq=o->irq_count;
361        atomic_thread_fence(memory_order_release);
362        ++o->sq_tail;
363        if (owner_write32(o,TOY_VF_SQ_TAIL,o->sq_tail)||owner_write32(o,TOY_VF_DOORBELL,1)) return -1;
364        int64_t deadline=lab_now_ms()+LAB_TIMEOUT_MS;
365        do {
366            if (owner_read32(o,TOY_VF_SQ_HEAD,&head)) return -1;
367            if (head==o->sq_tail) break;
368            if (lab_now_ms()>=deadline) { errno=ETIMEDOUT; return -1; }
369            usleep(1000);
370        } while (true);
371        if (!variant) {
372            uint32_t digests[OWNER_MAPS];
373            struct pollfd irq={.fd=o->eventfds[0],.events=POLLIN};
374            int ready;
375            do { ready=poll(&irq,1,20); } while (ready<0&&errno==EINTR);
376            if (ready!=0||owner_read32(o,TOY_VF_CQ_TAIL,&tail)||tail!=old_cq_tail||owner_unchanged(o,digests)) { errno=EPROTO; return -1; }
377        } else {
378            do {
379                uint64_t count;
380                if (lab_wait(o->eventfds[0],POLLIN,deadline)||read(o->eventfds[0],&count,sizeof(count))!=sizeof(count)) return -1;
381                o->irq_count+=count; ++o->irq_wakes;
382                if (owner_read32(o,TOY_VF_IRQ_CAUSE,&cause)||owner_write32(o,TOY_VF_IRQ_CAUSE,cause)) return -1;
383            } while (__atomic_load_n(cq+24,__ATOMIC_ACQUIRE)!=lab_ring_phase(o->cq_head,o->entries));
384            atomic_thread_fence(memory_order_acquire);
385            if (lab_get16(cq)!=(uint16_t)id||lab_get16(cq+2)!=TOY_STATUSES_BAD_DESC||lab_get32(cq+4)||
386                lab_get64(cq+8)!=o->epoch||lab_get64(cq+16)) { errno=EBADMSG; return -1; }
387            for (unsigned k=25;k<32;k++) if (cq[k]) { errno=EBADMSG; return -1; }
388            for (unsigned kind=0;kind<OWNER_MAPS;kind++) {
389                struct owner_map *map=&o->maps[kind];
390                unsigned char *actual=map->memory,*saved=o->held[kind];
391                for (size_t b=0;b<map->size;b++) {
392                    if (kind==OWNER_CQ&&actual+b>=cq&&actual+b<cq+32) continue;
393                    if (actual[b]!=saved[b]) { errno=EILSEQ; return -1; }
394                }
395            }
396            ++o->cq_head; ++o->completed;
397            for (unsigned k=0;k<32;k++) snprintf(completion_hex+k*2,3,"%02x",cq[k]);
398            if (owner_write32(o,TOY_VF_CQ_HEAD,o->cq_head)||owner_read32(o,TOY_VF_CQ_TAIL,&tail)||tail!=o->cq_head) { errno=EPROTO; return -1; }
399        }
400        printf("{\"event\":\"T09_VECTOR\",\"pid\":%d,\"name\":\"%s\",\"descriptor_hex\":\"%s\",\"completion_hex\":\"%s\",\"live_epoch\":%" PRIu64 ",\"sq_head\":%u,\"sq_tail\":%u,\"cq_before\":%u,\"cq_after\":%u,\"irq_before\":%" PRIu64 ",\"irq_after\":%" PRIu64 ",\"completion_status\":%d,\"payload_unchanged\":true,\"other_cq_bytes_unchanged\":true}\n",getpid(),names[variant],hex,completion_hex,o->epoch,head,o->sq_tail,old_cq_tail,tail,before_irq,o->irq_count,variant?(int)lab_get16(cq+2):-1);
401    }
402    return 0;
403}

405int owner_freeze(struct owner *o)
406{
407    for (unsigned i=0;i<OWNER_MAPS;i++) {
408        struct owner_map *m=&o->maps[i];
409        /* CPU guard 也涵蓋刻意未提供 DMA 映射的已配置記憶體。 */
410        if (!m->memory) { errno=EINVAL; return -1; }
411        if (!o->held[i]) { o->held[i]=malloc(m->size); if (!o->held[i]) return -1; }
412        memcpy(o->held[i],m->memory,m->size);
413    }
414    return 0;
415}
416int owner_unchanged(struct owner *o,uint32_t digests[OWNER_MAPS])
417{
418    for (unsigned i=0;i<OWNER_MAPS;i++) {
419        struct owner_map *m=&o->maps[i];
420        if (!o->held[i]||!m->memory||memcmp(o->held[i],m->memory,m->size)) { errno=EILSEQ; return -1; }
421        digests[i]=lab_crc32(m->memory,m->size);
422        printf("{\"event\":\"OWNER_HELD_CHECK\",\"pid\":%d,\"bdf\":\"%s\",\"kind\":%u,\"completed\":%" PRIu64 ",\"bytes\":%zu,\"pattern\":%u,\"crc\":%u,\"bytewise_unchanged\":true}\n",getpid(),o->vfio.bdf,i,o->completed,m->size,o->pattern,digests[i]);
423    }
424    if (o->recovery_case) {
425        uint64_t count;
426        errno=0;
427        if (read(o->eventfds[0],&count,sizeof(count))!=-1||errno!=EAGAIN) { errno=EPROTO; return -1; }
428        printf("{\"event\":\"RECOVERY_NO_IRQ\",\"pid\":%d,\"epoch\":%" PRIu64 ",\"completed\":%" PRIu64 ",\"eventfd_errno\":%d}\n",getpid(),o->epoch,o->completed,EAGAIN);
429    }
430    return owner_retired_unchanged(o);
431}
432/* 呼叫者先證明 fence 生效；每段 UNMAP 都核對實際長度，失敗的 allocation 保留給 close 收尾。 */
433int owner_unmap(struct owner *o,bool fence_verified)
434{
435    int failed=0;
436    if (!fence_verified) { errno=EACCES; return -1; }
437    for (unsigned i=OWNER_MAPS;i>0;i--) {
438        struct owner_map *m=&o->maps[i-1];
439        if (m->mapped) {
440            struct vfio_iommu_type1_dma_unmap unmap={.argsz=sizeof(unmap),.iova=m->iova,.size=m->dma_size};
441            /* 要求包含相鄰未映射範圍，取得真實 TYPE1 短 UNMAP 回傳。 */
442            bool short_test=fail(o,"owner.short_unmap");
443            if (short_test) unmap.size*=2;
444            uint64_t requested=unmap.size;
445            bool injected=fail(o,"owner.unmap");
446            int rc=injected?-1:ioctl(o->vfio.container,VFIO_IOMMU_UNMAP_DMA,&unmap);
447            if (rc||unmap.size!=requested) {
448                printf("{\"event\":\"OWNER_UNMAP_QUARANTINE\",\"pid\":%d,\"kind\":%u,\"requested\":%" PRIu64 ",\"returned\":%" PRIu64 ",\"ioctl_executed\":%s,\"ioctl_errno\":%d,\"buffer_held\":true}\n",getpid(),i-1,requested,(uint64_t)unmap.size,injected?"false":"true",rc?errno:0);
449                failed=1; o->cleanup_failed=true; continue;
450            }
451            m->mapped=false;
452            printf("{\"event\":\"OWNER_UNMAP\",\"pid\":%d,\"kind\":%u,\"requested\":%zu,\"returned\":%" PRIu64 "}\n",getpid(),i-1,m->dma_size,(uint64_t)unmap.size);
453        }
454        if (m->memory&&!o->keep_unmapped) {
455            if (munmap(m->memory,m->size)) { failed=1; continue; }
456            m->memory=NULL;
457        }
458    }
459    return failed?-1:0;
460}
461int owner_close(struct owner *o,bool fence_verified)
462{
463    int failed=0;
464    if (o->irq_enabled) {
465        struct vfio_irq_set irq={.argsz=sizeof(irq),.flags=VFIO_IRQ_SET_DATA_NONE|VFIO_IRQ_SET_ACTION_TRIGGER,.index=VFIO_PCI_MSIX_IRQ_INDEX};
466        if (ioctl(o->vfio.device,VFIO_DEVICE_SET_IRQS,&irq)) failed=1;
467        else o->irq_enabled=false;
468    }
469    if (fence_verified && !o->cleanup_failed && owner_unmap(o,true)) failed=1;
470    /* 未確認 fence 或短 UNMAP：保留記憶體到全部 owner fd 關閉之後。 */
471    if (o->vfio.device>=0) { if (close(o->vfio.device)) failed=1; o->vfio.device=-1; }
472    for (unsigned i=0;i<2;i++) if (o->eventfds[i]>=0) { if (close(o->eventfds[i])) failed=1; o->eventfds[i]=-1; }
473    if (o->vfio.attached) { if (ioctl(o->vfio.group,VFIO_GROUP_UNSET_CONTAINER)) failed=1; o->vfio.attached=0; }
474    if (o->vfio.group>=0) { if (close(o->vfio.group)) failed=1; o->vfio.group=-1; }
475    if (o->vfio.container>=0) { if (close(o->vfio.container)) failed=1; o->vfio.container=-1; }
476    for (unsigned i=0;i<OWNER_MAPS;i++) {
477        struct owner_map *m=&o->maps[i]; m->mapped=false;
478        if (m->memory) { if (munmap(m->memory,m->size)) failed=1; else m->memory=NULL; }
479        free(o->held[i]); o->held[i]=NULL;
480        if (o->retired[i].memory) {
481            if (munmap(o->retired[i].memory,o->retired[i].size)) failed=1;
482            else o->retired[i].memory=NULL;
483        }
484        free(o->retired_guard[i]); o->retired_guard[i]=NULL;
485    }
486    o->irq_enabled=false;
487    return failed||o->cleanup_failed?-1:0;
488}
